double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double total_weight = 0;
    double total_capacity = 0;
    double min_ratio = 1e9;

    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item];
        total_capacity += capacity[j];
        double ratio = weights[j][index_item] / capacity[j];
        if (ratio < min_ratio) {
            min_ratio = ratio;
        }
    }

    double average_ratio = total_weight / total_capacity;
    double weighted_profit = profit[index_item] * (1 - min_ratio);
    double neighborhood_score = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        neighborhood_score += profit[voisinage[i]];
    }
    neighborhood_score /= nb_voisinage;

    double heuristic_value = weighted_profit * average_ratio + neighborhood_score * (1 - average_ratio);
    if (heuristic_value < 0) {
        heuristic_value = 0; // Sparsify the heuristic by setting unpromising elements to zero
    }

    return heuristic_value;
}
