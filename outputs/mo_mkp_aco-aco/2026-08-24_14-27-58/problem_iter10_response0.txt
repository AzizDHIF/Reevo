double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double promising_factor = 0;
    int neighbor_index = voisinage[index_item];

    double remaining_capacity = 0;
    for (int j = 0; j < dimension; j++) {
        remaining_capacity += capacity[j];
    }

    if (remaining_capacity > 0) {
        double profit_density = profit[neighbor_index] / remaining_capacity;
        double weight_ratio = weights[0][neighbor_index] / weights[1][neighbor_index];
        promising_factor = profit_density * (1 - weight_ratio); // Adjusting based on specific weights
        return promising_factor > 0 ? promising_factor : 0; // Sparsify the heuristic
    } else {
        return 0;
    }
}
