double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double h = 0;
    for (int j = 0; j < dimension; j++)
    {
        h += weights[j][voisinage[index_item]] / capacity[j];
    }

    double value = profit[voisinage[index_item]] / h; // original value

    // Additional factors for determining promise
    double weight_factor = 0.5;
    double profit_factor = 0.3;
    double overweight_factor = 0.2;

    // Penalizing the value for weight exceeded capacity
    for (int j = 0; j < dimension; j++)
    {
        if (weights[j][voisinage[index_item]] > capacity[j])
        {
            value *= overweight_factor;
        }
    }

    // Combine multiple factors
    value = value * weight_factor + profit[voisinage[index_item]] * profit_factor;

    // Setting unpromising elements to zero
    if (value < 0)
    {
        value = 0;
    }

    return value;
}
