double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] )
{
    double h = 0;
    double promising_factor = 0;
    int neighbor_index = voisinage[index_item];
    
    for (int j = 0; j < dimension; j++) {
        h += weights[j][neighbor_index] / capacity[j];
    }

    if (h > 0) {
        promising_factor = profit[neighbor_index] / h;
        promising_factor *= 1 - weights[0][neighbor_index] / weights[1][neighbor_index]; // Adjusting based on specific weights
        return promising_factor > 0 ? promising_factor : 0; // Sparsify the heuristic
    } else {
        return 0;
    }
}
