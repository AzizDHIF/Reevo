double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double promising_factor = 0;

    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j];
    }

    if (h > 0) {
        promising_factor = profit[voisinage[index_item]] / h;
        promising_factor *= 1 - weights[0][voisinage[index_item]] / weights[1][voisinage[index_item]]; // Adjusting based on specific weights
        return promising_factor > 0 ? promising_factor : 0; // Sparsify the heuristic
    } else {
        return 0;
    }
}
