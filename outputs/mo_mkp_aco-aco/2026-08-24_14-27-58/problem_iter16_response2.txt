double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double promising_factor = 0;
    double weight_sum = 0;

    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][voisinage[index_item]];
    }

    double remaining_capacity = 0;
    for (int j = 0; j < dimension; j++) {
        if (j != index_item) {
            remaining_capacity += capacity[j];
        }
    }

    if (weight_sum > 0 && remaining_capacity > 0) {
        double weight_ratio = 0;
        for (int j = 0; j < dimension; j++) {
            weight_ratio += weights[j][voisinage[index_item]] / weight_sum;
        }

        promising_factor = profit[voisinage[index_item]] * (1 - weight_ratio) * remaining_capacity;

        return promising_factor > 0 ? promising_factor : 0;
    } else {
        return 0;
    }
}
