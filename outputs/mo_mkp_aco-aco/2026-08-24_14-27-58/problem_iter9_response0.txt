double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double promising_factor = 0;
    
    double weight_ratio = weights[0][voisinage[index_item]] / weights[1][voisinage[index_item]];
    
    if (weight_ratio == 0) {
        return 0;
    }
    
    promising_factor = profit[voisinage[index_item]] / weight_ratio;
    
    return promising_factor > 0 ? promising_factor : 0;
}
