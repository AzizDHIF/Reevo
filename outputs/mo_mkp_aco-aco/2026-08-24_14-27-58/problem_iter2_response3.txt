double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_factor = 0.3, profit_factor = 0.7;
    double h = 0;
    
    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / (capacity[j] + 1e-6); // Adding a small value to avoid division by zero
    }
    
    return (weight_factor * (1.0 / (h + 1e-6))) + (profit_factor * profit[voisinage[index_item]]);
}
