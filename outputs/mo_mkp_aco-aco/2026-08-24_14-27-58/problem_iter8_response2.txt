double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double promising_factor = 0;
    double weight_sum = 0;

    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][voisinage[index_item]];
    }

    if (weight_sum > 0) {
        promising_factor = profit[voisinage[index_item]] / weight_sum;
        double weight_ratio = weights[0][voisinage[index_item]] / weights[1][voisinage[index_item]];
        promising_factor *= 1 - weight_ratio;

        return promising_factor > 0 ? promising_factor : 0;
    } else {
        return 0;
    }
}
