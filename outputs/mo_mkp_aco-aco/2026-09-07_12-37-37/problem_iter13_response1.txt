double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double sum_weights = 0.00001; // Initialize to a small value to prevent division by zero
    for (int j = 0; j < dimension; j++) {
        sum_weights += weights[j][voisinage[index_item]]; // Accumulate the sum of weights without dividing by capacity
    }
    return profit[voisinage[index_item]] / sum_weights; // Return profit divided by the sum of weights
}
