double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0.00001; // Initialize to a small value for stability
    double combined_factor = 0.0001; // Small non-zero initialization for impact balancing

    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]]; // Accumulate the sum of weights without division
        combined_factor *= weights[j][voisinage[index_item]] * profit[voisinage[index_item]];
    }

    return combined_factor / h; // Return a combined factor divided by the sum of weights
}
