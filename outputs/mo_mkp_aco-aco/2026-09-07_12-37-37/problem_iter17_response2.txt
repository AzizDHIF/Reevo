double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0.00001; // Initialize to a small value to prevent division by zero
    for (int j = 0; j < nb_voisinage; j++) {
        for (int k = 0; k < dimension; k++) {
            h += weights[k][voisinage[j]]; // Accumulate the sum of weights without dividing by capacity
        }
    }
    return profit[voisinage[index_item]] / h; // Return profit divided by the sum of weights
}
