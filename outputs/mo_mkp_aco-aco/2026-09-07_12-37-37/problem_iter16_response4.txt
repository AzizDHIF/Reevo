double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double sum_weight = 0.00001; // Initialize to a small value to prevent division by zero
    for (int j = 0; j < nb_voisinage; j++) {
        sum_weight += weights[j][voisinage[j]]; // Sum only the weights of neighbors
    }
    return profit[voisinage[index_item]] / sum_weight; // Return profit divided by the sum of neighbor weights
}
