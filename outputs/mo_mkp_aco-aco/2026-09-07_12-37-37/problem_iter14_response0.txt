double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] )
{
    double h = 0.00001; // Initialize to a small value to prevent division by zero
    for (int j = 0; j < dimension; j++) {
        h += weights[j][index_item]; // Accumulate the sum of weights of the current item
    }
    return profit[index_item] / h; // Return profit divided by the sum of weights of the current item
}
