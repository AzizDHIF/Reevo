double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 1.0; // Initialize to 1 for proportional weight impact
    for (int j = 0; j < dimension; j++) {
        h *= (1.0 - weights[j][voisinage[index_item]] / capacity[j]); // Update h based on remaining capacity
    }
    return profit[voisinage[index_item]] * h; // Return profit multiplied by capacity factor
}
