#include <math.h>

double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double h=0;
    double weight_sum = 0;
    
    for(int j=0;j<dimension;j++){
        weight_sum += weights[j][voisinage[index_item]];
    	if (capacity[j] > 0){
			h += weights[j][voisinage[index_item]] / capacity[j];
		}		    
    }
    
    double max_profit = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (profit[voisinage[i]] > max_profit) {
            max_profit = profit[voisinage[i]];
        }
    }

    if (weight_sum == 0) {
        return 0;
    } 
    
    return ((max_profit - profit[voisinage[index_item]]) / weight_sum) + h;
}
