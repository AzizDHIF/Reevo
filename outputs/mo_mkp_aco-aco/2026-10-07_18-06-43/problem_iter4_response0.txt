double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double normalized_weight_sum = 0;
    
    for (int j = 0; j < dimension; j++) {
        normalized_weight_sum += weights[j][voisinage[index_item]] / capacity[j];
    }
    
    return profit[voisinage[index_item]] / normalized_weight_sum;
}
