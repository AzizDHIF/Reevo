double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double promising_factor = 0;
    
    // Factor 1: Measure of capacity utilization
    double capacity_utilization = 0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            capacity_utilization += weights[j][voisinage[index_item]] / capacity[j];
        }
    }
    
    // Factor 2: Importance of the item based on profit
    promising_factor += profit[voisinage[index_item]];
    
    // Factor 3: Structure of the neighborhood considering neighborhood importance
    promising_factor *= 1 + (1.0 / (nb_voisinage - index_item)); // Encourage exploration of different neighborhoods
    
    return promising_factor;
}
