double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h1 = 0;
    double h2 = 0;
    double h3 = 0;

    // Calculate the total weight of the item's neighbors
    for (int i = 0; i < nb_voisinage; i++) {
        for (int j = 0; j < dimension; j++) {
            h1 += weights[j][voisinage[i]];
        }
    }

    // Calculate the total profit of the item's neighbors
    for (int i = 0; i < nb_voisinage; i++) {
        h2 += profit[voisinage[i]];
    }

    // Calculate the minimum remaining capacity for each dimension
    for (int j = 0; j < dimension; j++) {
        double min_cap = capacity[j];
        for (int i = 0; i < nb_voisinage; i++) {
            min_cap = fmin(min_cap, capacity[j] - weights[j][voisinage[i]]);
        }
        h3 += min_cap;
    }

    // Combine the factors
    double h = (h2 / h1) * (h3 / dimension);

    // Sparsify the heuristic
    if (h < 0.1 || h > 10) {
        h = 0;
    }

    return h;
}
