double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double h = 0;
    double sum_weights = 0;
    double sum_profits = 0;
    double min_capacity = DBL_MAX;

    for (int j = 0; j < dimension; j++) {
        // Calculate a weighted sum of the weights of the neighbors of the current item
        sum_weights += weights[j][index_item];
        
        // Calculate the minimum capacity across all dimensions for the item
        min_capacity = (min_capacity < capacity[j] / weights[j][index_item]) ? min_capacity : capacity[j] / weights[j][index_item];
    }

    // Calculate a weighted sum of the profits of the neighbors of the current item
    for (int k = 0; k < nb_voisinage; k++) {
        sum_profits += profit[voisinage[k]];
    }

    // Sparsify the heuristic by setting unpromising elements to zero
    if (min_capacity <= 0 || sum_profits <= 0) {
        return 0;
    }

    // Calculate a score based on the profit, weight, and capacity of the item
    h = profit[index_item] / (sum_weights / min_capacity + 1);

    // Add a component to the heuristic that takes into account the profit and weight of the neighbors of the item
    double nb_voisinage_double = (double)nb_voisinage;
    if (nb_voisinage_double > 0) {
        h += (sum_profits / nb_voisinage_double) / (sum_weights / min_capacity + 1);
    }

    // Return the final heuristic score
    return h;
}
