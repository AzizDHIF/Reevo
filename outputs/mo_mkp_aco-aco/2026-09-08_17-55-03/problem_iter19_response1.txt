double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_factor = 0.3;
    double profit_factor = 0.5;
    double deviation_factor = 0.2;

    double max_profit = 0;
    double total_weight = 0;
    double average_profit = 0;

    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][voisinage[index_item]] / capacity[j];
    }

    for (int i = 0; i < nb_voisinage; i++) {
        int neighbour = voisinage[i];
        double neighbour_profit = profit[neighbour];
        double neighbour_utility = 0;

        for (int j = 0; j < dimension; j++) {
            neighbour_utility += weights[j][neighbour] / capacity[j];
        }

        if (neighbour_profit > max_profit) {
            max_profit = neighbour_profit;
        }

        average_profit += neighbour_profit;
    }

    average_profit /= nb_voisinage;

    double promising_score = (max_profit * profit_factor) / (total_weight * weight_factor) + deviation_factor * (max_profit - average_profit);

    if (promising_score < 0.1) {
        return 0;
    }

    return promising_score;
}
