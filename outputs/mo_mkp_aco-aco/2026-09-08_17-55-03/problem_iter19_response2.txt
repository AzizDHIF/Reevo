double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_factor = 0.3;
    double profit_factor = 0.5;
    double deviation_factor = 0.2;

    double utility = 0;
    for (int j = 0; j < dimension; j++) {
        utility += weights[j][voisinage[index_item]] / capacity[j];
    }

    double promising_score = (profit[voisinage[index_item]] * profit_factor) / (utility * weight_factor) + deviation_factor;

    // Aggressively filtering solutions by adjusting threshold
    if (promising_score < 0.2) {
        return 0;
    }

    // Dynamically fine-tune factors to balance trade-offs
    promising_score *= 1.2;

    return promising_score;
}
