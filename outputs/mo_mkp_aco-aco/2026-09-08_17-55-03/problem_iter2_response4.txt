double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_factor = 0.5;
    double profit_factor = 0.5;
    double deviation_factor = 0.001;

    double h = 0;
    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j];
    }

    double promising_score = (profit[voisinage[index_item]] * profit_factor) / (h * weight_factor) + deviation_factor;

    // Sparsifying the heuristic by setting unpromising elements to zero
    if (promising_score < 0.1) {
        return 0;
    }

    return promising_score;
}
