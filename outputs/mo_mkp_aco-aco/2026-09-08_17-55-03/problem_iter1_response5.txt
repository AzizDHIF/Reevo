double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double h = 0;
    for(int j=0; j<dimension; j++){
        h += weights[j][index_item] / capacity[j]; 
    }
    
    double feasibility_factor = 1;
    for(int j=0; j<dimension; j++){
        if(weights[j][index_item] <= capacity[j]){
            feasibility_factor *= 0.5; // Penalize features with better feasibility setting
        }
    }
    
    double diversity_factor = 0.5;
    double distance_factor = 1.0 / nb_voisinage; // Penalize neighbors
    
    return profit[index_item] / h * feasibility_factor * diversity_factor * distance_factor;
}
