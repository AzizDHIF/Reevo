double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] )
{
    double weight_factor = 0.3;
    double profit_factor = 0.5;
    double deviation_factor = 0.2;

    double utility = 0;
    for (int j = 0; j < dimension; j++) {
        utility += weights[j][voisinage[index_item]] / capacity[j];
    }

    double deviation = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        deviation += abs(profit[voisinage[i]] - profit[index_item]);
    }

    double promising_score = (profit[index_item] * profit_factor) / (utility * weight_factor) + deviation * deviation_factor;

    if (promising_score < 0.1) {
        return 0;
    }

    return promising_score;
}
