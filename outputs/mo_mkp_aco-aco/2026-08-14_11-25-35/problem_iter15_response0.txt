double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Simplify density calculation
    double density = 0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 1e-6) {
            density += weights[j][index_item] / capacity[j];
        } else {
            density += 1e6; // penalize if capacity is too small
        }
    }
    density /= dimension;

    // Calculate the remaining capacity after including the item
    double remaining_capacity = 1;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] - weights[j][index_item] > 1e-6) {
            remaining_capacity *= (capacity[j] - weights[j][index_item]) / capacity[j];
        } else {
            remaining_capacity = 0; // item cannot fit, discard
        }
    }

    // Calculate the average profit of the neighborhood
    double sum_profit_voisinage = 0;
    double count_voisinage = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (index_item != voisinage[i]) {
            sum_profit_voisinage += profit[voisinage[i]];
            count_voisinage++;
        }
    }
    double average_profit_voisinage = (count_voisinage > 0) ? sum_profit_voisinage / count_voisinage : 0;

    // Calculate the heuristic value
    double h = 0;
    if (remaining_capacity > 1e-6 && density < 1e6) {
        h = profit[index_item] / (density + 1e-6);
        h *= (1 + (average_profit_voisinage / (profit[index_item] + 1e-6)));
        h *= (remaining_capacity + 1e-6);
    }

    // Sparsify the heuristic, consider feasibility first
    if (h < 0.01 || remaining_capacity < 1e-6) {
        return 0;
    }

    return h;
}
