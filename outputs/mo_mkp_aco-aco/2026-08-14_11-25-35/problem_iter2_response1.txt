double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate the density of the current item
    double density = 0;
    for (int j = 0; j < dimension; j++) {
        density += weights[j][index_item] / capacity[j];
    }
    density /= dimension;

    // Calculate the average profit of the neighboring items
    double sum_profit_voisinage = 0;
    double count_voisinage = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (index_item != voisinage[i]) {
            sum_profit_voisinage += profit[voisinage[i]];
            count_voisinage++;
        }
    }
    double average_profit_voisinage = (count_voisinage > 0) ? sum_profit_voisinage / count_voisinage : 0;

    // Calculate the item's profit-to-density ratio
    double profit_to_density = profit[index_item] / (density + 1e-6);

    // Calculate the neighborhood bonus
    double neighborhood_bonus = (average_profit_voisinage / (profit[index_item] + 1e-6));

    // Combine the factors to determine how promising it is to select an item
    double h = profit_to_density * (1 + neighborhood_bonus);

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 0.01) {
        return 0;
    }

    return h;
}
