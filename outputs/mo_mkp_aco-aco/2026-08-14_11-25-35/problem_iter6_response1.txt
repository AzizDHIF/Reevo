double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate the density of the item, considering all dimensions
    double density = 0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 1e-6) {
            density += weights[j][index_item] / capacity[j];
        }
    }

    // Prioritize local properties: profit and density
    double local_priority = profit[index_item] / (density + 1e-6);

    // Calculate the remaining capacity after including the item
    double remaining_capacity = 1;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > weights[j][index_item] + 1e-6) {
            remaining_capacity *= (capacity[j] - weights[j][index_item]) / capacity[j];
        } else {
            remaining_capacity = 0;
        }
    }

    // Calculate the average profit of the neighborhood
    double sum_profit_voisinage = 0;
    double count_voisinage = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (index_item != voisinage[i]) {
            sum_profit_voisinage += profit[voisinage[i]];
            count_voisinage++;
        }
    }
    double average_profit_voisinage = (count_voisinage > 0) ? sum_profit_voisinage / count_voisinage : 0;

    // Consider neighborhood information with a lower priority, but prioritize items with high density and profit
    double neighborhood_contribution = (average_profit_voisinage / (profit[index_item] + 1e-6)) * (density / (density + 1e-6));

    // Calculate the heuristic value, balancing local properties and neighborhood effects
    double h = local_priority * (1 + neighborhood_contribution) * remaining_capacity;

    // Sparsify the heuristic, focusing on promising items
    if (h < 0.01 || remaining_capacity < 1e-6) {
        return 0;
    }

    return h;
}
