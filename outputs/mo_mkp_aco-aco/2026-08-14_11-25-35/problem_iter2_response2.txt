double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate the total weight of the item
    double total_weight = 0;
    double max_weight = 0;
    for(int j = 0; j < dimension; j++){
        total_weight += weights[j][index_item];
        max_weight = (max_weight > weights[j][index_item]) ? max_weight : weights[j][index_item];
    }

    // Calculate the ratio of the profit to the total weight
    double ratio = profit[index_item] / (total_weight + 1e-6);

    // Calculate the capacity utilization for each dimension
    double capacity_utilization = 0;
    for(int j = 0; j < dimension; j++){
        capacity_utilization += (weights[j][index_item] / capacity[j]);
    }
    capacity_utilization /= dimension;

    // Calculate the heuristic value based on local properties
    double h = ratio * (1 - capacity_utilization);

    // Consider neighborhood information with a lower priority
    if (nb_voisinage > 0) {
        double neighborhood_profit = 0;
        for (int i = 0; i < nb_voisinage; i++) {
            neighborhood_profit += profit[voisinage[i]];
        }
        double neighborhood_contribution = neighborhood_profit / (total_weight * nb_voisinage);
        h += 0.1 * neighborhood_contribution * ratio;
    }

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 1e-6) {
        h = 0;
    }

    return h;
}
