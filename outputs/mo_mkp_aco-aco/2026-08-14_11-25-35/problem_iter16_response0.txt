double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Prioritize feasibility
    double remaining_capacity = 1.0;
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > capacity[j]) {
            return 0.0; // item does not fit, no need to further calculate
        }
        remaining_capacity *= (capacity[j] - weights[j][index_item]) / capacity[j];
    }

    // Simplified density and average profit calculations
    double density = 0.0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            density += weights[j][index_item] / capacity[j];
        }
    }

    double sum_profit_voisinage = 0.0;
    double count_voisinage = 0.0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (index_item != voisinage[i]) {
            sum_profit_voisinage += profit[voisinage[i]];
            count_voisinage++;
        }
    }
    double average_profit_voisinage = (count_voisinage > 0) ? sum_profit_voisinage / count_voisinage : 0.0;

    // Calculate the heuristic value, prioritizing remaining capacity and neighborhood profit
    double h = (remaining_capacity > 0) ? (profit[index_item] / (density + 1e-6)) * (1 + (average_profit_voisinage / (profit[index_item] + 1e-6))) * (remaining_capacity + 1e-6) : 0.0;

    // Sparsify the heuristic
    return (h < 0.01) ? 0.0 : h;
}
