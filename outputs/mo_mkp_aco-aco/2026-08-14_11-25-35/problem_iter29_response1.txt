double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] )
{
    // Calculate the density of the item
    double density = 0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            density += weights[j][index_item] / capacity[j];
        }
    }

    // Calculate the remaining capacity after including the item
    double remaining_capacity = 1;
    double capacity_impact = 0;
    for (int j = 0; j < dimension; j++) {
        double remaining = (capacity[j] - weights[j][index_item]) / capacity[j];
        remaining_capacity *= (remaining > 0) ? remaining : 0;
        capacity_impact += (remaining < 0) ? -remaining : 0;
    }

    // Calculate the average profit of the neighborhood
    double sum_profit_voisinage = 0;
    double count_voisinage = 0;
    double max_profit_voisinage = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (index_item != voisinage[i]) {
            sum_profit_voisinage += profit[voisinage[i]];
            count_voisinage++;
            if (profit[voisinage[i]] > max_profit_voisinage) {
                max_profit_voisinage = profit[voisinage[i]];
            }
        }
    }
    double average_profit_voisinage = (count_voisinage > 0) ? sum_profit_voisinage / count_voisinage : 0;

    // Calculate the heuristic value
    double h = (profit[index_item] / (density + 1e-6));
    h *= (1 + (average_profit_voisinage / (profit[index_item] + 1e-6)));
    h *= (remaining_capacity + 1e-6);
    h *= (1 - capacity_impact);

    // Sparsify the heuristic
    if (h < 0.01 || density > 1 || capacity_impact > 0.5) {
        return 0;
    }

    return h;
}
