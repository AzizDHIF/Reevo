double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate the density of the item, considering the most limiting dimension
    double max_density = 0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            double density = weights[j][index_item] / capacity[j];
            if (density > max_density) {
                max_density = density;
            }
        }
    }

    // Calculate the average profit of the neighborhood, ignoring the item itself
    double sum_profit_voisinage = 0;
    double count_voisinage = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (index_item != voisinage[i]) {
            sum_profit_voisinage += profit[voisinage[i]];
            count_voisinage++;
        }
    }
    double average_profit_voisinage = (count_voisinage > 0) ? sum_profit_voisinage / count_voisinage : 0;

    // Calculate the standard deviation of the neighborhood profit
    double sum_squared_profit_voisinage = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (index_item != voisinage[i]) {
            sum_squared_profit_voisinage += (profit[voisinage[i]] - average_profit_voisinage) * (profit[voisinage[i]] - average_profit_voisinage);
        }
    }
    double std_dev_profit_voisinage = (count_voisinage > 0) ? sqrt(sum_squared_profit_voisinage / count_voisinage) : 0;

    // Calculate the remaining capacity after including the item
    double remaining_capacity = 1;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > weights[j][index_item]) {
            remaining_capacity *= (capacity[j] - weights[j][index_item]) / capacity[j];
        } else {
            remaining_capacity = 0;
        }
    }

    // Calculate the heuristic value, considering density, neighborhood, and capacity constraints
    double h = profit[index_item] / (max_density + 1e-6);
    h *= (1 + (average_profit_voisinage / (profit[index_item] + 1e-6)));
    h *= (remaining_capacity + 1e-6);
    h /= (std_dev_profit_voisinage + 1e-6);

    // Prioritize items with higher profit and lower density
    if (max_density > 1) {
        h *= 0.1;
    }

    // Sparsify the heuristic
    if (h < 0.01 || remaining_capacity == 0) {
        return 0;
    }

    return h;
}
