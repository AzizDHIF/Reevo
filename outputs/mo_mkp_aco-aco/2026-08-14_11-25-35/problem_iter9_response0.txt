double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate the density of the item, considering only the most limiting dimension
    double max_density = 0;
    for (int j = 0; j < dimension; j++) {
        double density = weights[j][index_item] / capacity[j];
        if (density > max_density) {
            max_density = density;
        }
    }

    // Calculate the average profit of the neighborhood, ignoring the item itself
    double sum_profit_voisinage = 0;
    double count_voisinage = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (index_item != voisinage[i]) {
            sum_profit_voisinage += profit[voisinage[i]];
            count_voisinage++;
        }
    }
    double average_profit_voisinage = (count_voisinage > 0) ? sum_profit_voisinage / count_voisinage : 0;

    // Calculate the heuristic value, prioritizing density and neighborhood
    double h = profit[index_item] / (max_density + 1e-6);
    h *= (1 + (average_profit_voisinage / (profit[index_item] + 1e-6)));

    // Sparsify the heuristic, focusing on the most promising items
    if (h < 0.01 || max_density > 1) {
        return 0;
    }

    return h;
}
