double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Prioritize feasibility
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > capacity[j]) {
            return 0.0; // infeasible item
        }
    }

    // Calculate the density of the item
    double density = 0.0;
    for (int j = 0; j < dimension; j++) {
        density += weights[j][index_item] / capacity[j];
    }

    // Balance density and profit
    double balance = (density > 0) ? profit[index_item] / density : profit[index_item];

    // Consider neighborhood quality
    double sum_profit_voisinage = 0.0;
    double count_voisinage = 0.0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (index_item != voisinage[i]) {
            sum_profit_voisinage += profit[voisinage[i]];
            count_voisinage++;
        }
    }
    double average_profit_voisinage = (count_voisinage > 0) ? sum_profit_voisinage / count_voisinage : 0.0;
    double neighborhood_quality = (average_profit_voisinage > 0) ? average_profit_voisinage / (profit[index_item] + 1e-6) : 0.0;

    // Calculate the heuristic value
    double h = balance * (1.0 + neighborhood_quality);

    // Sparsify the heuristic
    if (h < 0.01) {
        return 0.0;
    }

    return h;
}
