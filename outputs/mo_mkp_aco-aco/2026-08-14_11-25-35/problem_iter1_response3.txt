double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    // Calculate the total weight of the neighboring items
    double total_weight = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        for (int j = 0; j < dimension; j++) {
            total_weight += weights[j][voisinage[i]];
        }
    }

    // Calculate the average remaining capacity
    double avg_remaining_capacity = 0;
    for (int j = 0; j < dimension; j++) {
        avg_remaining_capacity += capacity[j];
    }
    avg_remaining_capacity /= dimension;

    // Calculate the total profit of the neighboring items
    double total_profit = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        total_profit += profit[voisinage[i]];
    }

    // Combine the factors to determine how promising it is to select an item
    double h = (profit[index_item] / (total_weight + 1e-6)) * (avg_remaining_capacity / (total_profit + 1e-6));

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 1e-6) {
        h = 0;
    }

    return h;
}
