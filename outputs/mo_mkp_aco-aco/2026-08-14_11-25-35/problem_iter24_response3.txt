#include <stdio.h>

#define dimension 3  // define the dimension
#define NBITEMS 10   // define the number of items

double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate the density of the item, considering the feasibility
    double density = 0;
    double is_feasible = 1;
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > capacity[j]) {
            is_feasible = 0;
            break;
        }
        density += weights[j][index_item] / capacity[j];
    }
    if (is_feasible == 0) {
        return 0;
    }

    // Calculate the remaining capacity after including the item
    double remaining_capacity = 1;
    for (int j = 0; j < dimension; j++) {
        remaining_capacity *= (capacity[j] - weights[j][index_item]) / capacity[j];
    }

    // Calculate the average profit of the neighborhood, excluding the item itself
    double sum_profit_voisinage = 0;
    double count_voisinage = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (index_item != voisinage[i]) {
            sum_profit_voisinage += profit[voisinage[i]];
            count_voisinage++;
        }
    }
    double average_profit_voisinage = (count_voisinage > 0) ? sum_profit_voisinage / count_voisinage : 0;

    // Calculate the heuristic value, considering density, remaining capacity, and neighborhood profit
    double h = is_feasible * profit[index_item] / (density + 1e-6);
    h *= (1 + (average_profit_voisinage / (profit[index_item] + 1e-6)));
    h *= (remaining_capacity + 1e-6);

    // Sparsify the heuristic
    if (h < 0.01) {
        return 0;
    }

    return h;
}

int main() {
    int index_item = 0;
    double weights[dimension][NBITEMS];
    double capacity[dimension];
    int nb_voisinage = 5;
    int voisinage[NBITEMS];
    double profit[NBITEMS];

    // initialize the variables
    for (int i = 0; i < dimension; i++) {
        capacity[i] = 10;
        for (int j = 0; j < NBITEMS; j++) {
            weights[i][j] = 1;
        }
    }
    for (int i = 0; i < NBITEMS; i++) {
        voisinage[i] = i;
        profit[i] = 10;
    }

    double result = heuristic_v2(index_item, weights, capacity, nb_voisinage, voisinage, profit);
    printf("Heuristic value: %f\n", result);

    return 0;
}
