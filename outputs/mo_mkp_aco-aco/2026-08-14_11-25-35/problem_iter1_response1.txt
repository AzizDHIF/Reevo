double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double h = 0;
    double total_weight = 0;
    double max_weight = 0;

    // Calculate the total weight and the maximum weight of the item for each dimension
    for(int j = 0; j < dimension; j++){
        total_weight += weights[j][index_item];
        max_weight = (max_weight > weights[j][index_item]) ? max_weight : weights[j][index_item];
    }

    // Calculate the average weight of the item for each dimension
    double average_weight = total_weight / dimension;

    // Calculate the ratio of the profit to the total weight
    double ratio = profit[index_item] / total_weight;

    // Calculate the heuristic value based on the ratio and the capacity
    for(int j = 0; j < dimension; j++){
        h += (weights[j][index_item] / capacity[j]) * (1 + ratio);
    }

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h > 1e6) {
        h = 0;
    }

    // Calculate the neighborhood density
    double neighborhood_density = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        neighborhood_density += profit[voisinage[i]] / total_weight;
    }

    // Calculate the final heuristic value
    h += neighborhood_density * (1 + ratio);

    return h;
}
