double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate the simplified density of the item
    double density = 0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            density += weights[j][index_item] / capacity[j];
        }
    }

    // Calculate the remaining capacity after including the item
    double remaining_capacity = 1;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > weights[j][index_item]) {
            remaining_capacity *= (capacity[j] - weights[j][index_item]) / capacity[j];
        } else {
            remaining_capacity = 0;
            break;
        }
    }

    // Calculate the average profit of the neighborhood
    double sum_profit_voisinage = 0;
    double count_voisinage = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (index_item != voisinage[i]) {
            sum_profit_voisinage += profit[voisinage[i]];
            count_voisinage++;
        }
    }
    double average_profit_voisinage = (count_voisinage > 0) ? sum_profit_voisinage / count_voisinage : 0;

    // Prioritize feasibility, balance density, profit, and neighborhood
    double h = 0;
    if (remaining_capacity > 0) {
        double profit_factor = profit[index_item] / (density + 1e-6);
        double neighborhood_factor = 1 + (average_profit_voisinage / (profit[index_item] + 1e-6));
        double feasibility_factor = remaining_capacity + 1e-6;
        h = profit_factor * neighborhood_factor * feasibility_factor;
    }

    // Consider the maximum weight dimension for the item
    double max_weight = 0;
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > max_weight) {
            max_weight = weights[j][index_item];
        }
    }

    // Adjust the heuristic value based on the maximum weight dimension
    if (max_weight > 0) {
        h /= (max_weight + 1e-6);
    }

    // Sparsify the heuristic
    if (h < 0.01) {
        return 0;
    }

    return h;
}
