double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Prioritize feasibility
    double remaining_capacity = 1;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] < weights[j][index_item]) {
            return 0;  // infeasible, do not consider
        }
        remaining_capacity *= (capacity[j] - weights[j][index_item]) / capacity[j];
    }

    // Calculate the density of the item
    double density = 0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            density += weights[j][index_item] / capacity[j];
        }
    }
    density /= dimension; // average density across dimensions

    // Calculate the average profit of the neighborhood
    double sum_profit_voisinage = 0;
    double count_voisinage = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (index_item != voisinage[i]) {
            sum_profit_voisinage += profit[voisinage[i]];
            count_voisinage++;
        }
    }
    double average_profit_voisinage = (count_voisinage > 0) ? sum_profit_voisinage / count_voisinage : 0;

    // Prioritize density, neighborhood quality, and profit
    double h = profit[index_item] / (density + 1e-6);  // prioritize density
    h *= (1 + (average_profit_voisinage / (profit[index_item] + 1e-6)));  // prioritize neighborhood quality
    h *= (remaining_capacity + 1e-6);  // prioritize feasibility

    // Balance density and profit
    double balance = (density < 0.5) ? h : h / (1 + density);

    // Sparsify the heuristic
    if (balance < 0.01) {
        return 0;
    }

    return balance;
}
