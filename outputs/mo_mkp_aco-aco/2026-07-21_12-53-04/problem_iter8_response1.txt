double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double utilization = 0;
    double neighbor_profits = 0;
    int neighbor_count = 0;
    double max_profit = 0;

    // Calculate the utilization of each dimension
    for (int j = 0; j < dimension; j++) {
        utilization += weights[j][index_item] / capacity[j];
    }
    utilization /= dimension;

    // Find the maximum profit among all items
    for (int i = 0; i < NBITEMS; i++) {
        if (profit[i] > max_profit) {
            max_profit = profit[i];
        }
    }

    // Consider the neighborhood of the current item
    for (int i = 0; i < nb_voisinage; i++) {
        int neighbor = voisinage[i];
        neighbor_profits += profit[neighbor];
        neighbor_count++;

        // Combine the utilization and profit of the neighbor
        double neighbor_utilization = 0;
        for (int j = 0; j < dimension; j++) {
            neighbor_utilization += weights[j][neighbor] / capacity[j];
        }

        // Calculate the relative profit of the neighbor
        double relative_neighbor_profit = profit[neighbor] / (max_profit + 1e-6); // avoid division by zero
    }

    // Calculate the relative profit of the current item
    double relative_profit = profit[index_item] / (max_profit + 1e-6); // avoid division by zero

    // Calculate the heuristic based on the relative profit and utilization
    double h = relative_profit / (utilization + 1e-9); // Add epsilon to avoid division by zero

    // Consider the neighborhood when calculating the heuristic
    if (neighbor_count > 0) {
        double average_neighbor_profit = neighbor_profits / neighbor_count;
        h += average_neighbor_profit / (utilization + 1e-9); // Add epsilon to avoid division by zero
    }

    // Sparsify the heuristic by setting unpromising elements to zero
    if (utilization > 1 || relative_profit < 0.1) {
        h = 0;
    }

    return h;
}
