double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double relative_profit = 0;
    double average_neighbor_utilization = 0;
    double relative_utilization = 0;
    int neighbor_count = 0;
    double max_profit = 0;

    // Calculate the relative utilization of the current item
    for (int j = 0; j < dimension; j++) {
        relative_utilization += weights[j][index_item] / capacity[j];
    }
    relative_utilization /= dimension;

    // Find the maximum profit among all items
    for (int i = 0; i < NBITEMS; i++) {
        if (profit[i] > max_profit) {
            max_profit = profit[i];
        }
    }

    // Calculate the relative profit of the current item
    relative_profit = profit[index_item] / (max_profit + 1e-6); // avoid division by zero

    // Consider the neighborhood of the current item
    for (int i = 0; i < nb_voisinage; i++) {
        int neighbor = voisinage[i];
        double neighbor_utilization = 0;
        for (int j = 0; j < dimension; j++) {
            neighbor_utilization += weights[j][neighbor] / capacity[j];
        }
        average_neighbor_utilization += neighbor_utilization / dimension; // Calculate relative utilization for each neighbor
        neighbor_count++;
    }

    // Calculate the average neighbor utilization
    if (neighbor_count > 0) {
        average_neighbor_utilization /= neighbor_count;
    }

    // Combine the relative profit and utilization of the current item
    h += relative_profit * (1 - relative_utilization);

    // Consider the average neighbor utilization
    if (neighbor_count > 0) {
        h += (1 - average_neighbor_utilization); // Give more priority to items with less utilized neighbors
    }

    // Sparsify the heuristic by setting unpromising elements to zero
    if (neighbor_count == 0 || relative_utilization > 0.5 || relative_profit < 0.1) {
        h = 0;
    }

    return h;
}
