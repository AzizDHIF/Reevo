double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double utilization = 0;
    double neighbor_profits = 0;
    int neighbor_count = 0;
    double max_profit = 0;
    double relative_utilization = 0;

    // Calculate the utilization of each dimension
    for(int j = 0; j < dimension; j++) {
        utilization += weights[j][index_item] / capacity[j];
    }

    // Find the maximum profit among all items
    for (int i = 0; i < NBITEMS; i++) {
        if (profit[i] > max_profit) {
            max_profit = profit[i];
        }
    }

    // Calculate relative utilization
    for (int i = 0; i < NBITEMS; i++) {
        double item_utilization = 0;
        for(int j = 0; j < dimension; j++) {
            item_utilization += weights[j][i] / capacity[j];
        }
        relative_utilization += item_utilization;
    }
    relative_utilization /= NBITEMS;

    // Consider the neighborhood of the current item
    for(int i = 0; i < nb_voisinage; i++) {
        int neighbor = voisinage[i];
        neighbor_profits += profit[neighbor];
        neighbor_count++;

        // Calculate the heuristic for the neighbor
        double neighbor_utilization = 0;
        for(int j = 0; j < dimension; j++) {
            neighbor_utilization += weights[j][neighbor] / capacity[j];
        }

        // Combine the utilization and profit of the neighbor
        h += profit[neighbor] / (neighbor_utilization + 1e-9); // Add epsilon to avoid division by zero
    }

    // Combine the utilization and profit of the current item
    double relative_profit = profit[index_item] / (max_profit + 1e-6); // avoid division by zero
    double utilization_weight = utilization / (relative_utilization + 1e-9); // Add epsilon to avoid division by zero
    h += relative_profit / (utilization_weight + 1e-9); // Add epsilon to avoid division by zero

    // Prioritize relative profit and neighbor utilization
    double priority = relative_profit * (neighbor_profits / (neighbor_count + 1e-6)) / (utilization + 1e-9);

    // Sparsify the heuristic by setting unpromising elements to zero
    if(neighbor_count == 0 || utilization > 1 || relative_profit < 0.1 || priority < 0.1) {
        h = 0;
    }

    return h;
}
