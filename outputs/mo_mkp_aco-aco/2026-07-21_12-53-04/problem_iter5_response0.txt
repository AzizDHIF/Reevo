double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double utilization = 0;
    double neighbor_profits = 0;
    int neighbor_count = 0;
    double max_profit = 0;
    double total_weight = 0;

    // Calculate the total weight of the item
    for(int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item];
        utilization += weights[j][index_item] / capacity[j];
    }

    // Find the maximum profit among all items
    for (int i = 0; i < NBITEMS; i++) {
        if (profit[i] > max_profit) {
            max_profit = profit[i];
        }
    }

    // Consider the neighborhood of the current item
    for(int i = 0; i < nb_voisinage; i++) {
        int neighbor = voisinage[i];
        neighbor_profits += profit[neighbor];
        neighbor_count++;
    }

    // Combine profitability and utilization metrics
    double relative_profit = profit[index_item] / (max_profit + 1e-6); // avoid division by zero
    double utilization_metric = 1 - utilization; // prefer items with lower utilization

    // Combine the metrics with neighborhood profits
    if(neighbor_count > 0) {
        h = relative_profit * utilization_metric * (neighbor_profits / neighbor_count);
    } else {
        h = relative_profit * utilization_metric;
    }

    // Sparsify the heuristic by setting unpromising elements to zero
    if(utilization > 1 || relative_profit < 0.1 || total_weight > 1) {
        h = 0;
    }

    return h;
}
