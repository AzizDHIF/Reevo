double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double weight_sum = 0;
    double remaining_capacity = 0;
    double total_profit = 0;
    double average_weight = 0;

    // calculate the sum of weights for the current item and its neighbors
    for(int i = 0; i < nb_voisinage; i++) {
        for(int j = 0; j < dimension; j++) {
            weight_sum += weights[j][voisinage[i]];
        }
        total_profit += profit[voisinage[i]];
    }

    // calculate the average weight per item
    for(int j = 0; j < dimension; j++) {
        remaining_capacity += capacity[j];
        average_weight += weight_sum / nb_voisinage;
    }

    // calculate the heuristic value
    double h = (total_profit / average_weight) * (remaining_capacity / (average_weight * nb_voisinage));

    // apply a penalty if the item's weight exceeds the capacity
    for(int j = 0; j < dimension; j++) {
        if(weights[j][index_item] > capacity[j]) {
            h *= 0.5; // penalize items that exceed capacity
        }
    }

    // sparsify the heuristic by setting unpromising elements to zero
    if(h < 0 || isnan(h)) {
        h = 0; // set to zero if heuristic value is invalid
    }

    return h;
}
