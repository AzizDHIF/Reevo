double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate the minimum capacity across all dimensions
    double min_capacity = capacity[0];
    for (int j = 1; j < dimension; j++) {
        if (capacity[j] < min_capacity) {
            min_capacity = capacity[j];
        }
    }

    // Calculate the total profit and weight of the neighbors
    double total_profit = 0;
    double total_weight = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        int neighbor = voisinage[i];
        total_profit += profit[neighbor];
        for (int j = 0; j < dimension; j++) {
            total_weight += weights[j][neighbor];
        }
    }

    // Calculate the average weight and profit of the neighbors
    double average_weight = total_weight / nb_voisinage;
    double average_profit = total_profit / nb_voisinage;

    // Calculate the heuristic value based on the average weight and profit of the neighbors
    double h = 0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] - average_weight > 0) {
            h += weights[j][index_item] / (capacity[j] - average_weight);
        } else {
            h += weights[j][index_item];
        }
    }

    // Add a sparsifying factor to set unpromising elements to zero
    double sparsifying_factor = (total_profit > 0 && h > 0) ? 1 : 0;

    // Combine various factors to determine how promising it is to select an item
    double heuristic_value = (profit[index_item] / (h + 1e-6)) * (1 + average_profit) * sparsifying_factor;

    // Adjust heuristic value according to remaining capacity
    double remaining_capacity = 0;
    for (int j = 0; j < dimension; j++) {
        remaining_capacity += capacity[j];
    }
    double total_weight_with_item = total_weight;
    for (int j = 0; j < dimension; j++) {
        total_weight_with_item += weights[j][index_item];
    }
    remaining_capacity -= total_weight_with_item;

    if (remaining_capacity < 0) {
        heuristic_value *= (1 - (total_weight_with_item - remaining_capacity) / total_weight_with_item);
    } else {
        heuristic_value *= (1 + remaining_capacity / total_weight_with_item);
    }

    return heuristic_value;
}
