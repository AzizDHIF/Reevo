double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double sum_profits = 0;
    double avg_capacity = 0;
    double num_voisis = 1; // include itself

    // Sum of profits and average capacity among neighboring items
    for (int i = 0; i < nb_voisinage; i++) {
        sum_profits += profit[voisinage[i]];
        for (int j = 0; j < dimension; j++) {
            avg_capacity += capacity[j];
        }
    }
    avg_capacity /= (double) (nb_voisinage * dimension); // normalize average capacity

    // calculate heurisitic value by incorporating both capacity and profit
    for (int i = 0; i < dimension; i++) {
        h += (1 - weights[i][index_item] / capacity[i]); // leave room in knapsack
    }
    h = h / (double) dimension; // normalize among dimensions

    // Adjust heuristic value according to neighborhood
    for (int i = 0; i < nb_voisinage; i++) {
        double temp_h = h * profit[voisinage[i]]; // factor in profit per item
        temp_h /= sum_profits; // divide by total profit among neighborhood for fairness
        if (profit[index_item] > temp_h) {
            h = profit[index_item];
        }
    }

    // Sparisity to focus only on promising elements
    if (h < 0 || sum_profits == 0 || avg_capacity < 1e-5) {
        h = 0;  // discard if conditions not satisfactory
    }

    // Balance between capacity and profit
    return h / (1 + h * (1 / avg_capacity));
}
