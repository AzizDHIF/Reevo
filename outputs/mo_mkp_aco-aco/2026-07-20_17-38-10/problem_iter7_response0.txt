double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double min_capacity = capacity[0];
    for(int j=1;j<dimension;j++){
        if (capacity[j] < min_capacity) {
            min_capacity = capacity[j];
        }
    }

    double total_profit = 0;
    double total_weight = 0;

    // Consider only the neighbors of the current item
    for(int i=0; i<nb_voisinage; i++){
        int neighbor = voisinage[i];
        total_profit += profit[neighbor];
        for(int j=0;j<dimension;j++){
            total_weight += weights[j][neighbor];
        }
    }

    double average_weight = (nb_voisinage > 0) ? total_weight / nb_voisinage : 0;
    double average_profit = (nb_voisinage > 0) ? total_profit / nb_voisinage : 0;

    // Calculate the impact of the current item on the capacity
    double item_impact = 0;
    for(int j=0;j<dimension;j++){
        item_impact += weights[j][index_item];
    }
    item_impact /= dimension;

    // Calculate the average capacity left
    double average_capacity_left = 0;
    for(int j=0;j<dimension;j++){
        average_capacity_left += capacity[j] - average_weight;
    }
    average_capacity_left /= dimension;

    // Calculate the heuristic value based on the item's impact and the average capacity left
    if (average_capacity_left > 0) {
        h = profit[index_item] / (item_impact / average_capacity_left);
    } else {
        h = 0;
    }

    // Add a sparsifying factor to set unpromising elements to zero
    double sparsifying_factor = (total_profit > 0 && profit[index_item] > 0) ? 1 : 0;

    // Combine various factors to determine how promising it is to select an item
    double heuristic_value = h * (1 + average_profit) * sparsifying_factor;

    return heuristic_value;
}
