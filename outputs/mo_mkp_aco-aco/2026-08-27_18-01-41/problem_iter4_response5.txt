double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double w_factor = 0.7; // Weight factor
    double p_factor = 0.3; // Profit factor
    double balance_factor = 0.5; // Factor to balance weight and profit
    double promise_level = 0;

    // Calculate promise level considering weight and profit factors
    for (int j = 0; j < dimension && promise_level < 1; j++) {
        promise_level += weights[j][voisinage[index_item]] / capacity[j];
    }

    if (promise_level >= 1) {
        return 0; // Zero value if including the item exceeds capacity
    }

    // Combine weight and profit factors to determine item's promise level
    return (p_factor * profit[voisinage[index_item]]) / ((1 - balance_factor) * promise_level + balance_factor);
}
