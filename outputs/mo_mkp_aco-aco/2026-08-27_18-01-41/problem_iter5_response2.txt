double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double w_factor = 0.7; // Weight factor
    double p_factor = 0.3; // Profit factor

    for (int j = 0; j < dimension && h < 1; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j];
    }

    if (h >= 1) {
        return 0; // Zero value if current item overflows capacity
    }

    double promising_value = (p_factor * profit[voisinage[index_item]]) / (w_factor * h + p_factor);
    
    double penalty = 0;
    for (int j = 0; j < dimension; j++) {
        double remaining_capacity = capacity[j] - weights[j][index_item];
        if (remaining_capacity < 0) {
            penalty += 0.5; // Penalize capacity violations
        }
    }

    return promising_value - penalty;
}
