double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h1 = 0, h2 = 0;

    for (int j = 0; j < dimension; j++) {
        h1 += weights[j][voisinage[index_item]] / capacity[j];
        h2 += weights[j][voisinage[index_item]] * capacity[j];
    }

    double weight_ratio = (h1 == 0) ? 0 : 1 / h1;
    double capacity_ratio = (h2 == 0) ? 0 : capacity[0] / h2;

    return 0.4 * profit[voisinage[index_item]] * weight_ratio + 0.6 * capacity_ratio;
}
