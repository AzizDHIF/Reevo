double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double a = 0.5, b = 0.3, c = 0.2; // coefficients for combining factors

    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j];
    }

    double factor_1 = b * profit[voisinage[index_item]] / h;

    if (capacity[0] < 0.6) {
        return 0;
    }

    double factor_2 = c * capacity[0];

    if (factor_1 > 0.1) {
        return a * factor_1 + b * factor_2;
    } else {
        return 0;
    }
}
