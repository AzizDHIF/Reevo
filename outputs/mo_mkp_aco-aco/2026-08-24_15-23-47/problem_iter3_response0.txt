double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h1 = 0, h2 = 0;

    for(int j = 0; j < dimension; j++) {
        if (capacity[j] != 0) {
            h1 += weights[j][voisinage[index_item]] / capacity[j];
            h2 += weights[j][voisinage[index_item]] * capacity[j];
        }
    }

    if (h1 == 0 || h2 == 0) {
        return 0;
    } else {
        return (0.5 * profit[voisinage[index_item]] / h1) + (0.5 * capacity[0] / h2);
    }
}
