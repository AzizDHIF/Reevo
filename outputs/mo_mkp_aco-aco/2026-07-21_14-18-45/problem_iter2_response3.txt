double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double density = 0.0;
    double capacity_ratio = 0.0;

    // Calculate the density of the item
    double total_weight = 0.0;
    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item];
    }
    density = profit[index_item] / total_weight;

    // Calculate the capacity ratio
    for (int j = 0; j < dimension; j++) {
        capacity_ratio += weights[j][index_item] / capacity[j];
    }

    // Calculate the heuristic value, prioritize density and capacity
    if (capacity_ratio > 1) {
        return 0.0; // Set heuristic to 0 if capacity constraint is not met
    } else {
        double h = density / (capacity_ratio + 1e-9);
        if (h < 0.5) {
            h = 0.0; // Sparsify the heuristic by setting unpromising elements to zero
        }
        return h;
    }
}
