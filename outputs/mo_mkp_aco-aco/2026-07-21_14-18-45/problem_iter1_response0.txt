double heuristic_v2(int index_item, double weights[/*dimension*/][NBITEMS], double capacity[/*dimension*/], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double numerator = profit[index_item];
    double denominator = 0;
    for (int j = 0; j < /*dimension*/; j++) {
        denominator += weights[j][index_item] / capacity[j];
    }
    double ratio = numerator / (denominator + 1e-9); // to avoid zero division

    // consider the neighboring items
    double neighbor_score = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        int neighbor_index = voisinage[i];
        double neighbor_weight_ratio = 0;
        for (int j = 0; j < /*dimension*/; j++) {
            neighbor_weight_ratio += weights[j][neighbor_index] / capacity[j];
        }
        neighbor_score += profit[neighbor_index] / (neighbor_weight_ratio + 1e-9); // to avoid zero division
    }

    // sparsify the heuristic by setting unpromising elements to zero
    if (ratio < 0.1) {
        return 0;
    }

    // combine various factors
    double heuristic_score = ratio * (1 + 0.1 * neighbor_score);

    return heuristic_score;
}
