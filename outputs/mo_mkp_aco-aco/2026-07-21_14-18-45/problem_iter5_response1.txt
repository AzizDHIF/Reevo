double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0.0;
    double total_weight = 0.0;
    double avg_weight = 0.0;
    double density = 0.0;
    double capacity_utilization = 0.0;
    double avg_weight_voisinage = 0.0;
    double sum_profit_voisinage = 0.0;

    // Calculate the total weight of the item
    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item];
    }
    avg_weight = total_weight / dimension;

    // Calculate the density of the item
    density = profit[index_item] / avg_weight;

    // Calculate the capacity utilization
    for (int j = 0; j < dimension; j++) {
        capacity_utilization += weights[j][index_item] / capacity[j];
    }
    capacity_utilization /= dimension;

    // Calculate the average weight and sum of profit of neighbors
    for (int i = 0; i < nb_voisinage; i++) {
        int voisinage_index = voisinage[i];
        for (int j = 0; j < dimension; j++) {
            avg_weight_voisinage += weights[j][voisinage_index];
        }
        sum_profit_voisinage += profit[voisinage_index];
    }
    avg_weight_voisinage /= (nb_voisinage * dimension);
    sum_profit_voisinage /= nb_voisinage;

    // Prioritize density, capacity utilization, and sparsity
    h = density * (1 - capacity_utilization);

    // Apply a soft threshold to promote sparsity
    if (h > 0 && density < 0.5) {
        h *= 0.5; // Reduce the heuristic value for low-density items
    }
    if (h > 0 && capacity_utilization > 0.8) {
        h *= 0.2; // Reduce the heuristic value for high-capacity utilization items
    }

    // Promote items with high profit and low weight among neighbors
    if (avg_weight_voisinage > 0) {
        double neighbor_ratio = (profit[index_item] / avg_weight) / (sum_profit_voisinage / (avg_weight_voisinage * nb_voisinage));
        h *= neighbor_ratio;
    }

    // Ensure non-negative heuristic values
    h = (h > 0) ? h : 0;

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 0.1) {
        h = 0;
    }

    return h;
}
