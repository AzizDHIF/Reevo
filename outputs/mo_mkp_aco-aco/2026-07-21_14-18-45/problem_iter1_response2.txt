double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double max_weight = 0;
    double max_profit = 0;
    double capacity_ratio = 0;

    // Calculate maximum weight and profit among neighbours
    for (int i = 0; i < nb_voisinage; i++) {
        double temp_weight = 0;
        for (int j = 0; j < dimension; j++) {
            temp_weight += weights[j][voisinage[i]];
        }
        if (temp_weight > max_weight) {
            max_weight = temp_weight;
        }
        if (profit[voisinage[i]] > max_profit) {
            max_profit = profit[voisinage[i]];
        }
    }

    // Calculate capacity ratio
    for (int j = 0; j < dimension; j++) {
        capacity_ratio += weights[j][index_item] / capacity[j];
    }

    // Calculate heuristic value
    if (capacity_ratio > 1) {
        h = 0; // Set heuristic to 0 if capacity constraint is not met
    } else {
        double weight_ratio = 0;
        for (int j = 0; j < dimension; j++) {
            weight_ratio += weights[j][index_item] / (max_weight + 1e-9);
        }
        h = profit[index_item] / (weight_ratio + capacity_ratio + 1e-9);
        // Sparsify the heuristic by setting unpromising elements to zero
        if (h < max_profit * 0.1) {
            h = 0;
        }
    }
    return h;
}
