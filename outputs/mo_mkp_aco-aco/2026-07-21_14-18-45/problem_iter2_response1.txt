double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0.0;
    double avg_weight = 0.0;
    double density = 0.0;
    double avg_neighbor_weight = 0.0;

    // Calculate the average weight of the item
    for (int j = 0; j < dimension; j++) {
        avg_weight += weights[j][index_item];
    }
    avg_weight /= dimension;

    // Calculate the average weight of the neighbors
    for (int i = 0; i < nb_voisinage; i++) {
        for (int j = 0; j < dimension; j++) {
            avg_neighbor_weight += weights[j][voisinage[i]];
        }
    }
    avg_neighbor_weight /= (nb_voisinage * dimension);

    // Calculate the density of the item
    density = profit[index_item] / avg_weight;

    // Calculate the heuristic value
    double capacity_ratio = 0.0;
    for (int j = 0; j < dimension; j++) {
        capacity_ratio += weights[j][index_item] / capacity[j];
    }
    h = profit[index_item] / (capacity_ratio + 1e-9);

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 0.1 || density < 0.5 || avg_weight > avg_neighbor_weight * 2) {
        h = 0.0;
    }

    return h;
}
