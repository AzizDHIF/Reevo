double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate how well the item fits into the knapsack for each dimension
    double fit = 1.0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0.0) {
            fit *= (1.0 - weights[j][index_item] / capacity[j]);
        }
    }
    fit = 1.0 - (1.0 - fit) / dimension;

    // Calculate the potential conflict with neighboring items
    double neighborhood_conflict = 0.0;
    int valid_neighbors = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        int item = voisinage[i];
        if (item == index_item) continue;  // Skip current item
        double competing_weight = 0.0;
        for (int j = 0; j < dimension; j++) {
            competing_weight += (weights[j][item] / capacity[j]);
        }
        if (competing_weight > 0.0) {
            neighborhood_conflict += competing_weight / (1.0 + competing_weight);
            valid_neighbors++;
        }
    }
    if (valid_neighbors > 0) {
        neighborhood_conflict /= valid_neighbors;
    } else {
        neighborhood_conflict = 0.0;
    }

    // Combine factors: prioritize fit and conflict over weighted sums
    double final_heuristic = profit[index_item] * fit * (1.0 - neighborhood_conflict);

    return final_heuristic;
}
