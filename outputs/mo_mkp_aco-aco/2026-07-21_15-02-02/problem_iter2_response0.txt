double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double sum_weights = 0.0;
    double min_capacity = 1e6;
    double profit_sum = 0.0;

    // Calculate the average weight and minimal capacity
    for (int j = 0; j < dimension; j++) {
        sum_weights += weights[j][index_item];
        if (capacity[j] < min_capacity) {
            min_capacity = capacity[j];
        }
        if (j < nb_voisinage) {
            profit_sum += profit[voisinage[j]];
        }
    }

    // Calculate the heuristic value based on a weighted sum of objectives
    double h = (profit[index_item] / (1 + sum_weights)) * (min_capacity / (1 + sum_weights)) * (1 + profit_sum / (nb_voisinage + 1));

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 1e-6 || sum_weights > min_capacity) {
        h = 0.0;
    }

    // Additional consideration: encourage exploration by giving more weight to less frequently visited items
    int visit_count = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (voisinage[i] == index_item) {
            visit_count++;
        }
    }
    h *= (1 + 1.0 / (1 + visit_count));

    // Diversification term: penalize items that are too similar to already selected items
    double similarity_sum = 0.0;
    for (int i = 0; i < NBITEMS; i++) {
        if (i != index_item) {
            double similarity = 0.0;
            for (int j = 0; j < dimension; j++) {
                similarity += weights[j][i] * weights[j][index_item];
            }
            similarity_sum += similarity;
        }
    }
    h *= (1 - similarity_sum / (NBITEMS * dimension));

    return h;
}
