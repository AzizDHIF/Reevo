double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double promising_factor = 0;
    double capacity_factor = 0;

    for (int j = 0; j < dimension; j++) {
        double w = weights[j][voisinage[index_item]];
        double cap = capacity[j];
        promising_factor += (w > capacity[j]) ? profit[voisinage[index_item]] / w : profit[voisinage[index_item]] / cap;
        capacity_factor += capacity[j] - weights[j][index_item];
    }

    promising_factor = (promising_factor > 0) ? promising_factor : 0;
    capacity_factor = 1 / (1 + capacity_factor);

    return promising_factor * capacity_factor;
}
