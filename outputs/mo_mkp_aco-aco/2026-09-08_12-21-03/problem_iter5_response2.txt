double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] )
{
    double h = 0;
    for (int i = 0; i < dimension; i++) {
        if (weights[i][index_item] > 0) {
            h += weights[i][index_item] / capacity[i];
        }
    }

    double promising_factor = profit[index_item] / h;
    
    double capacity_filled = 0;
    for (int i = 0; i < dimension; i++) {
        if (weights[i][index_item] < capacity[i]) {
            capacity_filled += capacity[i] - weights[i][index_item];
        }
    }
    
    double decision_factor = 1.0 / (1.0 + capacity_filled);

    return promising_factor * decision_factor;
}
