double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double promising_factor = 0;

    double remaining_capacity = 0;
    for (int j = 0; j < dimension; j++) {
        remaining_capacity += capacity[j] - weights[j][index_item];
    }
    remaining_capacity = 1 / (1 + remaining_capacity);

    if (remaining_capacity <= 0) {
        return 0;
    }

    double max_ratio = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        double curr_weight = 0;
        double ratio = 0;
        for (int j = 0; j < dimension; j++) {
            curr_weight = weights[j][voisinage[i]];
            ratio = curr_weight / (capacity[j] - weights[j][index_item]);
            if (ratio > max_ratio) {
                max_ratio = ratio;
            }
        }

        promising_factor += max_ratio * profit[voisinage[i]];
    }

    return promising_factor * remaining_capacity;
}
