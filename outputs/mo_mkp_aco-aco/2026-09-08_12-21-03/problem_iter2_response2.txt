double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 1;
    double w_sum = 0;

    for(int j = 0; j < dimension; j++) {
        w_sum += weights[j][voisinage[index_item]] / capacity[j];
        if(weights[j][voisinage[index_item]] > capacity[j]){
            h = 0;
            break;
        }
    }

    if(h > 0) {
        h = profit[voisinage[index_item]] * (1 / w_sum);
    }

    int isPromising = profit[voisinage[index_item]] > 0 && h > 0;
    return isPromising ? h : 0;
}
