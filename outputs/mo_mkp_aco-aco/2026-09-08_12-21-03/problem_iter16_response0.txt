double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double promising_factor = 0;
    double capacity_factor = 0;

    for (int j = 0; j < dimension; j++) {
        double curr_weight = weights[j][voisinage[index_item]];
        double remaining_capacity = capacity[j] - weights[j][index_item];

        if (remaining_capacity > 0) {
            double capacity_ratio = curr_weight / capacity[j];
            promising_factor += curr_weight * capacity_ratio;
            capacity_factor += remaining_capacity;
        }
    }

    promising_factor = (promising_factor > 0) ? profit[voisinage[index_item]] / promising_factor : 0;
    capacity_factor = 1.0 / (1.0 + capacity_factor);

    return promising_factor * capacity_factor;
}
