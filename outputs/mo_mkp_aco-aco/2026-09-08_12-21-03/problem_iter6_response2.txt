double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double sum_weights = 0;
    
    for (int i = 0; i < dimension; i++) {
        if (weights[i][index_item] > capacity[i]) {
            return 0; // Early stopping condition
        }
        sum_weights += weights[i][voisinage[index_item]] / capacity[i];
    }

    if (sum_weights == 0) {
        return 0; // Avoid division by zero
    }

    double h = profit[voisinage[index_item]] / sum_weights;
    return (profit[voisinage[index_item]] > 0) ? h : 0;
}
