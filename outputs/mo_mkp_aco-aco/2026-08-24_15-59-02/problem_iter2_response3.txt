double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double promising_factor = 1.0;

    for(int j = 0; j < dimension; j++){
        h += weights[j][voisinage[index_item]] / capacity[j];
    }

    if(h > 0){
        double neighbor_factor = 1.0 / nb_voisinage;
        double objective_factor = 0.7 * profit[voisinage[index_item]] / h;
        double remaining_capacity_factor = 0.3 * (1.0 - h / dimension);

        promising_factor = 0.5 * neighbor_factor + 0.3 * objective_factor + 0.2 * remaining_capacity_factor;
    } else {
        promising_factor = 0;
    }

    return promising_factor;
}
