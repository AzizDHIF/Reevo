double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double h_weights = 0;

    for (int j = 0; j < dimension; j++) {
        h_weights += weights[j][index_item] / capacity[j];
    }

    if (h_weights == 0) {
        return 0;
    }

    for (int k = 0; k < nb_voisinage; k++) {
        double obj_score = (weights[0][voisinage[k]] / h_weights) + profit[voisinage[k]];

        // Dynamic weight adjustments and non-linear transformations for efficiency
        obj_score = obj_score * obj_score * obj_score;

        h += obj_score;
    }

    return h;
}
