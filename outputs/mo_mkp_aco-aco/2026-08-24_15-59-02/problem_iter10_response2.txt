double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double h = 0;
    double total_weight = 0;

    // Calculate total normalized weight of the item
    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item] / capacity[j];
    }

    // If the total normalized weight is zero, return 0
    if (total_weight == 0) {
        return 0;
    }

    // Calculate heuristic value based on neighbors and profits
    for (int k = 0; k < nb_voisinage; k++) {
        double normalized_weight = weights[0][voisinage[k]] / total_weight;
        double obj_score = normalized_weight + profit[voisinage[k]];
        h += obj_score * obj_score; // Square for higher value emphasis
    }

    return h;
}
