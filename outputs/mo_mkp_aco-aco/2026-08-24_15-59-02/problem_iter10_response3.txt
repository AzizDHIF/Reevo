double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double h_weights = 0;

    // Calculate the sum of relative weights for the current item
    for (int j = 0; j < dimension; j++) {
        h_weights += weights[j][index_item] / capacity[j];
    }

    // If the sum of relative weights is zero, return 0 since the item cannot be included
    if (h_weights == 0) {
        return 0;
    }

    // Calculate the heuristic value by considering the relative weight, profit, and a non-linear transformation
    for (int k = 0; k < nb_voisinage; k++) {
        double relative_weight = weights[0][voisinage[k]] / h_weights;
        double obj_score = relative_weight + profit[voisinage[k]];
        h += obj_score * obj_score * relative_weight; // Dynamic weighting adjustment with non-linear transformation
    }

    return h;
}
