double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double h = 0, sum_weights = 0;

    for (int j = 0; j < dimension; j++)
    {
        sum_weights += weights[j][voisinage[index_item]] / capacity[j];
    }

    h = sum_weights / dimension;

    if (h < 0.000001) // Sparsify low weight elements
    {
        return 0;
    }

    return profit[voisinage[index_item]] / (h + 1);
}
