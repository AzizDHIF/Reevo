double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weighted_capacity = 0;
    double importance_factor = 0.75;
    
    for(int j = 0; j < dimension; j++){
        weighted_capacity += weights[j][index_item] / capacity[j]; 
    }
    
    double importance = profit[index_item] * 2 - weighted_capacity * 0.5;
    
    if(importance < 0) {
        return 0;
    } else {
        return importance * importance_factor;
    }
}
