double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double promising_item = 0.0;
    double total_weight = 0.0;

    // Combine factors to determine how promising it is to select an item
    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item];
    }

    double density = profit[index_item] / total_weight;

    // Calculate the "pressure" factor based on the available capacity and the item's weight
    double pressure = 0.0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            pressure += (weights[j][index_item] / capacity[j]);
        }
    }

    // Incorporate the neighborhood information
    double neighborhood_promising = 0.0;
    for (int i = 0; i < nb_voisinage; i++) {
        int voisin = voisinage[i];
        neighborhood_promising += profit[voisin];
    }
    neighborhood_promising /= (double) nb_voisinage;

    // Combine the factors
    promising_item = density * neighborhood_promising / (1.0 + pressure);

    // Sparsify the heuristic
    if (promising_item < 0.1 * neighborhood_promising) {
        promising_item = 0.0;
    }

    return promising_item;
}
