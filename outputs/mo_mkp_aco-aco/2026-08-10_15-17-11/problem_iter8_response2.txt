double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double max_capacity = 0.0;
    double sum_weights = 0.0;
    double utilization_ratio = 1.0;
    double max_neighbor_profit = 0.0;

    // Find max capacity
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > max_capacity) {
            max_capacity = capacity[j];
        }
        sum_weights += weights[j][index_item];
        if (capacity[j] != 0) {
            utilization_ratio *= (1 - weights[j][index_item] / capacity[j]);
        }
    }

    // Find max neighbor profit
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_sum_weights = 0.0;
        for (int j = 0; j < dimension; j++) {
            neighbor_sum_weights += weights[j][voisinage[i]];
        }
        if (neighbor_sum_weights <= max_capacity) {
            if (profit[voisinage[i]] > max_neighbor_profit) {
                max_neighbor_profit = profit[voisinage[i]];
            }
        }
    }

    // Calculate item utility and total weight factor
    double item_utility = (profit[index_item] - max_neighbor_profit) * utilization_ratio;
    double total_weight_factor = sum_weights * (1.0 / max_capacity);

    // Calculate heuristic value
    double h = item_utility - total_weight_factor;
    if (h < 0) {
        h = 0.0;
    }

    return h;
}
