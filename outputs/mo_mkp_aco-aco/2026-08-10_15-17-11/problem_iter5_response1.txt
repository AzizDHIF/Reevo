double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0.0;
    double max_capacity = capacity[0];
    for (int j = 1; j < dimension; j++) {
        if (capacity[j] > max_capacity) {
            max_capacity = capacity[j];
        }
    }
    double sum_weights = 0.0;
    double min_capacity = capacity[0];
    for (int j = 0; j < dimension; j++) {
        sum_weights += weights[j][index_item];
        if (capacity[j] < min_capacity) {
            min_capacity = capacity[j];
        }
    }
    double density = profit[index_item] / sum_weights;
    double pressure = sum_weights / max_capacity;
    double utilization_ratio = 1.0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] != 0) {
            utilization_ratio *= (1 - weights[j][index_item] / capacity[j]);
        }
    }
    double item_utility = density * utilization_ratio;
    double total_weight_factor = pressure * (1.0 - utilization_ratio);
    h = item_utility - total_weight_factor;
    if (h < 0) {
        h = 0.0;
    }
    // Sparsifying the heuristic
    if (sum_weights > max_capacity || utilization_ratio < 0.1) {
        h = 0.0;
    }
    return h;
}
