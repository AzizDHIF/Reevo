double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double total_weight = 0.0;
    double pressure = 0.0;

    // Calculate total weight and pressure
    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item];
        if (capacity[j] > 0) {
            pressure += (weights[j][index_item] / capacity[j]);
        }
    }

    // Calculate density
    double density = profit[index_item] / total_weight;

    // Combine the factors
    double promising_item = density / (1.0 + pressure);

    // Handle division by zero and negative values
    if (total_weight == 0 || promising_item < 0) {
        promising_item = 0.0;
    }

    return promising_item;
}
