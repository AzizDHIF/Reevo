double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double density = profit[index_item];
    for (int j = 0; j < dimension; j++) {
        h += weights[j][index_item] / capacity[j];
        density /= (1 + weights[j][index_item]);
    }
    double neighbor_density = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        double tmp = profit[voisinage[i]];
        for (int j = 0; j < dimension; j++) {
            tmp /= (1 + weights[j][voisinage[i]]);
        }
        neighbor_density += tmp;
    }
    double diversity = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        double similarity = 0;
        for (int j = 0; j < dimension; j++) {
            similarity += weights[j][index_item] * weights[j][voisinage[i]];
        }
        diversity += similarity;
    }
    double score = (density / (1 + h)) * (1 - diversity / nb_voisinage) * neighbor_density;
    // Sparsifying the heuristic by setting unpromising elements to zero
    if (score < 1e-6) {
        score = 0;
    }
    return score;
}
