double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ) 
{
    double h = 0.0;
    double density = profit[index_item];
    for (int j = 0; j < dimension; j++) {
        density /= weights[j][index_item];
    }
    double pressure = 0.0;
    for (int j = 0; j < dimension; j++) {
        pressure += (1 - (capacity[j] / weights[j][index_item]));
    }
    double utilization_ratio = 1.0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] != 0) {
            utilization_ratio *= (1 - weights[j][index_item] / capacity[j]);
        }
    }
    double neighbor_sum_profits = 0.0;
    for (int i = 0; i < nb_voisinage; i++) {
        neighbor_sum_profits += profit[voisinage[i]];
    }
    double item_utility = density * utilization_ratio * (1 - (neighbor_sum_profits / nb_voisinage));
    double total_weight_factor = 0.0;
    for (int j = 0; j < dimension; j++) {
        total_weight_factor += weights[j][index_item];
    }
    double total_weight_pressure = total_weight_factor * pressure;
    h = item_utility - total_weight_pressure;
    if (h < 0) {
        h = 0.0;
    }
    if (utilization_ratio < 0.1) {
        h = 0.0; // Sparsify the heuristic by setting unpromising elements to zero
    }
    return h;
}
