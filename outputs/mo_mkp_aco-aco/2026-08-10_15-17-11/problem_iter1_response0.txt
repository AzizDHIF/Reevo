double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double profit_item = profit[index_item];
    double weight_item = 0;
    for (int j = 0; j < dimension; j++) {
        weight_item += weights[j][index_item];
        double sum_weights_voisinage = 0;
        for (int i = 0; i < nb_voisinage; i++) {
            sum_weights_voisinage += weights[j][voisinage[i]];
        }
        h += (weights[j][index_item] + sum_weights_voisinage) / capacity[j];
    }
    double profit_over_weight_item = profit_item / weight_item;
    double h_normalized = h / dimension;
    return profit_over_weight_item / (1 + h_normalized);
}
