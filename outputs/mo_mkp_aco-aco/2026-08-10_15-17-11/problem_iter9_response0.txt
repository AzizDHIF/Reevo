double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) 
{
    double h = 0.0;
    double density = profit[index_item];
    for (int j = 0; j < dimension; j++) {
        density /= weights[j][index_item];
    }
    double min_capacity = capacity[0];
    for (int j = 1; j < dimension; j++) {
        if (capacity[j] < min_capacity) {
            min_capacity = capacity[j];
        }
    }
    double utilization_ratio = 1.0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] != 0) {
            utilization_ratio *= (1 - weights[j][index_item] / capacity[j]);
        }
    }
    int neighbor_max_profit_index = -1;
    double max_neighbor_profit = 0.0;
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_sum_weights = 0.0;
        for (int j = 0; j < dimension; j++) {
            neighbor_sum_weights += weights[j][voisinage[i]];
        }
        if (neighbor_sum_weights <= min_capacity) {
            if (profit[voisinage[i]] > max_neighbor_profit) {
                neighbor_max_profit_index = i;
                max_neighbor_profit = profit[voisinage[i]];
            }
        }
    }
    double item_utility = (profit[index_item] - max_neighbor_profit) * utilization_ratio;
    double total_weight_factor = 0.0;
    for (int j = 0; j < dimension; j++) {
        total_weight_factor += (weights[j][index_item] / capacity[j]);
    }
    total_weight_factor /= dimension;
    h = density * item_utility - total_weight_factor;
    if (h < 0) {
        h = 0.0;
    }
    return h;
}
