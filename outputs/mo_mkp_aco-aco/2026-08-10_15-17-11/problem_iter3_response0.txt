double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ) 
{
    double h = 0.0;
    double sum_weights = 0.0;
    double min_capacity = capacity[0];
    double max_capacity = capacity[0];
    for (int j = 0; j < dimension; j++) {
        sum_weights += weights[j][index_item];
        if (capacity[j] < min_capacity) {
            min_capacity = capacity[j];
        }
        if (capacity[j] > max_capacity) {
            max_capacity = capacity[j];
        }
    }

    double utilization_ratio = 1.0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] != 0) {
            utilization_ratio *= (1 - weights[j][index_item] / capacity[j]);
        }
    }

    double density = profit[index_item] / sum_weights;

    double neighbor_sum_profits = 0.0;
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_sum_weights = 0.0;
        for (int j = 0; j < dimension; j++) {
            neighbor_sum_weights += weights[j][voisinage[i]];
        }
        if (neighbor_sum_weights <= max_capacity) {
            neighbor_sum_profits += profit[voisinage[i]];
        }
    }

    double item_utility = density * utilization_ratio * profit[index_item];
    double pressure = (sum_weights / max_capacity);
    double neighbor_pressure = neighbor_sum_profits / max_capacity;

    h = item_utility - pressure - neighbor_pressure;

    if (h < 0) {
        h = 0.0;
    }

    return h;
}
