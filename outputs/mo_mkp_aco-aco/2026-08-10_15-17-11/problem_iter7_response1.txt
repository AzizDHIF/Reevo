double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0.0;
    double max_capacity = capacity[0];
    for (int j = 1; j < dimension; j++) {
        if (capacity[j] > max_capacity) {
            max_capacity = capacity[j];
        }
    }
    int neighbor_max_profit_index = -1;
    double max_neighbor_profit = 0.0;
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_sum_weights = 0.0;
        for (int j = 0; j < dimension; j++) {
            neighbor_sum_weights += weights[j][voisinage[i]];
        }
        if (neighbor_sum_weights <= max_capacity) {
            if (profit[voisinage[i]] > max_neighbor_profit) {
                neighbor_max_profit_index = i;
                max_neighbor_profit = profit[voisinage[i]];
            }
        }
    }
    double sum_weights = 0.0;
    double min_capacity = capacity[0];
    for (int j = 0; j < dimension; j++) {
        sum_weights += weights[j][index_item];
        if (capacity[j] < min_capacity) {
            min_capacity = capacity[j];
        }
    }
    double utilization_ratio = 1.0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] != 0) {
            utilization_ratio *= (1 - weights[j][index_item] / capacity[j]);
        }
    }
    double item_density = profit[index_item] / sum_weights;
    double pressure = sum_weights / max_capacity;
    double item_utility = item_density * (1 - pressure) * utilization_ratio;
    if (neighbor_max_profit_index != -1) {
        item_utility -= (max_neighbor_profit / sum_weights) * pressure;
    }
    if (item_utility < 0) {
        item_utility = 0.0;
    }
    h = item_utility;
    return h;
}
