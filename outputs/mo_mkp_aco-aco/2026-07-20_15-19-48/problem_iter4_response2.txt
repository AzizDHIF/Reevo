#include <stdio.h>

#define NBITEMS 10
#define dimension 3

double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double weight_sum = 0;
    double capacity_ratio = 1.0;
    double total_profit = 0;
    double total_weight = 0;

    // Calculate weight sum, total profit and total weight
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
        total_weight += weights[j][index_item] / capacity[j];
    }
    total_profit = profit[index_item];

    // Calculate capacity ratio considering all dimensions
    for (int j = 0; j < dimension; j++) {
        capacity_ratio *= (1 - weights[j][index_item] / capacity[j]);
        if (capacity_ratio < 0) capacity_ratio = 0;
    }

    // Calculate the promising value based on profit, weight, and capacity ratio
    double promising_value = total_profit * capacity_ratio;

    // Consider the influence of neighbors on the heuristic value
    double neighbor_influence = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_weight_sum = 0;
        for (int j = 0; j < dimension; j++) {
            neighbor_weight_sum += weights[j][voisinage[i]];
        }
        neighbor_influence += profit[voisinage[i]] / (weight_sum + neighbor_weight_sum + 1e-6);
    }

    // Combine various factors to determine the promise of the item
    h = (promising_value / (total_weight + 1)) + neighbor_influence / (nb_voisinage + 1e-6);

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 0 || weight_sum > 1) {
        h = 0;
    }

    return h;
}

int main() {
    double weights[dimension][NBITEMS];
    double capacity[dimension];
    int voisinage[NBITEMS];
    double profit[NBITEMS];

    // Initialize weights, capacity, voisinage, and profit
    for (int i = 0; i < dimension; i++) {
        capacity[i] = 1.0;
        for (int j = 0; j < NBITEMS; j++) {
            weights[i][j] = (double)rand() / RAND_MAX;
            if (i == 0) {
                profit[j] = (double)rand() / RAND_MAX;
                voisinage[j] = j;
            }
        }
    }

    int index_item = 0;
    int nb_voisinage = 3;

    double h = heuristic_v2(index_item, weights, capacity, nb_voisinage, voisinage, profit);

    printf("Heuristic value: %f\n", h);

    return 0;
}
