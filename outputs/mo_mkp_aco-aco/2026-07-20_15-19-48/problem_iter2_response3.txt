double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double weight_sum = 0;
    double capacity_ratio = 1.0;
    double profit_sum = 0;

    // Calculate weight sum, capacity ratio, and profit sum
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
        capacity_ratio *= (1 - weights[j][index_item] / capacity[j]);
    }
    profit_sum = profit[index_item];

    // Calculate the promising value based on profit, weight, and capacity ratio
    double promising_value = profit_sum * capacity_ratio;

    // Take into account the number of possible neighbours and their weights
    double neighbour_influence = 0;
    double neighbour_weight_sum = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        neighbour_influence += profit[voisinage[i]];
        for (int j = 0; j < dimension; j++) {
            neighbour_weight_sum += weights[j][voisinage[i]];
        }
    }

    // Calculate the average weight of neighbours
    double average_neighbour_weight = neighbour_weight_sum / (nb_voisinage * dimension);

    // Calculate the neighbour weight influence
    double neighbour_weight_influence = average_neighbour_weight / (weight_sum + 1);

    // Combine various factors
    return (promising_value / (weight_sum + 1)) + neighbour_influence * neighbour_weight_influence / (nb_voisinage * profit_sum + 1);
}
