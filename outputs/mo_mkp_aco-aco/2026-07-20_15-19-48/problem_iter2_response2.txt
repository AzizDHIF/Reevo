double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double profit_ratio = 0;
    double weight_ratio = 0;
    double neighbor_weight_ratio = 0;
    double min_capacity_ratio = 1.0;

    // Calculate the minimum capacity constraint among all dimensions for the item
    for (int j = 0; j < dimension; j++) {
        double capacity_ratio = weights[j][index_item] / capacity[j];
        if (capacity_ratio < min_capacity_ratio) {
            min_capacity_ratio = capacity_ratio;
        }
    }

    // Calculate the profit-to-weight ratio for the current item
    for (int j = 0; j < dimension; j++) {
        weight_ratio += weights[j][index_item] / capacity[j];
    }
    profit_ratio = profit[index_item] / (weight_ratio + 1e-6);

    // Calculate the average weight ratio of the neighbors
    for (int i = 0; i < nb_voisinage; i++) {
        for (int j = 0; j < dimension; j++) {
            neighbor_weight_ratio += weights[j][voisinage[i]] / capacity[j];
        }
    }
    neighbor_weight_ratio /= nb_voisinage;

    // Combine various factors to determine the promise of the item
    h = profit_ratio / (1 + neighbor_weight_ratio);

    // Consider the impact of the item's neighbors
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_profits = profit[voisinage[i]];
        double neighbor_weight_sum = 0;
        for (int j = 0; j < dimension; j++) {
            neighbor_weight_sum += weights[j][voisinage[i]];
        }
        h += neighbor_profits / (1 + neighbor_weight_sum);
    }

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 0 || weight_ratio > 1 || min_capacity_ratio > 1) {
        h = 0;
    }

    return h;
}
