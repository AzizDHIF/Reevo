double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) 
{
    double weight_sum = 0;
    double capacity_ratio = 1.0;
    double average_weight = 0.0;

    // Calculate weight sum and average
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
        capacity_ratio *= (1 - weights[j][index_item] / capacity[j]);
    }

    average_weight = weight_sum / dimension;

    // Calculate the promising value based on profit, capacity ratio, and average weight
    double promising_value = profit[index_item] * capacity_ratio / (average_weight + 1e-6);

    // Sparsify the heuristic by setting unpromising elements to zero
    if (promising_value < 0.1 * profit[index_item]) {
        return 0;
    }

    // Prioritize capacity ratio and balance with average weight
    double priority = capacity_ratio * (1 / (average_weight + 1e-6));

    // Take into account the number of possible neighbours and balance with dimension
    double neighbour_influence = nb_voisinage > 0 ? (double)nb_voisinage / (NBITEMS * dimension) : 0;

    // Combine various factors
    return promising_value * priority * (1 + neighbour_influence);
}
