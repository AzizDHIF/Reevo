double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) 
{
    double capacity_ratio = 1.0;
    double weight_sum = 0.0;

    // Prioritize capacity ratio and calculate weight sum
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
        capacity_ratio *= (1 - weights[j][index_item] / capacity[j]);
    }

    // Simplify calculations and minimize divisions
    double promising_value = profit[index_item] * capacity_ratio;
    if (weight_sum > 1.0 || promising_value < 0.0) {
        return 0.0;
    }

    // Take into account the number of possible neighbours and their profits
    double neighbour_influence = 0.0;
    for (int i = 0; i < nb_voisinage; i++) {
        neighbour_influence += profit[voisinage[i]];
    }

    // Combine factors and minimize divisions
    if (nb_voisinage > 0) {
        return promising_value + neighbour_influence / nb_voisinage;
    } else {
        return promising_value;
    }
}
