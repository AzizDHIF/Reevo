double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double weight_sum = 0;
    double capacity_ratio = 1.0;
    double max_capacity_constraint = 0;

    // Calculate weight sum, max capacity constraint and average
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
        if (weights[j][index_item] / capacity[j] > max_capacity_constraint) {
            max_capacity_constraint = weights[j][index_item] / capacity[j];
        }
        capacity_ratio *= (1 - weights[j][index_item] / capacity[j]);
    }

    // Calculate the promising value based on profit, weight, and capacity ratio
    double promising_value = profit[index_item] * capacity_ratio;

    // Sparsify the heuristic by setting unpromising elements to zero
    if (promising_value < 0.1 * weight_sum) {
        return 0;
    }

    // Take into account the number of possible neighbours and their weights
    double neighbour_influence = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbour_weight_sum = 0;
        double neighbour_capacity_ratio = 1.0;
        for (int j = 0; j < dimension; j++) {
            neighbour_weight_sum += weights[j][voisinage[i]];
            neighbour_capacity_ratio *= (1 - weights[j][voisinage[i]] / capacity[j]);
        }
        double neighbour_promising_value = profit[voisinage[i]] * neighbour_capacity_ratio;
        neighbour_influence += neighbour_promising_value / (weight_sum + 1);
    }

    // Combine various factors and consider capacity ratios and neighbour influences
    return (promising_value / (max_capacity_constraint + 1)) + neighbour_influence / nb_voisinage;
}
