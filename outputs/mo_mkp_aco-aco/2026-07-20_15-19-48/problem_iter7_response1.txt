double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) 
{
    double weight_sum = 0;
    double capacity_ratio = 1.0;
    double average_weight = 0;

    // Calculate weight sum and average
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
        capacity_ratio *= (1 - weights[j][index_item] / capacity[j]);
    }
    average_weight = weight_sum / dimension;

    // Calculate the promising value based on profit and capacity ratio
    double promising_value = profit[index_item] * capacity_ratio / (average_weight + 1e-9);

    // Sparsify the heuristic by setting unpromising elements to zero
    if (promising_value < 0.1 * (profit[index_item] / (average_weight + 1e-9))) {
        return 0;
    }

    // Prioritize capacity ratio
    double priority = capacity_ratio;

    // Take into account the number of possible neighbours
    double neighbour_influence = nb_voisinage > 0 ? (double)nb_voisinage / NBITEMS : 0;

    // Combine various factors with more emphasis on capacity ratio and profit
    return promising_value * priority * priority * (1 + neighbour_influence);
}
