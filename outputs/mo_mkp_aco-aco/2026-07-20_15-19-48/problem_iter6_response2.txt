double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double capacity_ratio = 1.0;
    double weight_sum = 0;

    // Calculate weight sum and prioritize capacity ratio
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
        capacity_ratio *= (1 - weights[j][index_item] / capacity[j]);
    }

    // Simplify calculations and calculate promising value
    double promising_value = profit[index_item] * capacity_ratio;

    // Minimize divisions and consider neighbor influence
    double neighbor_influence = 0;
    if (nb_voisinage > 0) {
        for (int i = 0; i < nb_voisinage; i++) {
            neighbor_influence += profit[voisinage[i]];
        }
        neighbor_influence /= (nb_voisinage + 1e-6);
    }

    // Combine factors and balance them
    h = promising_value + 0.5 * neighbor_influence;

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 0 || weight_sum > 1) {
        return 0;
    }

    return h;
}
