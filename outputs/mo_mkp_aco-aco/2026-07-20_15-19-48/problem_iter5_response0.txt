double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_sum = 0;
    double capacity_ratio = 1.0;

    // Calculate weight sum and average
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
        capacity_ratio *= (1 - weights[j][index_item] / capacity[j]);
    }

    // Calculate the promising value based on profit and capacity ratio
    double promising_value = profit[index_item] * capacity_ratio;

    // Sparsify the heuristic by setting unpromising elements to zero
    if (promising_value < 0.1 * weight_sum) {
        return 0;
    }

    // Prioritize capacity ratio
    double priority = capacity_ratio;

    // Take into account the number of possible neighbours
    double neighbour_influence = nb_voisinage > 0 ? (double)nb_voisinage / NBITEMS : 0;

    // Combine various factors
    return promising_value * priority * neighbour_influence;
}
