double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) 
{
    double h = 0;
    double weight_sum = 0;
    double capacity_ratio = 1.0;
    double profit_ratio = 0;

    // Calculate weight sum and average
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
        h += weights[j][voisinage[index_item]] / capacity[j];
        capacity_ratio *= (1 - weights[j][index_item] / capacity[j]);
        if (capacity[j] > 0) {
            profit_ratio += profit[index_item] / (weights[j][index_item] + 1);
        }
    }

    // Calculate the promising value based on profit, weight, and capacity ratio
    double promising_value = profit[index_item] * capacity_ratio * profit_ratio;

    // Sparsify the heuristic by setting unpromising elements to zero
    if (promising_value < 0.1 * weight_sum) {
        return 0;
    }

    // Take into account the number of possible neighbours and their weights
    double neighbour_influence = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbour_weight_sum = 0;
        for (int j = 0; j < dimension; j++) {
            neighbour_weight_sum += weights[j][voisinage[i]];
        }
        neighbour_influence += profit[voisinage[i]] / (neighbour_weight_sum + 1);
    }

    // Combine various factors
    return (promising_value / (h + 1)) + neighbour_influence / (nb_voisinage + 1);
}
