double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) 
{
    double h = 0;
    double capacity_ratio = 1.0;

    // Prioritize capacity ratio
    for (int j = 0; j < dimension; j++) {
        h += weights[j][index_item];
        capacity_ratio *= (1 - weights[j][index_item] / capacity[j]);
    }

    // Calculate the promising value based on profit and capacity ratio
    double promising_value = profit[index_item] * capacity_ratio / (h + 1);

    // Sparsify the heuristic by setting unpromising elements to zero
    if (promising_value < 0.1) {
        return 0;
    }

    // Take into account the number of possible neighbours and their profits
    double neighbour_influence = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        neighbour_influence += profit[voisinage[i]];
    }

    // Combine various factors, minimizing divisions
    return promising_value + neighbour_influence / (nb_voisinage + 1);
}
