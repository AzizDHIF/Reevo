double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_sum = 0;
    double capacity_ratio = 1.0;
    double total_profit = profit[index_item];

    // Calculate weight sum and capacity ratio
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item] / capacity[j];
        capacity_ratio *= (1 - weights[j][index_item] / capacity[j]);
    }

    // Calculate the promising value based on profit, weight, and capacity ratio
    double promising_value = total_profit * capacity_ratio;

    // Consider the influence of neighbors on the heuristic value
    double neighbor_influence = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_weight_sum = 0;
        for (int j = 0; j < dimension; j++) {
            neighbor_weight_sum += weights[j][voisinage[i]] / capacity[j];
        }
        neighbor_influence += profit[voisinage[i]] / (weight_sum + neighbor_weight_sum + 1e-6);
    }

    // Combine various factors to determine the promise of the item
    double h = (promising_value / (weight_sum + 1)) + neighbor_influence / (nb_voisinage + 1e-6);

    // Normalize the heuristic value to be between 0 and 1
    h = h / (h + 1);

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 0 || weight_sum > 1) {
        h = 0;
    }

    return h;
}
