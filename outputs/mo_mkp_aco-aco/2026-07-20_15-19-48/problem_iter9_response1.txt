double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) 
{
    double weight_sum = 0;
    double capacity_ratio = 1.0;
    double average_weight = 0;

    // Calculate weight sum and average
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
        capacity_ratio *= (1 - weights[j][index_item] / capacity[j]);
    }
    average_weight = weight_sum / dimension;

    // Consider neighbors in the heuristic calculation
    double neighbor_factor = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        int neighbor_index = voisinage[i];
        double neighbor_weight_sum = 0;
        for (int j = 0; j < dimension; j++) {
            neighbor_weight_sum += weights[j][neighbor_index];
        }
        double neighbor_average_weight = neighbor_weight_sum / dimension;
        neighbor_factor += profit[neighbor_index] / (neighbor_average_weight + 1e-9);
    }
    neighbor_factor /= (double)nb_voisinage;

    // Calculate the promising value based on profit, capacity ratio, and neighbor factor
    double promising_value = profit[index_item] * capacity_ratio / (average_weight + 1e-9) * (1 + 0.5 * neighbor_factor);

    // Sparsify the heuristic by setting unpromising elements to zero
    if (promising_value < 0.1 * (profit[index_item] / (average_weight + 1e-9))) {
        return 0;
    }

    // Prioritize capacity ratio
    double priority = capacity_ratio;

    // Combine various factors with more emphasis on capacity ratio and profit
    return promising_value * priority * priority * (1 + 0.2 * neighbor_factor);
}
