double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double promising_factor = 1.0; // Promising factor initialization

    for (int j = 0; j < dimension; j++) {
        if (capacity[j] != 0.0) { // Avoid division by zero
            h += weights[j][voisinage[index_item]] / capacity[j];
        } else {
            promising_factor = 0.0; // If division by zero is encountered, set promising factor to 0
        }
    }

    return promising_factor * profit[voisinage[index_item]] / (h + 0.0001); // Sparsifying and preventing division by close-to-zero values
}
