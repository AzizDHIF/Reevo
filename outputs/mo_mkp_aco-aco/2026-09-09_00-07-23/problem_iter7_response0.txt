double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double sum_weights = 0;
    
    // Calculate the sum of weights for the current item over all dimensions
    for (int j = 0; j < dimension; j++) {
        sum_weights += weights[j][index_item];
    }
    
    // Normalize weights directly by the sum of weights
    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / sum_weights;
    }
    
    // Promising selection based on profit-to-weight ratio
    return profit[voisinage[index_item]] / h;
}
