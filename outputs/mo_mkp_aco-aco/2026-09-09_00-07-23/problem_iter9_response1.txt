double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double h = 0;
    double normalized_weight = 0;
    
    for (int j = 0; j < dimension; j++) {
        normalized_weight += weights[j][index_item] / capacity[j];
    }
    
    if (normalized_weight == 0) {
        return 0;
    }
    
    int promising = 1;
    for (int i = 0; i < nb_voisinage; i++) {
        int neighbor = voisinage[i];
        double neighbor_weight = 0;
        
        for (int j = 0; j < dimension; j++) {
            neighbor_weight += weights[j][neighbor] / capacity[j];
        }
        
        if (neighbor_weight == 0) {
            continue;
        }
        
        if (profit[neighbor] / neighbor_weight < profit[index_item] / normalized_weight) {
            promising = 0;
            break;
        }
    }
    
    if (promising == 0) {
        return 0;
    }
    
    return profit[index_item] / normalized_weight;
}
