double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] == 0) {
            return 0; // Sparsify by setting unpromising elements to zero
        }
        h += weights[j][index_item] / capacity[j];
    }
    if (h == 0) {
        return 0;
    }
    return profit[index_item] / h; // Consider item influence and normalize by the combined weight and normalized profit
}
