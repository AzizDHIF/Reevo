double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;

    for (int j = 0; j < dimension; j++) {
        double weight = weights[j][voisinage[index_item]];
        double remaining_capacity = capacity[j];

        if (weight != 0 && remaining_capacity > 0) {
            h += weight / remaining_capacity;
        }
    }

    return (h != 0) ? profit[voisinage[index_item]] / h : 0;
}
