double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double h = 0;
    double weight_sum = 0;
    double profit_sum = 0;
    double neighbor_sum = 0;

    // Calculate the total weight and profit of the item and its neighbors
    for(int j = 0; j < dimension; j++){
        weight_sum += weights[j][index_item];
    }
    profit_sum = profit[index_item];

    // Calculate the sum of weights and profits of the neighbors
    for(int i = 0; i < nb_voisinage && i < NBITEMS; i++){
        for(int j = 0; j < dimension; j++){
            neighbor_sum += weights[j][voisinage[i]];
        }
        profit_sum += profit[voisinage[i]];
    }

    // Calculate the heuristic value based on the item's weight, profit, and its neighbors' influence
    if(weight_sum > 0){
        h = profit_sum / (weight_sum + EPSILON);
    }

    // Check the capacity constraint and adjust the heuristic value accordingly
    double capacity_left = 1;
    for(int j = 0; j < dimension; j++){
        if(capacity[j] < weights[j][index_item]){
            capacity_left -= weights[j][index_item];
        }
    }
    if(capacity_left < 0){
        h *= (1 + capacity_left);
    }

    return h;
}
