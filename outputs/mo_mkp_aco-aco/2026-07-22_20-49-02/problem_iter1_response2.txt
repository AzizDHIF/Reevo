#include <stdio.h>
#include <limits.h>

#define NBITEMS 100
#define dimension 5

// Define a function to compute the heuristic value
double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Initialize variables to keep track of the weight and profit of neighboring items
    double neighbor_weight = 0.0;
    double neighbor_profit = 0.0;

    // Compute the weight and profit of neighboring items
    for (int i = 0; i < nb_voisinage; i++) {
        for (int j = 0; j < dimension; j++) {
            neighbor_weight += weights[j][voisinage[i]];
        }
        neighbor_profit += profit[voisinage[i]];
    }

    // Compute the average weight and profit of neighboring items
    double avg_neighbor_weight = neighbor_weight / nb_voisinage;
    double avg_neighbor_profit = neighbor_profit / nb_voisinage;

    // Initialize the heuristic value
    double h = 0.0;

    // Compute the heuristic value based on multiple factors
    for (int j = 0; j < dimension; j++) {
        h += (weights[j][index_item] / capacity[j]) * (1 + (avg_neighbor_profit / (1 + avg_neighbor_weight)));
    }

    // Compute the final heuristic value by dividing by the weight of the item
    double final_h = (profit[index_item] / (1 + h));

    // Apply sparsification to the heuristic value by setting it to zero if it's below a certain threshold
    double threshold = 0.01;
    if (final_h < threshold) {
        final_h = 0.0;
    }

    return final_h;
}

int main() {
    double weights[dimension][NBITEMS];
    double capacity[dimension];
    int voisinage[NBITEMS];
    double profit[NBITEMS];

    // Initialize the weights, capacity, voisinage, and profit arrays
    for (int i = 0; i < dimension; i++) {
        capacity[i] = 1.0;
        for (int j = 0; j < NBITEMS; j++) {
            weights[i][j] = (double)rand() / RAND_MAX;
        }
    }

    for (int i = 0; i < NBITEMS; i++) {
        voisinage[i] = i;
        profit[i] = (double)rand() / RAND_MAX;
    }

    int index_item = 0;
    int nb_voisinage = 10;

    double h = heuristic_v2(index_item, weights, capacity, nb_voisinage, voisinage, profit);
    printf("%f\n", h);

    return 0;
}
