double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    
    // Calculate the sum of weight ratios for each dimension
    for (int j = 0; j < dimension; j++) {
        double weight_ratio = weights[j][voisinage[index_item]] / capacity[j];
        h += weight_ratio;
        
        // Early pruning if weight ratio exceeds a threshold
        if (weight_ratio > 1.5) {
            return 0;
        }
    }
    
    // Return the profit to weight sum ratio multiplied by the inverse of the normalized number of neighbors
    return (profit[voisinage[index_item]] / h) * (1.0 / nb_voisinage);
}
