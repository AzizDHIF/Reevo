double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double h = 0;
    for (int j = 0; j < dimension; j++) {
        double weight_ratio = weights[j][voisinage[index_item]] / capacity[j];
        h += weight_ratio;
        // Limit unnecessary computations by early stopping if weight ratio exceeds a threshold
        if (weight_ratio > 1.5) {
            return 0; // Return 0 if weight ratio exceeds the threshold
        }
    }
    // Calculate the heuristic value based on profit to weight sum ratio and normalized number of neighbors
    return (profit[voisinage[index_item]] / h) * (1.0 / nb_voisinage);
}
