double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_factor = 0;
    double profit_factor = 0;

    for (int j = 0; j < dimension; j++) {
        weight_factor += weights[j][voisinage[index_item]] / capacity[j];
    }

    if (weight_factor < 1e-6) { // Check for nearly zero weight factor to avoid division by zero
        return 0;
    } else {
        profit_factor = profit[voisinage[index_item]] / weight_factor;
        return profit_factor;
    }
}
