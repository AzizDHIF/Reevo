double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ) {
    if (nb_voisinage <= 0) {
        return 0.0;
    }
    
    double total_weight_factor = 0;
    for (int j = 0; j < dimension; j++) {
        total_weight_factor += weights[j][index_item] / capacity[j];
    }
    
    if (total_weight_factor == 0) {
        return 0.0;
    }
    
    return profit[index_item] / total_weight_factor;
}
