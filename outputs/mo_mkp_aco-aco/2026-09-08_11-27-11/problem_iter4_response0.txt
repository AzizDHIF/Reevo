double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double weight_factor = 0;
    double profit_factor = 0;

    for (int j = 0; j < dimension; j++) {
        weight_factor += weights[j][index_item] / capacity[j];
    }

    if (weight_factor == 0) {
        return 0;
    } else {
        profit_factor = profit[index_item] / weight_factor;
        return profit_factor;
    }
}
