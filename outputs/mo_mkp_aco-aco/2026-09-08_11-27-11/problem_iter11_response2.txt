double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double contribution = 0;
    
    for (int j = 0; j < dimension; j++) {
        double normalized_weight = weights[j][voisinage[index_item]] / capacity[j];
        contribution += normalized_weight;
    }
    
    if (contribution == 0) {
        return 0;
    }
    
    return profit[voisinage[index_item]] / contribution;
}
