double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double ratio = 0.7;
    double weight_factor = 0.8;
    double profit_sum = 0;
    double weight_sum = 0;
    
    for(int j=0; j<dimension; j++){
        weight_sum += weights[j][voisinage[index_item]] / capacity[j];
    }
    
    for(int i=0; i<nb_voisinage; i++){
        profit_sum += profit[voisinage[i]];
    }
    
    double promise = profit[voisinage[index_item]] * ratio * (1.0 / (weight_sum * weight_factor)) + (profit[voisinage[index_item]] / profit_sum);
    
    return promise <= 0 ? 0 : promise;
}
