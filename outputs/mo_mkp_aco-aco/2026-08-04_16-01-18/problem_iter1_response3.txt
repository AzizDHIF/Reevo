double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double weight_sum = 0;
    double profit_sum = 0;
    double density = 0;

    // combine various factors to determine how promising it is to select an item
    for (int i = 0; i < nb_voisinage; i++) {
        int item_index = voisinage[i];
        weight_sum += weights[0][item_index]; // consider only one dimension for simplicity
        profit_sum += profit[item_index];
    }

    // calculate the average weight and profit of the neighbors
    double avg_weight = weight_sum / nb_voisinage;
    double avg_profit = profit_sum / nb_voisinage;

    // calculate the density of the item
    for (int j = 0; j < dimension; j++) {
        h = h + weights[j][index_item] / capacity[j];
        density += weights[j][index_item];
    }

    // try sparsifying the heuristic by setting unpromising elements to zero
    if (h > 1 || density > 1) {
        return 0; // set to zero if the item is too heavy or dense
    }

    // calculate the heuristic value based on the combined factors
    return profit[index_item] / (h + 1) * (avg_profit / (avg_weight + 1));
}
