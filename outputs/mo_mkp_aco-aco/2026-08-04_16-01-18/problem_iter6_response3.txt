double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double density = 0;
    double total_weight = 0;
    double total_profit = 0;

    // Calculate total weight and profit of neighbors
    for(int i = 0; i < nb_voisinage; i++){
        int neighbor = voisinage[i];
        for(int j = 0; j < dimension; j++){
            total_weight += weights[j][neighbor];
            h += weights[j][neighbor] / capacity[j];
        }
        total_profit += profit[neighbor];
    }

    // Calculate density of the current item, considering capacity
    for(int j = 0; j < dimension; j++){
        density += (weights[j][index_item] / capacity[j]);
    }

    // Normalize by the maximum possible weight
    double max_weight = 0;
    for(int i = 0; i < NBITEMS; i++) {
        double item_weight = 0;
        for(int j = 0; j < dimension; j++) {
            item_weight += weights[j][i];
        }
        if (item_weight > max_weight) {
            max_weight = item_weight;
        }
    }

    // Calculate heuristic value, considering density and profit
    double heuristic_value = (profit[index_item] / (density + 1e-6)) * ((total_profit / (total_weight + 1e-6)) + 1);

    // Normalize heuristic value by the maximum possible profit
    double max_profit = 0;
    for(int i = 0; i < NBITEMS; i++) {
        if (profit[i] > max_profit) {
            max_profit = profit[i];
        }
    }
    heuristic_value /= (max_profit + 1e-6);

    // Sparsify the heuristic
    if(heuristic_value < 0.1 * (profit[index_item] / (max_profit + 1e-6))){
        heuristic_value = 0;
    }

    return heuristic_value;
}
