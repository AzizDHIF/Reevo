double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double density = 0;
    double total_weight = 0;
    double total_profit = 0;
    double local_factor = 0;
    double global_factor = 0;

    // Calculate total weight and profit of neighbors
    for(int i = 0; i < nb_voisinage; i++){
        int neighbor = voisinage[i];
        for(int j = 0; j < dimension; j++){
            total_weight += weights[j][neighbor];
            h += weights[j][neighbor] / capacity[j];
        }
        total_profit += profit[neighbor];
    }

    // Calculate density of the current item
    for(int j = 0; j < dimension; j++){
        density += weights[j][index_item] / capacity[j];
    }

    // Calculate local factor (profit-to-density ratio)
    if(density > 0) {
        local_factor = profit[index_item] / density;
    } else {
        local_factor = profit[index_item];
    }

    // Calculate global factor (average profit-to-weight ratio of neighbors)
    if(total_weight > 0) {
        global_factor = total_profit / total_weight;
    } else {
        global_factor = total_profit;
    }

    // Calculate heuristic value as a balance between local and global factors
    double heuristic_value = local_factor * (global_factor + 1);

    // Sparsify the heuristic
    if(heuristic_value < 0.1 * profit[index_item]){
        heuristic_value = 0;
    }

    return heuristic_value;
}
