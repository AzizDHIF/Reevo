double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double density = 0;
    double neighbor_impact = 0;
    double total_weight = 0;
    double total_profit = 0;

    // Calculate the impact of neighbors
    for(int i = 0; i < nb_voisinage; i++){
        int neighbor = voisinage[i];
        for(int j = 0; j < dimension; j++){
            total_weight += weights[j][neighbor];
            h += weights[j][neighbor] / capacity[j];
        }
        total_profit += profit[neighbor];
    }

    // Calculate the average impact of neighbors
    if (nb_voisinage > 0) {
        neighbor_impact = total_profit / (total_weight + 1e-6);
    }

    // Calculate the density of the current item
    for(int j = 0; j < dimension; j++){
        density += weights[j][index_item] / capacity[j];
    }

    // Calculate the heuristic value considering the density and neighbors' impact
    double heuristic_value = (profit[index_item] / (density + 1e-6)) * (neighbor_impact + 1);

    // Sparsify the heuristic by setting low values to zero
    if(heuristic_value < 0.1 * profit[index_item]){
        heuristic_value = 0;
    }

    return heuristic_value;
}
