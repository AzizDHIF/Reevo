double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double density = 0;
    double total_weight = 0;
    double total_profit = 0;

    // Calculate total weight and profit of neighbors
    for(int i = 0; i < nb_voisinage; i++){
        int neighbor = voisinage[i];
        for(int j = 0; j < dimension; j++){
            total_weight += weights[j][neighbor];
            h += weights[j][neighbor] / capacity[j];
        }
        total_profit += profit[neighbor];
    }

    // Calculate density of the current item considering capacity
    for(int j = 0; j < dimension; j++){
        density += (weights[j][index_item] / capacity[j]);
    }

    // Check for division by zero
    if (density < 1e-6 || total_weight < 1e-6) {
        return 0;
    }

    // Calculate heuristic value considering density, capacity and neighbors
    double heuristic_value = (profit[index_item] / density) * ((total_profit / (total_weight + 1e-6)) + 1);

    // Consider the current item's weight and capacity ratio
    double item_weight_capacity_ratio = 0;
    for(int j = 0; j < dimension; j++){
        item_weight_capacity_ratio += (weights[j][index_item] / capacity[j]);
    }
    heuristic_value *= item_weight_capacity_ratio;

    // Sparsify the heuristic
    if(heuristic_value < 0.1 * profit[index_item]){
        heuristic_value = 0;
    }

    return heuristic_value;
}
