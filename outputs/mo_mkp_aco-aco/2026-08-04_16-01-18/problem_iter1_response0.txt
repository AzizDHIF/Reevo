double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double density = 0;
    double min_excess = 1e10;

    // Calculate the total weight and minimum excess for the item and its neighbors
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_weight = 0;
        double neighbor_excess = 0;
        for (int j = 0; j < dimension; j++) {
            neighbor_weight += weights[j][voisinage[i]];
            if (weights[j][voisinage[i]] > capacity[j]) {
                neighbor_excess += weights[j][voisinage[i]] - capacity[j];
            }
        }
        density += profit[voisinage[i]] / neighbor_weight;
        if (neighbor_excess < min_excess) {
            min_excess = neighbor_excess;
        }
    }

    // Penalize the item if its neighbors have high excess
    if (min_excess > 0) {
        density /= (1 + min_excess);
    }

    // Calculate the combined heuristic value
    for (int j = 0; j < dimension; j++) {
        h += weights[j][index_item] / capacity[j];
    }
    double heuristic_value = profit[index_item] / (h + 1e-10) * density;

    // Sparsify the heuristic
    if (heuristic_value < 0.1) {
        return 0;
    }

    return heuristic_value;
}
