double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ) 
{
    double h = 0;
    double density = 0;
    double total_weight = 0;
    double total_profit = 0;

    // Simplify calculations
    for(int i = 0; i < nb_voisinage; i++){
        int neighbor = voisinage[i];
        total_weight += weights[0][neighbor]; // Focus on key factors, consider only one dimension
        total_profit += profit[neighbor];
    }

    // Simplify density calculation
    density = weights[0][index_item]; // Focus on key factors, consider only one dimension

    // Avoid divisions by near-zero values
    double remaining_capacity = 1; // Consider the remaining capacity for one dimension
    for(int j = 0; j < dimension; j++){
        if(capacity[j] > 0){
            remaining_capacity *= capacity[j];
            break;
        }
    }

    // Calculate heuristic value
    double heuristic_value = (profit[index_item] / (density + 1e-6)) * ((total_profit / (total_weight + 1e-6)) + 1) * remaining_capacity;

    // Sparsify the heuristic
    if(heuristic_value < 0.1 * profit[index_item]){
        heuristic_value = 0;
    }

    return heuristic_value;
}
