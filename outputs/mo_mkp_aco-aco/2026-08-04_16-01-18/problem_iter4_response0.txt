double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double total_weight = 0;
    double total_profit = 0;
    double max_capacity = 0;

    // Calculate total weight and profit of neighbors
    for(int i = 0; i < nb_voisinage; i++){
        int neighbor = voisinage[i];
        for(int j = 0; j < dimension; j++){
            total_weight += weights[j][neighbor];
            if(capacity[j] > max_capacity) {
                max_capacity = capacity[j];
            }
        }
        total_profit += profit[neighbor];
    }

    // Calculate heuristic value considering multi-dimensions and neighbors' total weight
    double heuristic_value = 0;
    if (max_capacity > 1e-6 && total_weight + 1e-6 > 0) {
        double density = 0;
        for(int j = 0; j < dimension; j++){
            density += weights[j][index_item];
        }
        double ratio = density / max_capacity;
        heuristic_value = (profit[index_item] * (total_profit / (total_weight + 1e-6))) / (ratio + 1e-6);
    }

    // Sparsify the heuristic
    if(heuristic_value < 0.1 * profit[index_item]){
        heuristic_value = 0;
    }

    return heuristic_value;
}
