// Define the dimension of the weights
#define dimension 3
#define NBITEMS 10

// Heuristic function that combines various factors to determine how promising it is to select an item
double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate a weighted sum of the item's weights per dimension
    double weight_sum = 0;
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item] / capacity[j];
    }

    // Calculate the item's profit-to-weight ratio
    double profit_ratio = profit[index_item] / weight_sum;

    // Incorporate the item's neighbourhood
    double neighbourhood_score = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        neighbourhood_score += profit[voisinage[i]] / weight_sum;
    }
    neighbourhood_score /= nb_voisinage;

    // Calculate a final heuristic score by combining the profit ratio and neighbourhood score
    double heuristic_score = profit_ratio * (1 + neighbourhood_score);

    // Sparsify the heuristic by setting unpromising elements to zero
    if (heuristic_score < 0.5) {
        heuristic_score = 0;
    }

    return heuristic_score;
}

// Define a variant of the heuristic function that emphasizes neighbourhood influence
double heuristic_v3(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double neighbourhood_score = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbour_weight_sum = 0;
        for (int j = 0; j < dimension; j++) {
            neighbour_weight_sum += weights[j][voisinage[i]] / capacity[j];
        }
        neighbourhood_score += profit[voisinage[i]] / neighbour_weight_sum;
    }
    neighbourhood_score /= nb_voisinage;

    double item_weight_sum = 0;
    for (int j = 0; j < dimension; j++) {
        item_weight_sum += weights[j][index_item] / capacity[j];
    }

    double heuristic_score = neighbourhood_score * profit[index_item] / item_weight_sum;

    if (heuristic_score < 0.5) {
        heuristic_score = 0;
    }

    return heuristic_score;
}
