double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double total_weight = 0;
    double total_remaining_capacity = 0;
    double total_profit_neighborhood = 0;

    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item];
        total_remaining_capacity += capacity[j] - weights[j][index_item];
    }

    for (int i = 0; i < nb_voisinage; i++) {
        total_profit_neighborhood += profit[voisinage[i]];
    }

    double weight_to_capacity_ratio = (total_remaining_capacity > 0) ? total_weight / total_remaining_capacity : 0;
    double average_profit_neighborhood = (nb_voisinage > 0) ? total_profit_neighborhood / nb_voisinage : 0;

    double h1 = profit[index_item] / (weight_to_capacity_ratio + 1e-6);
    double h2 = average_profit_neighborhood / (weight_to_capacity_ratio + 1e-6);
    double h3 = profit[index_item] * average_profit_neighborhood;

    double capacity_consistency = 1;
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > capacity[j]) {
            capacity_consistency -= 0.2;
        }
    }

    // consider capacity constraints
    double capacity_constraint_factor = 1;
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > 0.8 * capacity[j]) {
            capacity_constraint_factor -= 0.1;
        }
    }

    // consider neighborhood proximity
    double neighborhood_proximity_factor = 1;
    for (int i = 0; i < nb_voisinage; i++) {
        if (profit[voisinage[i]] < 0.5 * profit[index_item]) {
            neighborhood_proximity_factor -= 0.05;
        }
    }

    // sparsify the heuristic
    if (weight_to_capacity_ratio > 2 || average_profit_neighborhood < 0.5 * profit[index_item] || capacity_consistency < 0) {
        return 0;
    }

    return (h1 * 0.4 + h2 * 0.3 + h3 * 0.3) * capacity_consistency * capacity_constraint_factor * neighborhood_proximity_factor;
}
