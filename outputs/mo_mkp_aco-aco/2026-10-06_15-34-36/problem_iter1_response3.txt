double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double h=0, promisibility=1;
    for(int j=0; j<dimension; j++){
        if(weights[j][voisinage[index_item]] <= 0 || capacity[j] <= 0) {
            promisibility = 0; // If any weight or capacity is non-positive, set promisibility to 0
            break;
        }
        h += weights[j][voisinage[index_item]] / capacity[j]; 
    }
    return promisibility * profit[voisinage[index_item]] / h;
}
