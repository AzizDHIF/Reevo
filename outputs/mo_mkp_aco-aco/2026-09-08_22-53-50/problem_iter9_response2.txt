double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0.0;
    
    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / (capacity[j] + 0.000001);
    }
    
    double promising_factor = 0.6 * profit[voisinage[index_item]] / h;
    double weight_factor = 0.4 * (1.0 - h / dimension);
    
    double promising_threshold = 0.001;
    double weight_threshold = 0.001;
    
    if (promising_factor < promising_threshold || weight_factor < weight_threshold) {
        return 0.0;
    }
    
    return promising_factor + weight_factor;
}
