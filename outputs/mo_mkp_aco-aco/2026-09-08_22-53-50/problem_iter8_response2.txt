double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0.0;

    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / (capacity[j] + 0.000001);
    }

    double normalized_profit = profit[voisinage[index_item]] / (double)(NBITEMS); // Normalize profit
    double promising_factor = 0.6 * normalized_profit / h; // Prioritize promising factor more
    double weight_factor = 0.4 * (1.0 - h / dimension); // Reduced weight factor contribution

    return promising_factor + weight_factor;
}
