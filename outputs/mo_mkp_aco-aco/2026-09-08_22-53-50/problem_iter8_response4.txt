double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double promising_factor = 0.0, weight_factor = 0.0;

    for (int j = 0; j < dimension; j++) {
        promising_factor += weights[j][voisinage[index_item]] / (capacity[j] + 1e-6);
    }

    promising_factor = 0.6 * profit[voisinage[index_item]] / promising_factor;
    weight_factor = 0.4 * (1.0 - promising_factor / dimension);

    return promising_factor + weight_factor;
}
