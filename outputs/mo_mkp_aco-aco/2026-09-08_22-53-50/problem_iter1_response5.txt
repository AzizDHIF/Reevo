double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double h=0, total_capacity = 0, total_weight = 0, total_score = 0;
    
	for(int j=0;j<dimension;j++){
		h += weights[j][voisinage[index_item]]/capacity[j]; 
		total_capacity += capacity[j];
		total_weight += weights[j][index_item];
	}
	
    total_score = profit[voisinage[index_item]] / total_weight + total_capacity - h; 
    
    if(total_score < 0) {
	    return 0;
	}
	
    return total_score;
}
