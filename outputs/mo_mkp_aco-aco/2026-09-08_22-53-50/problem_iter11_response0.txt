double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0.0;
    
    for (int j = 0; j < dimension; j++) {
        h += 1.0 - (weights[j][voisinage[index_item]] / capacity[j]);
    }
    
    double promising_factor = 0.8 * profit[voisinage[index_item]] / (h + 1e-6);
    double weight_factor = 0.2 * (1.0 - h / dimension);
    
    return promising_factor + weight_factor;
}
