double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double sum_weights = 0;
    double sum_capacities = 0;
    double average_capacity = 0;
    double max_capacity_violation = 0;

    // Calculate sum of weights and capacities
    for (int j = 0; j < dimension; j++) {
        sum_weights += weights[j][index_item];
        sum_capacities += capacity[j];
    }

    // Calculate average capacity
    if (sum_capacities > 0) {
        average_capacity = sum_capacities / dimension;
    }

    // Calculate max capacity violation for neighbors
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_violation = 0;
        for (int j = 0; j < dimension; j++) {
            if (capacity[j] - weights[j][voisinage[i]] < 0) {
                double violation = weights[j][voisinage[i]] - capacity[j];
                if (violation > neighbor_violation) {
                    neighbor_violation = violation;
                }
            }
        }
        if (neighbor_violation > max_capacity_violation) {
            max_capacity_violation = neighbor_violation;
        }
    }

    // Calculate heuristic value
    if (sum_weights < sum_capacities && max_capacity_violation == 0) {
        h = profit[index_item] * (1 - sum_weights / sum_capacities) / dimension;
    } else {
        h = 0;
    }

    // Penalize if capacity is exceeded
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] - weights[j][index_item] < 0) {
            h -= weights[j][index_item] - capacity[j];
        }
    }

    return h;
}
