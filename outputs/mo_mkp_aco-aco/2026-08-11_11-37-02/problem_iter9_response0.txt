double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ) 
{
    // Prioritize constraint satisfaction and simplify calculations
    double weight_to_capacity_ratio = 0;
    double valid_dimensions = 0;
    double max_weight_to_capacity_ratio = 0;
    for(int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            if (weights[j][index_item] > capacity[j]) {
                return 0; // immediately return 0 if any dimension's constraint is violated
            } else {
                double ratio = weights[j][index_item] / capacity[j];
                weight_to_capacity_ratio += ratio;
                valid_dimensions++;
                if (ratio > max_weight_to_capacity_ratio) {
                    max_weight_to_capacity_ratio = ratio;
                }
            }
        }
    }

    if (valid_dimensions == 0) {
        return 0; // if no valid dimensions, the item is not promising
    }

    // Simplify calculations
    weight_to_capacity_ratio /= valid_dimensions;

    // Simplify neighbor comparison
    double best_neighbor_profit = 0;
    if (nb_voisinage > 0) {
        for (int i = 0; i < nb_voisinage; i++) {
            if (profit[voisinage[i]] > best_neighbor_profit) {
                best_neighbor_profit = profit[voisinage[i]];
            }
        }
    }

    // Combine various factors to determine how promising it is to select an item
    double h = profit[index_item] / (weight_to_capacity_ratio + 1e-9) / (max_weight_to_capacity_ratio + 1e-9);
    if (best_neighbor_profit > 0) {
        h /= (best_neighbor_profit / profit[index_item] + 1e-9);
    }

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 1e-6) { // a small threshold to determine if an item is unpromising
        h = 0;
    }

    return h;
}
