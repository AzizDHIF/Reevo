double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double sum_weights = 0;
    double sum_capacities = 0;
    double average_capacity = 0;
    double max_capacity_violation = 0;

    for (int j = 0; j < dimension; j++) {
        sum_weights += weights[j][index_item];
        sum_capacities += capacity[j];
        if (capacity[j] - weights[j][index_item] < 0) {
            max_capacity_violation = max_capacity_violation > weights[j][index_item] - capacity[j] ? max_capacity_violation : weights[j][index_item] - capacity[j];
        }
    }

    if (sum_capacities > 0) {
        average_capacity = sum_capacities / dimension;
    }

    double neighbor_violation = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        double temp_max_capacity_violation = 0;
        for (int j = 0; j < dimension; j++) {
            if (capacity[j] - weights[j][voisinage[i]] < 0) {
                temp_max_capacity_violation = temp_max_capacity_violation > weights[j][voisinage[i]] - capacity[j] ? temp_max_capacity_violation : weights[j][voisinage[i]] - capacity[j];
            }
        }
        neighbor_violation += (temp_max_capacity_violation > 0) ? temp_max_capacity_violation : 0;
    }

    if (sum_weights < sum_capacities && max_capacity_violation == 0 && neighbor_violation == 0) {
        h = profit[index_item] * (1 - sum_weights / sum_capacities) / dimension;
    } else {
        h = 0;
    }

    return h;
}
