double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double h = 0;
    double sum_weights = 0;
    double max_weight = 0;
    double avg_weight = 0;
    double item_value = profit[index_item];
    
    // Calculate average weight and max weight per dimension for the neighbors
    for (int j = 0; j < dimension; j++) {
        double sum_weight_j = 0;
        double max_weight_j = 0;
        for (int i = 0; i < nb_voisinage; i++) {
            sum_weight_j += weights[j][voisinage[i]];
            if (weights[j][voisinage[i]] > max_weight_j) {
                max_weight_j = weights[j][voisinage[i]];
            }
        }
        sum_weights += sum_weight_j;
        if (max_weight_j > max_weight) {
            max_weight = max_weight_j;
        }
        avg_weight += sum_weight_j / nb_voisinage;
        
        // Calculate heuristic per dimension
        h += weights[j][index_item] / capacity[j];
    }
    
    avg_weight /= dimension;
    
    // Normalize and combine heuristic factors
    if (sum_weights > 0 && h > 0) {
        return (item_value / h) * (1 - (sum_weights / (max_weight * dimension))) * (avg_weight / max_weight);
    } else {
        return 0;
    }
}
