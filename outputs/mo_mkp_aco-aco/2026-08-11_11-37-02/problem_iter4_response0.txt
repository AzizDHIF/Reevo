double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate the weight-to-capacity ratio for each dimension
    double weight_to_capacity_ratio = 0.0;
    double valid_dimensions = 0.0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0.0) {
            weight_to_capacity_ratio += weights[j][index_item] / capacity[j];
            valid_dimensions += 1.0;
        }
    }

    // Adjust the ratio based on the number of valid dimensions
    if (valid_dimensions > 0.0) {
        weight_to_capacity_ratio /= valid_dimensions;
    } else {
        // If all dimensions are fully utilized, the item is not promising
        return 0.0;
    }

    // Calculate the neighbor factor based on the profit ratio
    double neighbor_factor = 0.0;
    if (nb_voisinage > 0) {
        for (int i = 0; i < nb_voisinage; i++) {
            neighbor_factor += profit[voisinage[i]] / profit[index_item];
        }
        neighbor_factor /= (double)nb_voisinage;
    }

    // Compute the total heuristic value as a combination of factors
    double h = profit[index_item] / (weight_to_capacity_ratio + 1e-9) * (1.0 / (1.0 + neighbor_factor));

    // Apply a bonus for items with high profit and low weight-to-capacity ratio
    if (weight_to_capacity_ratio < 0.5 && profit[index_item] > 0.5 * MAX_PROFIT) {
        h *= 1.2;
    }

    return h;
}
