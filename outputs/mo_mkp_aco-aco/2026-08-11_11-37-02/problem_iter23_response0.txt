double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ) 
{
    // Prioritize constraint satisfaction and simplify calculations
    double weight_to_capacity_ratio = 0;
    double valid_dimensions = 0;
    for(int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            if (weights[j][index_item] <= capacity[j]) {
                weight_to_capacity_ratio += weights[j][index_item] / capacity[j];
                valid_dimensions++;
            }
        }
    }

    if (valid_dimensions == 0) {
        return 0;
    } else {
        weight_to_capacity_ratio /= valid_dimensions;
    }

    // Simplify neighbor comparison and prioritize capacity
    double best_neighbor_profit = 0;
    if (nb_voisinage > 0) {
        for (int i = 0; i < nb_voisinage; i++) {
            if (profit[voisinage[i]] > best_neighbor_profit) {
                best_neighbor_profit = profit[voisinage[i]];
            }
        }
    }

    double neighbor_factor = (best_neighbor_profit > 0) ? best_neighbor_profit / profit[index_item] : 1;
    double constraint_satisfaction_factor = 1 - weight_to_capacity_ratio;
    double h = profit[index_item] * constraint_satisfaction_factor / (neighbor_factor + 1e-9);

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 1e-6 || constraint_satisfaction_factor <= 0) {
        return 0;
    }

    return h;
}
