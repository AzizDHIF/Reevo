double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ) 
{
    // Prioritize constraint satisfaction and feasibility
    double capacity_violation = 0;
    double weight_to_remaining_capacity_ratio = 0;
    double valid_dimensions = 0;
    for(int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            if (weights[j][index_item] > capacity[j]) {
                capacity_violation += weights[j][index_item] - capacity[j];
            } else {
                weight_to_remaining_capacity_ratio += weights[j][index_item] / capacity[j];
                valid_dimensions++;
            }
        }
    }

    if (valid_dimensions > 0) {
        weight_to_remaining_capacity_ratio /= valid_dimensions;
    } else {
        return 0;
    }

    // Simplify neighbor comparison and balance factors
    double best_neighbor_profit = 0;
    if (nb_voisinage > 0) {
        for (int i = 0; i < nb_voisinage; i++) {
            if (profit[voisinage[i]] > best_neighbor_profit) {
                best_neighbor_profit = profit[voisinage[i]];
            }
        }
    }

    // Prioritize capacity constraints and simplify calculations
    double h = 0;
    if (capacity_violation == 0) {
        double neighbor_factor = (best_neighbor_profit > 0) ? (1 - best_neighbor_profit / profit[index_item]) : 1;
        double capacity_factor = (weight_to_remaining_capacity_ratio < 1) ? (1 - weight_to_remaining_capacity_ratio) : 1;
        h = profit[index_item] * capacity_factor * neighbor_factor;
    }

    // Sparsify the heuristic by setting unpromising elements to zero
    if(h < 1e-9) {
        h = 0;
    }

    return h;
}
