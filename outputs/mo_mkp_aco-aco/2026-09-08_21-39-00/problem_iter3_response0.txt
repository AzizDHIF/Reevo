double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double sum_weights = 0;
    double sum_capacity = 0;

    for (int j = 0; j < dimension; j++) {
        sum_weights += weights[j][voisinage[index_item]];
        sum_capacity += capacity[j];
    }

    // Adjust profit-weight balance based on capacity utilization
    double balance = 1.0 - (sum_weights / sum_capacity);

    // Consider worst-case improvements for heuristic selection
    double worst_improvement = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (capacity[0] - sum_weights + weights[0][voisinage[i]] < 0) {
            double improvement = (profit[voisinage[i]] - profit[voisinage[index_item]]) / sum_weights;
            if (improvement < worst_improvement) {
                worst_improvement = improvement;
            }
        }
    }

    return balance * (profit[voisinage[index_item]] / sum_weights) + worst_improvement;
}
