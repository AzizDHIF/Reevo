double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h=0;
    double factor1 = 0.7; // Weight factor to be tuned
    double factor2 = 0.3; // Profit factor to be tuned

    if (nb_voisinage == 0) {
        return 0.0;
    }
    
    for(int j=0; j<dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j];
    }

    if (h == 0) {
        return 0.0;
    }

    double promising_item_score = (1 - factor1) * (1 - factor2) * (profit[voisinage[index_item]] / h); // Combining factors
    return promising_item_score;
}
