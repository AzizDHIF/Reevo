double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_ratios_sum = 0;
    double weight_plus_one_inv_sum = 0;
    double capacity_sum = 0;

    for (int j = 0; j < dimension; j++) {
        double weight_item = weights[j][voisinage[index_item]];
        double weight_plus_one = weight_item + 1;
        double weight_ratio = weight_item / capacity[j];

        weight_ratios_sum += weight_ratio;
        weight_plus_one_inv_sum += 1 / weight_plus_one;
        capacity_sum += capacity[j];
    }

    double factor1 = profit[voisinage[index_item]] / (weight_ratios_sum + 1);
    double factor2 = weight_plus_one_inv_sum / nb_voisinage;
    double factor3 = 1 / (capacity_sum + 1);

    return factor1 * factor2 * factor3;
}
