double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double total_weight_ratio = 0.0;

    for (int j = 0; j < dimension; j++) {
        total_weight_ratio += weights[j][voisinage[index_item]] / capacity[j];
    }

    if (total_weight_ratio > dimension) {
        return 0.0; // Penalize items where weight exceeds capacity
    }

    return profit[voisinage[index_item]] - total_weight_ratio;
}
