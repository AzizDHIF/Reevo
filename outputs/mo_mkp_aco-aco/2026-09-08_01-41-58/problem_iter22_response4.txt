double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double sum_weight_ratio = 0, sum_inv_weight = 0, sum_capacity = 0;

    for (int j = 0; j < dimension; j++) {
        double weight = weights[j][voisinage[index_item]];
        double weight_ratio = weight / capacity[j];
        double weight_inv = 1 / (weight + 1);

        sum_weight_ratio += weight_ratio;
        sum_inv_weight += weight_inv;
        sum_capacity += capacity[j];
    }

    double objective_ratio = profit[voisinage[index_item]] / (sum_weight_ratio + 1);
    double neighbor_ratio = sum_inv_weight / nb_voisinage;
    double capacity_ratio = 1 / (sum_capacity + 1);

    return objective_ratio * neighbor_ratio * capacity_ratio;
}
