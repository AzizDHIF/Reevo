double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h1 = 0, h2 = 0;
    
    for (int j = 0; j < dimension; j++) {
        h1 += weights[j][voisinage[index_item]] / capacity[j];
        h2 += 1 / (weights[j][voisinage[index_item]] + 1);
    }
    
    double factor1 = profit[voisinage[index_item]] / h1;
    double factor2 = h2 / nb_voisinage;
    
    return factor1 * factor2;
}
