double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double w_sum_normalized = 0, w_inv_sum = 0, c_inv_sum = 0;

    for (int j = 0; j < dimension; j++) {
        double w_normalized = weights[j][voisinage[index_item]] / (capacity[j] + 1);
        w_sum_normalized += w_normalized;
        w_inv_sum += 1 / (weights[j][voisinage[index_item]] + 1);
        c_inv_sum += 1 / (capacity[j] + 1);
    }

    double factor1 = profit[voisinage[index_item]] / (w_sum_normalized + 1);
    double factor2 = w_inv_sum / nb_voisinage;
    double factor3 = c_inv_sum / dimension;

    return factor1 * factor2 * factor3;
}
