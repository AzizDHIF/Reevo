double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] )
{
    double sum_weight_ratio = 0;
    double sum_inverse_weights = 0;
    double sum_capacities = 0;

    for (int j = 0; j < nb_voisinage; j++) {
        int neighbor_index = voisinage[j];
        for (int k = 0; k < dimension; k++) {
            sum_weight_ratio += weights[k][neighbor_index] / capacity[k];
            sum_inverse_weights += 1 / (weights[k][neighbor_index] + 1);
            sum_capacities += capacity[k];
        }
    }

    double average_weight_ratio = sum_weight_ratio / (dimension * nb_voisinage);
    double average_inverse_weight = sum_inverse_weights / (dimension * nb_voisinage);
    double average_capacity = sum_capacities / (dimension * nb_voisinage);

    double factor1 = profit[index_item] / (average_weight_ratio + 1);
    double factor2 = average_inverse_weight;
    double factor3 = 1 / (average_capacity + 1);

    return factor1 * factor2 * factor3;
}
