double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_sum = 0, weight_inv_sum = 0, capacity_sum = 0;

    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][voisinage[index_item]] / capacity[j];
        weight_inv_sum += 1 / (weights[j][voisinage[index_item]] + 1);
        capacity_sum += capacity[j];
    }

    double factor1 = profit[voisinage[index_item]] / (weight_sum + 1);
    double factor2 = weight_inv_sum / nb_voisinage;
    double factor3 = 1 / (capacity_sum + 1);

    return factor1 * factor2 * factor3;
}
