double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double sum_weight_ratio = 0;
    double sum_weight_plus_one_inv = 0;

    for (int j = 0; j < dimension; j++) {
        double weight_item = weights[j][voisinage[index_item]];
        double weight_ratio = weight_item / capacity[j];
        sum_weight_ratio += weight_ratio;
        sum_weight_plus_one_inv += 1 / (weight_item + 1);
    }

    double factor1 = profit[voisinage[index_item]] / sum_weight_ratio;
    double factor2 = sum_weight_plus_one_inv / nb_voisinage;

    return factor1 * factor2;
}
