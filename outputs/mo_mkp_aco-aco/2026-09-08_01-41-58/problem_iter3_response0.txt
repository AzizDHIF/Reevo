double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double normalized_weight = 0;
    double weight_penalty = 0;
    for (int j = 0; j < dimension; j++)
    {
        normalized_weight += weights[j][voisinage[index_item]] / capacity[j];
        weight_penalty += 1 / (1 + weights[j][voisinage[index_item]]);
    }

    double objective_ratio = profit[voisinage[index_item]] / normalized_weight;
    double exploration_factor = 1 - weight_penalty / dimension;
    double capacity_factor = 1 / (1 + capacity[0]);

    return objective_ratio * exploration_factor * capacity_factor;
}
