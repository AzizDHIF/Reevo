double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double sum_weight_ratio = 0;
    double sum_capacities = 0;

    for (int j = 0; j < dimension; j++) {
        double neighbor_weight = weights[j][voisinage[index_item]];

        if (neighbor_weight > 0) {
            sum_weight_ratio += neighbor_weight / capacity[j];
            sum_capacities += 1 / (neighbor_weight + 1);
        }
    }

    double profit_weight_ratio = profit[index_item] / (sum_weight_ratio + 1);
    double neighbors_weight_ratio = sum_capacities / nb_voisinage;

    return profit_weight_ratio * neighbors_weight_ratio;
}
