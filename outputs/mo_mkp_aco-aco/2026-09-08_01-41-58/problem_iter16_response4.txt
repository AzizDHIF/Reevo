double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_ratio_sum = 0;
    double weight_plus_one_inv_sum = 0;

    for (int j = 0; j < dimension; j++) {
        double weight_item = weights[j][index_item];
        double weight_plus_one = weight_item + 1;
        weight_ratio_sum += weight_item / capacity[j];
        weight_plus_one_inv_sum += 1 / weight_plus_one;
    }

    double factor1 = profit[index_item] / (weight_ratio_sum + 1);
    double factor2 = weight_plus_one_inv_sum / dimension;

    return factor1 * factor2;
}
