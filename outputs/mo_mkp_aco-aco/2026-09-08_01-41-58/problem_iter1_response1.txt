double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double h=0, w_sum = 0, total_sum = 0;
    for(int j=0; j<dimension; j++){
        w_sum += weights[j][voisinage[index_item]];
        total_sum += weights[j][index_item];
    }
    for(int j=0; j<dimension; j++){
        if(capacity[j] - weights[j][index_item] > 0)
            h += (weights[j][voisinage[index_item]] - weights[j][index_item])/(capacity[j] - total_sum);
        else
            h += (weights[j][voisinage[index_item]] - weights[j][index_item]);
    }
    return h > 0 ? profit[voisinage[index_item]] / h : 0;
}
