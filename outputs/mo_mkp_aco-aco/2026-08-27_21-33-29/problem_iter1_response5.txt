double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], 
                    int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double h[dimension];
    for(int j=0; j<dimension; j++){
        h[j] = weights[j][voisinage[index_item]] / capacity[j];
    }
      
    double combined_factor = 0.3 * h[0] + 0.5 * h[1]; // Combine factors for how promising it is
    
    if(combined_factor <= 0.2) return 0.0; // Sparsify: zero out if combined factor is unpromising
    else return profit[voisinage[index_item]] / h[0];
}
