double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0.0; // Initialize h as zero

    for (int j = 0; j < dimension; j++) {
        if (capacity[j] <= 0) {
            return 0; // Early termination if capacity is exhausted or invalid
        }
        h += weights[j][voisinage[index_item]] / capacity[j]; // Utilize available capacity efficiently
    }

    return profit[voisinage[index_item]] + h; // Combine factors to determine promise
}
