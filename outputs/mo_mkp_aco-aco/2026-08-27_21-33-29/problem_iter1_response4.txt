double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double promising_factor = 0;
    double promising_weight_sum = 0;
    
    for(int j=0; j<dimension; j++){
        promising_weight_sum += weights[j][voisinage[index_item]] / capacity[j];
    }
    
    if(promising_weight_sum == 0){
        return 0;
    }
    
    promising_factor += promising_weight_sum;
    promising_factor += profit[voisinage[index_item]] * nb_voisinage;
    
    return promising_factor;
}
