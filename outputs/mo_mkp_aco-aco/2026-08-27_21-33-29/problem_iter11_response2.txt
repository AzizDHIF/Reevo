double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double sum_weight_ratio = 0;
    
    if (nb_voisinage <= 0) {
        return 0;
    }
    
    for (int j = 0; j < dimension; j++) {
        sum_weight_ratio += weights[j][voisinage[index_item]] / capacity[j];
    }
    
    if (sum_weight_ratio == 0) {
        return 0;
    }
    
    return profit[voisinage[index_item]] / sum_weight_ratio;
}
