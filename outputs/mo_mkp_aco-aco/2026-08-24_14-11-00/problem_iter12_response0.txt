double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    const double epsilon = 0.0001;
    const double factor = 0.7;

    double capacity_norm = 0.0;
    for (int j = 0; j < dimension; j++) {
        capacity_norm += capacity[j];
    }

    if (capacity_norm < epsilon) {
        return 0.0;
    }

    double h = epsilon;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > epsilon) {
            h += weights[j][voisinage[index_item]] / (capacity_norm * epsilon);
        }
    }

    double heuristic_value = profit[voisinage[index_item]] / (h + epsilon);

    if (heuristic_value < 0.5) {
        return 0.0;
    }

    return heuristic_value * factor;
}
