double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double factor1 = 0.5;
    double factor2 = 0.7;

    for (int j = 0; j < dimension; j++) {
        if (capacity[j] != 0) {
            h += weights[j][voisinage[index_item]] / capacity[j];
        }
        else {
            return 0.0; // Reject solution early if division by zero
        }
    }

    if (h < 0.00001) {
        return 0.0;
    }

    double heuristic_value = profit[voisinage[index_item]] / h;

    if (heuristic_value < factor1) {
        return 0.0;
    }

    return heuristic_value * factor2;
}
