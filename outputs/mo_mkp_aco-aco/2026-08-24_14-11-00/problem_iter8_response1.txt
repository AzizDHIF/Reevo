double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0.0001; // Prevent division by zero
    double factor1 = 0.5;
    double factor2 = 0.7; // Experiment with different factors here

    // Normalize weights
    double normalized_weights[dimension][NBITEMS];
    for (int i = 0; i < dimension; i++) {
        for (int j = 0; j < NBITEMS; j++) {
            normalized_weights[i][j] = weights[i][j] / capacity[i];
        }
    }

    for (int j = 0; j < dimension; j++) {
        h += normalized_weights[j][voisinage[index_item]];
    }

    double heuristic_value = profit[voisinage[index_item]] / h;

    if (heuristic_value < factor1) {
        return 0.0;
    }

    return heuristic_value * factor2;
}
