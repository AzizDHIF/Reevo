double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double h = 0.0001; // Prevent division by zero
    double weight_sum = 0.00001; // Prevent division by zero
    double factor = 0.7;

    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            h += weights[j][voisinage[index_item]] / (capacity[j] * 0.001);
            weight_sum += weights[j][voisinage[index_item]];
        }
    }

    double normalized_weight = weight_sum / (double)(dimension); // Normalize total weight
    double normalized_capacity = 0.0;
    for (int c = 0; c < dimension; c++) {
        normalized_capacity += capacity[c];
    }
    normalized_capacity /= (double)(dimension); // Average capacity across dimensions

    double heuristic_value = profit[voisinage[index_item]] / ((h + normalized_capacity) * 0.001 * (normalized_weight + 0.001));

    if (heuristic_value < 0.5) {
        return 0.0;
    }

    return heuristic_value * factor;
}
