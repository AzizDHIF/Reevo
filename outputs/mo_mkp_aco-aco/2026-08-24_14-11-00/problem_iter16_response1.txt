double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0.0001; // Prevent division by zero
    double factor = 0.7;

    for (int j = 0; j < dimension; j++) {
        double weight_item = weights[j][voisinage[index_item]];
        double weight_ratio = weight_item * 1000; // Multiply by 1000 for better precision
        if (capacity[j] > weight_ratio) {
            h += weight_ratio / capacity[j];
        }
    }

    double heuristic_value = profit[voisinage[index_item]] / (h + 0.001);

    return (heuristic_value < 0.5) ? 0.0 : heuristic_value * factor;
}
