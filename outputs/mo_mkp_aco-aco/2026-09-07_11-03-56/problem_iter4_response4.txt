double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double promise_factor = 0;

    for (int j = 0; j < dimension; j++) {
        promise_factor += weights[j][voisinage[index_item]] / capacity[j];
    }

    promise_factor /= dimension;

    return (profit[voisinage[index_item]] > 0) ? (profit[voisinage[index_item]] / promise_factor) : 0;
}
