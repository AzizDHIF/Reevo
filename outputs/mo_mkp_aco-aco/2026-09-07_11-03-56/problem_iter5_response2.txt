double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] )
{
    double sum_ratio = 0;
    double max_ratio = 0;

    for (int j = 0; j < dimension; j++)
    {
        double weight_ratio = weights[j][voisinage[index_item]] / capacity[j];
        sum_ratio += weight_ratio;
        max_ratio = fmax(max_ratio, weight_ratio);
    }

    if (sum_ratio == 0 && max_ratio == 0)
    {
        return 0;
    }
    else
    {
        return profit[voisinage[index_item]] / (sum_ratio + max_ratio);
    }
}
