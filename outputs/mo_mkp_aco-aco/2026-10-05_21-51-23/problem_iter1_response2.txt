double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 0;
    double perspective_factor = 2.0;
    
    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j]; 
    }
    
    double promising_factor = 1.0 / h;
    double weight_diff_factor = 0.0;
    
    for (int j = 0; j < dimension; j++) {
        weight_diff_factor += (capacity[j] - weights[j][voisinage[index_item]]) > 0 ? (capacity[j] - weights[j][voisinage[index_item]]) : 0;
    }
    
    double heuristic_score = (profit[voisinage[index_item]] + perspective_factor * promising_factor) / (1 + weight_diff_factor);
    
    return heuristic_score;
}
