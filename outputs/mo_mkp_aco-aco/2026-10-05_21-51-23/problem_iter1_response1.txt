double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double normalized_weight=0;
    for(int j=0;j<dimension;j++){
        normalized_weight += weights[j][voisinage[index_item]] / capacity[j];
    }
    double weight_factor = 1/normalized_weight;
    double profit_factor = profit[voisinage[index_item]];
    
	// Penalize heavily if the weight of the neighbor exceeds capacity
    if (normalized_weight > 1.0){
        return 0.0;
    }
    
    // Reward items based on their profit and weight factors
    return (10.0*profit_factor + 100.0*weight_factor);
}
