double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double h=0, w=0, p=0;
    for(int j=0;j<dimension;j++){
        h += weights[j][voisinage[index_item]] / capacity[j];
        w += weights[j][index_item];
    }
    
    h = h != 0 ? h : 1; // Avoid division by zero
    
    if(w > capacity[dimension-1]){
        return 0;  // If total weight of item exceeds remaining capacity in last dimension, unfeasible move
    }
    
    p = profit[voisinage[index_item]];
    
    return p / h; // Higher profit-to-capacity ratio for promising items
}
