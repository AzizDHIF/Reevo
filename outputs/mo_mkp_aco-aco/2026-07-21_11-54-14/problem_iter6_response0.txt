double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_sum = 0;
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
    }
    if (weight_sum == 0) {
        return 0;
    }
    double density = profit[index_item] / weight_sum;

    double neighbor_density_sum = 0;
    double neighbor_count = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_weight_sum = 0;
        for (int j = 0; j < dimension; j++) {
            neighbor_weight_sum += weights[j][voisinage[i]];
        }
        if (neighbor_weight_sum > 0) {
            neighbor_density_sum += profit[voisinage[i]] / neighbor_weight_sum;
            neighbor_count++;
        }
    }
    double avg_neighbor_density = neighbor_count > 0 ? neighbor_density_sum / neighbor_count : 0;

    double neighborhood_effect = 1 - abs(density - avg_neighbor_density);
    double capacity_effect = 1;
    for (int j = 0; j < dimension; j++) {
        capacity_effect *= (1 - weights[j][index_item] / capacity[j]);
    }
    double heuristic_value = density + neighborhood_effect + capacity_effect;

    // Sparsification: set unpromising elements to zero
    if (heuristic_value < 0.1) {
        return 0;
    }
    return heuristic_value;
}
