```c
double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double max_weight = 0;
    for (int j = 0; j < dimension; j++) {
        h += weights[j][index_item] / capacity[j];
        if (weights[j][index_item] > max_weight) {
            max_weight = weights[j][index_item];
        }
    }
    double avg_weight = h / dimension;
    double weight_to_capacity_ratio = max_weight / (max_weight + 1); // To avoid division by zero
    double density = profit[index_item] / max_weight;

    // Combine various factors
    double combined_heuristic = alpha * density + beta * (1 - weight_to_capacity_ratio);
    combined_heuristic *= sparsification_factor * (profit[index_item] > 0);

    // Sparsification: set unpromising elements to zero
    if (combined_heuristic < sparsification_threshold) {
        combined_heuristic = 0;
    }

    return combined_heuristic;
}

// Parameters for the heuristic function
double alpha = 0.5;  // Weight for density
double beta = 0.5;   // Weight for weight-to-capacity ratio
double sparsification_factor = 0.1;  // Factor for sparsification
double sparsification_threshold = 0.01;  // Threshold for sparsification
```
