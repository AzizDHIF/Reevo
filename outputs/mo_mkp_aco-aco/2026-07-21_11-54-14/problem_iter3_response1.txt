double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate the item's weight sum and density
    double weight_sum = 0;
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
    }
    double density = weight_sum > 0 ? profit[index_item] / weight_sum : 0;

    // Calculate the average neighbor density
    double neighbor_density = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_weight_sum = 0;
        for (int j = 0; j < dimension; j++) {
            neighbor_weight_sum += weights[j][voisinage[i]];
        }
        double neighbor_profit = profit[voisinage[i]];
        neighbor_density += neighbor_weight_sum > 0 ? neighbor_profit / neighbor_weight_sum : 0;
    }
    double avg_neighbor_density = nb_voisinage > 0 ? neighbor_density / nb_voisinage : 0;

    // Calculate the heuristic value considering both density and objective factors
    double h = 0;
    for (int j = 0; j < dimension; j++) {
        h += weights[j][index_item] / capacity[j];
    }
    double objective_factor = 1 / (1 + h);
    double heuristic_value = density * objective_factor * (1 + density / (1 + abs(density - avg_neighbor_density)));

    // Sparsification: set unpromising elements to zero
    if (heuristic_value < 0.1 || density < 0.1 || h > 1) {
        return 0;
    }
    return heuristic_value;
}
