double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_sum = 0;
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
    }
    if (weight_sum == 0) {
        return 0;
    }
    double density = profit[index_item] / weight_sum;

    double neighbor_density_sum = 0;
    double neighbor_count = 0;
    double max_neighbor_density = 0;
    double min_neighbor_density = 1e6;
    double neighbor_densities[dimension];
    for (int i = 0; i < dimension; i++) {
        neighbor_densities[i] = 0;
    }
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_weight_sum = 0;
        for (int j = 0; j < dimension; j++) {
            neighbor_weight_sum += weights[j][voisinage[i]];
        }
        if (neighbor_weight_sum > 0) {
            double neighbor_density = profit[voisinage[i]] / neighbor_weight_sum;
            neighbor_density_sum += neighbor_density;
            neighbor_count++;
            for (int j = 0; j < dimension; j++) {
                neighbor_densities[j] += neighbor_density * weights[j][voisinage[i]] / neighbor_weight_sum;
            }
            if (neighbor_density > max_neighbor_density) {
                max_neighbor_density = neighbor_density;
            }
            if (neighbor_density < min_neighbor_density) {
                min_neighbor_density = neighbor_density;
            }
        }
    }
    double avg_neighbor_density = neighbor_count > 0 ? neighbor_density_sum / neighbor_count : 0;
    double neighborhood_similarity = 1 - abs(density - avg_neighbor_density) / (density + avg_neighbor_density + 1e-6);
    double neighborhood_diversity = abs(density - max_neighbor_density) / (density + max_neighbor_density + 1e-6) + abs(density - min_neighbor_density) / (density + min_neighbor_density + 1e-6);

    double capacity_effect = 1;
    double capacity_sum = 0;
    for (int j = 0; j < dimension; j++) {
        capacity_effect *= (1 - weights[j][index_item] / capacity[j]);
        capacity_sum += weights[j][index_item] / capacity[j];
    }
    double capacity_pressure = capacity_sum / dimension;

    double heuristic_value = density * neighborhood_similarity * (1 - capacity_pressure);
    // Combine neighborhood similarity and diversity
    heuristic_value *= (1 + neighborhood_diversity) / 2;

    // Incorporate diversity, similarity metrics for each dimension
    double dimension_diversity = 0;
    double dimension_similarity = 0;
    for (int j = 0; j < dimension; j++) {
        dimension_diversity += abs(neighbor_densities[j] / neighbor_count - density) / (density + neighbor_densities[j] / neighbor_count + 1e-6);
        dimension_similarity += 1 - abs(neighbor_densities[j] / neighbor_count - density) / (density + neighbor_densities[j] / neighbor_count + 1e-6);
    }
    dimension_diversity /= dimension;
    dimension_similarity /= dimension;

    heuristic_value *= (1 + dimension_diversity) / 2;
    heuristic_value *= (1 + dimension_similarity) / 2;

    // Sparsification: set unpromising elements to zero
    if (heuristic_value < 0.1) {
        return 0;
    }
    return heuristic_value;
}
