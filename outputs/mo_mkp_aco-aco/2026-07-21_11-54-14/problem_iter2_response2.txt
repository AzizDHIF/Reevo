double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double weight_sum = 0;
    double density = 0;
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
        h += weights[j][index_item] / capacity[j];
    }
    if (weight_sum == 0) {
        return 0;
    }
    density = profit[index_item] / weight_sum;
    
    double neighbor_density = 0;
    int valid_neighbors = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_weight_sum = 0;
        for (int j = 0; j < dimension; j++) {
            neighbor_weight_sum += weights[j][voisinage[i]];
        }
        if (neighbor_weight_sum > 0) {
            neighbor_density += profit[voisinage[i]] / neighbor_weight_sum;
            valid_neighbors++;
        }
    }
    double avg_neighbor_density = valid_neighbors > 0 ? neighbor_density / valid_neighbors : 0;
    
    double heuristic_value = density / (h * (1 + abs(density - avg_neighbor_density)));
    // Sparsification: set unpromising elements to zero
    if (heuristic_value < 0.1) {
        return 0;
    }
    return heuristic_value;
}
