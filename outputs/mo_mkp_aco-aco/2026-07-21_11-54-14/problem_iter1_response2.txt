double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double total_weight = 0;
    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item];
    }
    double average_weight = total_weight / dimension;

    double total_capacity = 0;
    for (int j = 0; j < dimension; j++) {
        total_capacity += capacity[j];
    }
    double average_capacity = total_capacity / dimension;

    double utilization = average_weight / average_capacity;

    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_utilization = 0;
        double neighbor_weight = 0;
        for (int j = 0; j < dimension; j++) {
            neighbor_weight += weights[j][voisinage[i]];
        }
        neighbor_utilization = neighbor_weight / average_capacity;
        h += utilization + neighbor_utilization;
    }

    double sparsification_threshold = 0.5;
    if (h < sparsification_threshold) {
        return 0;
    }

    double total_profit = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        total_profit += profit[voisinage[i]];
    }

    double average_profit = total_profit / nb_voisinage;
    double item_profit_ratio = profit[index_item] / average_profit;

    double score = item_profit_ratio / h;

    return score;
}
