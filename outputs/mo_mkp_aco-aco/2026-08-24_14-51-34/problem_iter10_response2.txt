double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] )
{
    double weight_factor = 0.6;
    double profit_factor = 0.4;
    
    double weight_contribution = 0;
    for (int j = 0; j < dimension; j++) {
        double item_weight = weights[j][index_item];
        if (capacity[j] < item_weight) {
            weight_contribution += 1.0; // Penalize if weight exceeds capacity
        } else {
            weight_contribution += item_weight / capacity[j];
        }
    }
    
    double promising_score = weight_factor * (1.0 / weight_contribution) + profit_factor * profit[index_item];
    
    return promising_score;
}
