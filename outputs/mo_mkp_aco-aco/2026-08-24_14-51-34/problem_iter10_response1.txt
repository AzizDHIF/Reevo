double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double weight_factor = 0.6;
    double profit_factor = 0.4;

    double weight_contribution = 0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] < weights[j][index_item]) {
            return -1; // Penalize violating capacity constraints
        }
        weight_contribution += weights[j][index_item] / capacity[j];
    }

    return weight_factor / weight_contribution + profit_factor * profit[index_item];
}
