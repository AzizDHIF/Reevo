double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_factor = 0.6;
    double profit_factor = 0.4;

    double weight_contribution = 0;
    double profit_contribution = 0;

    for(int j = 0; j < dimension; j++) {
        double weight_ratio = (capacity[j] != 0) ? weights[j][voisinage[index_item]] / capacity[j] : 0;
        weight_contribution += (weight_ratio > 1 ? 0 : 1 - weight_ratio); // Penalize weights exceeding capacity
    }

    if (weight_contribution > 0) {
        profit_contribution = profit_factor * profit[voisinage[index_item]];
    }

    return weight_factor * weight_contribution + profit_contribution;
}
