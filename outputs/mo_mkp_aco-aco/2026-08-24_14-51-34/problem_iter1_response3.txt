double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]){
    double h = 0;
    double weight_ratio = 0;
    double profit_ratio = 0;
    
    for(int j=0; j<dimension; j++){
        if(capacity[j] > 0){
            h += (weights[j][voisinage[index_item]] + 1) / (1 + capacity[j]);
            weight_ratio += weights[j][voisinage[index_item]] / (1 + capacity[j]);
        }
    }
    
    if(h > 0){
        profit_ratio = profit[voisinage[index_item]] / h;
    } else {
        profit_ratio = 0;
    }
    
    return profit_ratio * weight_ratio;
}
