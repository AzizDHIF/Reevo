double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double density = 0;
    double proximity = 0;
    double weight_sum = 0;

    // Calculate the total weight of the item
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
    }

    // Calculate the average weight of the item
    double average_weight = weight_sum / dimension;

    // Calculate the item's weight-to-profit ratio (density)
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            density += profit[index_item] / weights[j][index_item];
        }
    }
    density /= dimension;

    // Calculate the proximity to neighbors
    for (int i = 0; i < nb_voisinage; i++) {
        for (int j = 0; j < dimension; j++) {
            if (weights[j][voisinage[i]] > 0 && weights[j][index_item] > 0) {
                proximity += (profit[voisinage[i]] + profit[index_item]) / (weights[j][voisinage[i]] + weights[j][index_item]);
            }
        }
    }
    proximity /= nb_voisinage * dimension;

    // Check if item can fit in any knapsack
    int can_fit = 1;
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > capacity[j]) {
            can_fit = 0;
            break;
        }
    }

    // Combine factors to get the final heuristic value
    double h;
    if (can_fit) {
        h = (density * 0.5 + (profit[index_item] / average_weight) * 0.3 + proximity * 0.2);
    } else {
        h = (density * 0.5 + (profit[index_item] / average_weight) * 0.3 + proximity * 0.2) * 0.1;
    }

    return h;
}
