double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate the total weight of the item
    double weight_sum = 0;
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
    }

    // Calculate density
    double density = 0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0 && weights[j][index_item] > 0) {
            density += profit[index_item] / weights[j][index_item];
        }
    }
    if (weight_sum > 0) {
        density /= weight_sum;
    } else {
        density = 0;
    }

    // Calculate proximity to neighbors, with a simplified calculation
    double proximity = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_weight_sum = 0;
        for (int j = 0; j < dimension; j++) {
            neighbor_weight_sum += weights[j][voisinage[i]];
        }
        if (neighbor_weight_sum > 0) {
            proximity += (profit[voisinage[i]] + profit[index_item]) / (neighbor_weight_sum + weight_sum);
        }
    }
    if (nb_voisinage > 0) {
        proximity /= nb_voisinage;
    } else {
        proximity = 0;
    }

    // Apply penalty if the item cannot fit in any knapsack
    double total_remaining_capacity = 0;
    for (int j = 0; j < dimension; j++) {
        total_remaining_capacity += capacity[j];
    }
    double penalty = 1;
    if (weight_sum > total_remaining_capacity) {
        penalty = 0.1;
    }

    // Combine factors to get the final heuristic value, with a simplified calculation and prioritizing density
    double h = penalty * (density * 0.7 + (profit[index_item] / weight_sum) * 0.2 + proximity * 0.1);

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 0.01) {
        h = 0;
    }

    return h;
}
