double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) 
{
    double weight_sum = 0;
    double density = 0;
    double proximity = 0;

    // Calculate total weight and density
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
        if (weights[j][index_item] > 0) {
            density += profit[index_item] / weights[j][index_item];
        }
    }
    if (dimension > 0) {
        density /= dimension;
    }

    // Simplify proximity calculation
    if (nb_voisinage > 0) {
        for (int i = 0; i < nb_voisinage; i++) {
            proximity += profit[voisinage[i]];
        }
        proximity /= nb_voisinage;
    }

    // Apply penalty if item cannot fit
    double penalty = 1.0;
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > capacity[j]) {
            penalty = 0.1;
            break;
        }
    }

    // Calculate heuristic value
    double h = penalty * (density * 0.7 + (weight_sum > 0 ? profit[index_item] / weight_sum : 0) * 0.2 + proximity * 0.1);

    // Sparsify heuristic
    if (h < 0.01) {
        h = 0;
    }

    return h;
}
