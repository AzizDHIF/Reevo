double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double profit_ratio = 0;
    double weight_ratio = 0;

    // Calculate the total capacity
    double total_capacity = 0;
    for (int j = 0; j < dimension; j++) {
        total_capacity += capacity[j];
    }

    // Calculate the weight ratio for each neighbor of the item
    for (int i = 0; i < nb_voisinage; i++) {
        double weight_sum = 0;
        for (int j = 0; j < dimension; j++) {
            weight_sum += weights[j][voisinage[i]];
        }
        weight_ratio += weight_sum / total_capacity;
    }

    // Calculate the profit ratio for the item
    for (int i = 0; i < nb_voisinage; i++) {
        profit_ratio += profit[voisinage[i]];
    }
    profit_ratio /= nb_voisinage;

    // Calculate the heuristic value
    double max_weightRatio = 0;
    double neighbor_value = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        double weight_sum = 0;
        for (int j = 0; j < dimension; j++) {
            weight_sum += weights[j][voisinage[i]];
        }
        double ratio = weight_sum / total_capacity;
        if (ratio > max_weightRatio)
            max_weightRatio = ratio;
        neighbor_value += profit[voisinage[i]] * (1 - ratio) / max_weightRatio;
    }

    if (weight_ratio > 0) {
        h = profit_ratio / weight_ratio;
    }
    else {
        h = profit_ratio;
    }

    // Combine the heuristic values and apply a weight
    h *= (profit[index_item] * neighbor_value) / (nb_voisinage * max_weightRatio);

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 0.1 * profit[index_item])
        h = 0;

    return h;
}
