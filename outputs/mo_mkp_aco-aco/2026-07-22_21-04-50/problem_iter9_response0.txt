double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate the total weight of the item
    double weight_sum = 0;
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
    }

    // Calculate item density, avoiding division by zero
    double density = 0;
    int non_zero_weights = 0;
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > 0) {
            density += profit[index_item] / weights[j][index_item];
            non_zero_weights++;
        }
    }
    if (non_zero_weights > 0) {
        density /= non_zero_weights;
    } else {
        density = 0;
    }

    // Simplify proximity calculation
    double proximity = 0;
    if (nb_voisinage > 0) {
        for (int i = 0; i < nb_voisinage; i++) {
            proximity += profit[voisinage[i]];
        }
        proximity /= nb_voisinage;
    }

    // Apply a penalty if the item cannot fit in any knapsack
    double penalty = 1;
    double total_remaining_capacity = 0;
    for (int j = 0; j < dimension; j++) {
        total_remaining_capacity += capacity[j];
    }
    if (weight_sum > total_remaining_capacity) {
        penalty = 0.1;
    }

    // Calculate capacity utilization
    double capacity_utilization = 0;
    for (int j = 0; j < dimension; j++) {
        capacity_utilization += weights[j][index_item] / capacity[j];
    }
    capacity_utilization /= dimension;

    // Combine factors to get the final heuristic value
    double h = penalty * (density * 0.5 + capacity_utilization * 0.3 + proximity * 0.1 + profit[index_item] / (weight_sum + 1e-9) * 0.1);

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 0.01) {
        h = 0;
    }

    return h;
}
