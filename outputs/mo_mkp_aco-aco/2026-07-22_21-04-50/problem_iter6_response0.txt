double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate the total weight of the item
    double weight_sum = 0;
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
    }

    // Check if the item can fit in the knapsacks
    double penalty = 1.0;
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > capacity[j]) {
            penalty = 0.1;
            break;
        }
    }

    // Calculate density, prioritizing items with higher profit-to-weight ratio
    double density = 0;
    if (weight_sum > 0) {
        for (int j = 0; j < dimension; j++) {
            if (capacity[j] > 0 && weights[j][index_item] > 0) {
                density += profit[index_item] / weights[j][index_item];
            }
        }
        density /= weight_sum;
    }

    // Simplify proximity calculation
    double proximity = 0;
    if (nb_voisinage > 0) {
        double total_neighbor_profit = 0;
        double total_neighbor_weight = 0;
        for (int i = 0; i < nb_voisinage; i++) {
            total_neighbor_profit += profit[voisinage[i]];
            for (int j = 0; j < dimension; j++) {
                total_neighbor_weight += weights[j][voisinage[i]];
            }
        }
        if (total_neighbor_weight + weight_sum > 0) {
            proximity = (total_neighbor_profit + profit[index_item]) / (total_neighbor_weight + weight_sum);
        }
    }

    // Combine factors to get the final heuristic value, prioritizing density and balancing factors
    double h = penalty * (density * 0.8 + proximity * 0.2);

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 0.01) {
        h = 0;
    }

    return h;
}
