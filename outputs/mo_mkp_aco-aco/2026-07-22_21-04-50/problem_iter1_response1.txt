double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double h = 0;
    double min_cap_ratio = 1e10;
    double sum_cap_ratio = 0;
    int count = 0;
    double weight_sum = 0;

    // Calculate the sum of weights for the item and its neighbors
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
        for (int i = 0; i < nb_voisinage; i++) {
            weight_sum += weights[j][voisinage[i]];
        }
    }

    // Calculate the weight-to-capacity ratio for each dimension
    for (int j = 0; j < dimension; j++) {
        double cap_ratio = weights[j][index_item] / capacity[j];
        if (cap_ratio < min_cap_ratio) {
            min_cap_ratio = cap_ratio;
        }
        sum_cap_ratio += cap_ratio;
        count++;
    }

    // Calculate the average weight-to-capacity ratio
    double avg_cap_ratio = sum_cap_ratio / count;

    // Calculate the heuristic value
    if (weight_sum > 0 && min_cap_ratio < 1e10) {
        h = profit[index_item] / (weight_sum * min_cap_ratio * avg_cap_ratio);
    } else {
        h = 0;
    }

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 1e-6) {
        h = 0;
    }

    return h;
}
