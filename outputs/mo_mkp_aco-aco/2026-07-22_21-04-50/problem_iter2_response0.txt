double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_sum = 0;
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][index_item];
    }
    double average_weight = weight_sum / dimension;

    double density = 0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            density += profit[index_item] / weights[j][index_item];
        }
    }
    density /= dimension;

    double proximity = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        for (int j = 0; j < dimension; j++) {
            if (weights[j][voisinage[i]] > 0 && weights[j][index_item] > 0) {
                proximity += (profit[voisinage[i]] + profit[index_item]) / (weights[j][voisinage[i]] + weights[j][index_item]);
            }
        }
    }
    proximity /= nb_voisinage * dimension;

    double penalty = 1.0;
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > capacity[j]) {
            penalty = 0.1;
            break;
        }
    }

    double h = (density * 0.4 + (profit[index_item] / average_weight) * 0.3 + proximity * 0.3) * penalty;
    return h;
}
