double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double combined_factor = 0;
    double weight_factor = 0;
    for(int j=0; j<dimension; j++){
        weight_factor += weights[j][voisinage[index_item]] / capacity[j]; 
    }
    
    if(weight_factor > 0){
        double profit_factor = profit[voisinage[index_item]] / weight_factor;
        combined_factor = 1.0 / (1.0 + 1.0/profit_factor + weight_factor);
    }
    return combined_factor;
}
