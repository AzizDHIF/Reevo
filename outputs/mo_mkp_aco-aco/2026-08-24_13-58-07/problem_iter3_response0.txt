double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0, num = 0, den = 0;
    
    for (int j = 0; j < dimension; j++) {
        h += weights[j][index_item] / capacity[j];
    }
    
    if (h == 0) return 0;
    
    for (int i = 0; i < nb_voisinage; i++) {
        int neighbour_index = voisinage[i];
        den += weights[0][neighbour_index]; // Considering impact of the neighbour on the decision
    }
    
    num = profit[index_item] / (h + den); // considering both item characteristic and neighbour impact
    
    return num;
}
