double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double importance_ratio = 0.0;
    double total_weights = 0.0;

    for (int j = 0; j < dimension; j++) {
        total_weights += weights[j][index_item];
    }

    if (total_weights == 0) {
        return 0;
    }

    importance_ratio = profit[index_item] / total_weights;

    return importance_ratio;
}
