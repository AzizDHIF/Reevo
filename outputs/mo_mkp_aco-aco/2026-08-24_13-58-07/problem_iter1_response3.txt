double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double h_dim = 0;
    for(int j=0; j<dimension; ++j) {
        h_dim += weights[j][voisinage[index_item]] / capacity[j];
    }
    
    double ratio_value_weight = profit[voisinage[index_item]] / h_dim; 
    
    // Combining factors to determine promise
    double selection_promisability = ratio_value_weight * (1.0 + nb_voisinage * 0.1); 
    
    // Sparsifying heuristic by zeroing unpromising elements
    if (h_dim <= 0 || ratio_value_weight <= 0) {
        selection_promisability = 0.0;
    }
    
    return selection_promisability;
}
