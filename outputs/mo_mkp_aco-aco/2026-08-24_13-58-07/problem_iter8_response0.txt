double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0, num = 0;

    // Balance between profit and weight with respect to capacity constraints for each dimension
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] != 0) {
            h += weights[j][index_item] / capacity[j];
        }
    }

    if (h == 0) {
        return 0;
    }

    // Promisingness factor considering profit to weight balance
    num = profit[index_item] / h;

    return num;
}
