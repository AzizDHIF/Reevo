double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0.0, score = 0.0;
    
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            h += weights[j][index_item] / capacity[j]; // Utilize weight/capacity ratio
        }
    }
    
    if (h == 0.0) {
        return 0.0; // Consider invalid if total weight contribution is zero
    }
    
    score = profit[index_item] / h; // Normalize profit by weight
    
    return score;
}
