double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double promisiness = 0;

    // Calculate average weight across all dimensions for the selected item's neighbors
    double avg_weight = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        for (int j = 0; j < dimension; j++) {
            avg_weight += weights[j][voisinage[i]];
        }
    }
    avg_weight /= (nb_voisinage * dimension);

    // Calculate the promisiness based on average weight and profit of the selected item
    if (avg_weight > 0) {  // Avoid division by zero
        promisiness = profit[voisinage[index_item]] / avg_weight;
    }

    return promisiness;
}
