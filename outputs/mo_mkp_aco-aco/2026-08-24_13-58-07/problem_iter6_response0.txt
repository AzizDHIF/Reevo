double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double h = 0, num = 0;

    for (int j = 0; j < dimension; j++) {
        h += weights[j][index_item] / capacity[j];
    }

    if (h == 0) return 0;

    double normalized_profit = profit[index_item] / nb_voisinage; // Normalize profit by the number of neighbors

    num = normalized_profit / h; // Consider normalized profit and normalized weights in heuristic calculation

    return num;
}
