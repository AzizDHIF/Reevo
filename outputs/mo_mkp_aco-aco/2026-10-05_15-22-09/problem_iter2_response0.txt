double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double promising_factor = 0.0;

    for(int j = 0; j < dimension; j++){
        if(capacity[j] > 0 && weights[j][voisinage[index_item]] > 0){
            promising_factor += weights[j][voisinage[index_item]] * profit[voisinage[index_item]] / capacity[j];

    }

    return promising_factor;
}
