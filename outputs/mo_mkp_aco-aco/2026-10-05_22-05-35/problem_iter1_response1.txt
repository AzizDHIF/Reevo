double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double h=0, weight_ratio = 0;
    
    for(int j=0;j<dimension;j++){
        if(weights[j][voisinage[index_item]] > 0 && capacity[j] > 0){
            weight_ratio = weight_ratio + weights[j][voisinage[index_item]]/capacity[j];
        }
	}
    if(weight_ratio == 0){
        return 0;
    }
    
    h = (double)nb_voisinage / weight_ratio;
    
    return profit[voisinage[index_item]] * weight_ratio / h;
}
