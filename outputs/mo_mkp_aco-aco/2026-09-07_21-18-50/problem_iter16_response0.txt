double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double sum_weight = 0;
    double sum_ratio = 0;

    // Calculate the sum of weights and sum of ratios for neighboring items
    for (int j = 0; j < dimension; j++) {
        sum_weight += weights[j][voisinage[index_item]];
        if (weights[j][voisinage[index_item]] > 0 && capacity[j] > 0) {
            sum_ratio += weights[j][voisinage[index_item]] / capacity[j];
        }
    }

    // Applying balanced factor combination for holistic evaluation with a smoothing factor
    double heuristic_value = profit[voisinage[index_item]] / (sum_weight + 1) * (1 / (sum_ratio + 1)) * nb_voisinage;

    // Sparsifying by setting unpromising element to zero based on non-zero weights
    if (sum_weight == 0 || sum_ratio == 0) {
        heuristic_value = 0;
    }

    return heuristic_value;
}
