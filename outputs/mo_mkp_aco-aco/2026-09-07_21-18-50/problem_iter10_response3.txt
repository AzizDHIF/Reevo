double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double sum_weight = 0;
    double sum_ratio = 0;

    // Calculate the sum of weights and sum of ratios for neighbouring items
    for (int j = 0; j < dimension; j++) {
        sum_weight += weights[j][voisinage[index_item]];
        sum_ratio += weights[j][voisinage[index_item]] / capacity[j];
    }

    double heuristic_value = 0;

    // Check if any weight or ratio is non-zero before calculation
    if (sum_weight != 0 && sum_ratio != 0) {
        // Normalize based on the number of neighbors
        double normalized_sum_weight = sum_weight / nb_voisinage;
        double normalized_sum_ratio = sum_ratio / nb_voisinage;

        // Consider additional context-aware features in heuristic calculation
        heuristic_value = (profit[voisinage[index_item]] + 1) / (normalized_sum_weight + 1) * (1 / (normalized_sum_ratio + 1));
    }

    return heuristic_value;
}
