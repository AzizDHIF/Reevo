double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double sum_weight = 0;

    // Calculate the sum of weights for neighbouring items
    for (int j = 0; j < dimension; j++) {
        sum_weight += weights[j][voisinage[index_item]];
    }

    // Applying balanced factor combination for holistic evaluation
    double heuristic_value = profit[voisinage[index_item]] / (sum_weight + 1) * nb_voisinage;

    // Sparsifying by setting unpromising element to zero based on balance factors
    if (sum_weight == 0) {
        heuristic_value = 0;
    }

    return heuristic_value;
}
