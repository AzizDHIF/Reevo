double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]){
    double h = 0;
    double promising_factor = 0.75;
    
    for(int j = 0; j < dimension; j++){
        h += weights[j][voisinage[index_item]] / capacity[j]; 
    }
    
    double utilization_ratio = 0;
    if(h > 0) {
        for(int k = 0; k < dimension; k++){
            utilization_ratio += weights[k][index_item] / (capacity[k] * h);
        }
        utilization_ratio /= dimension;
    }
    
    double weighted_profit = profit[voisinage[index_item]] / h;
    
    double heuristic_value = promising_factor * utilization_ratio * weighted_profit;
    
    if(heuristic_value <= 0){
        return 0;
    }
    
    return heuristic_value;
}
