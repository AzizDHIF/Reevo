double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double sum_weights_capacities = 0;

    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            sum_weights_capacities += weights[j][voisinage[index_item]] / capacity[j];
        }
    }

    if (sum_weights_capacities > 0) {
        double weight_factor = 1 / sum_weights_capacities;
        double normalized_profit = (profit[voisinage[index_item]] - 1) / (NBITEMS - 1);

        return weight_factor * normalized_profit;
    } else {
        return 0;
    }
}
