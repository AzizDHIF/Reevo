double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double sum_weights_capacities = 0;
    double weight_factor = 0.0;
    double profit_factor = 0.0;

    for (int j = 0; j < dimension; j++) {
        sum_weights_capacities += weights[j][voisinage[index_item]] / capacity[j];
    }

    if (sum_weights_capacities > 0) {
        weight_factor = 1 / sum_weights_capacities;
        profit_factor = sqrt(profit[voisinage[index_item]]);
        return weight_factor * profit_factor;
    } else {
        return 0;
    }
}
