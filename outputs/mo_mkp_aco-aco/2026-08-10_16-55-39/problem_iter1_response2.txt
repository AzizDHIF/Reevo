double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double h = 0;
    double density = 0;
    double ratio = 0;
    double utilization = 0;

    // Calculate item density
    for(int j = 0; j < dimension; j++){
        density += weights[j][index_item];
    }
    density /= dimension;

    // Calculate capacity utilization ratio
    for(int j = 0; j < dimension; j++){
        utilization += (1 - (capacity[j] / (weights[j][index_item] + 1e-6)));
    }
    utilization /= dimension;

    // Calculate item profit ratio
    double max_profit = 0;
    for(int i = 0; i < NBITEMS; i++){
        if(profit[i] > max_profit){
            max_profit = profit[i];
        }
    }
    ratio = profit[index_item] / max_profit;

    // Calculate neighborhood density
    double neighborhood_density = 0;
    double neighborhood_utilization = 0;
    for(int i = 0; i < nb_voisinage; i++){
        for(int j = 0; j < dimension; j++){
            neighborhood_density += weights[j][voisinage[i]];
            neighborhood_utilization += (1 - (capacity[j] / (weights[j][voisinage[i]] + 1e-6)));
        }
    }
    neighborhood_density /= (nb_voisinage * dimension);
    neighborhood_utilization /= (nb_voisinage * dimension);

    // Calculate heuristic value
    h = ratio * utilization * (1 - neighborhood_density) * (1 - neighborhood_utilization);

    // Sparsify heuristic
    if(h < 0.1){
        h = 0;
    }

    return h;
}
