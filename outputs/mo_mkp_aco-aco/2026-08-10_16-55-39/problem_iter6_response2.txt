double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate the total weight of the item
    double total_weight = 0;
    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item];
    }
    double average_weight = total_weight / dimension;

    // Calculate the profit-to-weight ratio of the item
    double ptw_ratio = profit[index_item] / (average_weight + 1e-9);

    // If the item's weight exceeds the capacity, penalize it
    double h = ptw_ratio;
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > capacity[j]) {
            return 0;
        }
    }

    // Consider neighborhood effect if it exists
    if (nb_voisinage > 0) {
        double total_neighborhood_profit = 0;
        double total_neighborhood_weight = 0;
        for (int i = 0; i < nb_voisinage; i++) {
            total_neighborhood_profit += profit[voisinage[i]];
            for (int j = 0; j < dimension; j++) {
                total_neighborhood_weight += weights[j][voisinage[i]];
            }
        }
        double neighborhood_ptw_ratio = total_neighborhood_profit / (total_neighborhood_weight + 1e-9);
        h = h / (neighborhood_ptw_ratio + 1e-9);
    }

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 0.5 || average_weight > 1.0) {
        h = 0;
    }

    return h;
}
