double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double ptw_ratio[dimension];
    double h = 1.0;
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > capacity[j]) {
            return 0; // Penalize the item if its weight exceeds the capacity in any dimension
        }
        if (weights[j][index_item] < 1e-9) {
            ptw_ratio[j] = profit[index_item]; // Avoid division by zero
        } else {
            ptw_ratio[j] = profit[index_item] / weights[j][index_item];
        }
        if (ptw_ratio[j] < 0.1) { // Simplify by considering a threshold for the ratio
            return 0;
        }
        h *= ptw_ratio[j]; // Accumulate the ratios along all dimensions
    }

    // Consider neighborhood effect with simplified calculation
    if (nb_voisinage > 0) {
        double total_neighborhood_profit = 0;
        for (int i = 0; i < nb_voisinage; i++) {
            total_neighborhood_profit += profit[voisinage[i]];
        }
        double neighborhood_ratio = total_neighborhood_profit / (nb_voisinage * profit[index_item] + 1e-9);
        h *= neighborhood_ratio; // Incorporate neighborhood effect into the heuristic
    }

    // Sparsify the heuristic by setting unpromising elements to zero
    double average_weight = 0;
    for (int j = 0; j < dimension; j++) {
        average_weight += weights[j][index_item];
    }
    average_weight /= dimension;
    if (h < 0.5 || average_weight > 1.0) {
        h = 0;
    }

    return h;
}
