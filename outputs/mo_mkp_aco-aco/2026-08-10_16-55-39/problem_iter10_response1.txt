double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Prioritize capacity checks
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > capacity[j]) return 0; // Item exceeds capacity
    }

    // Calculate average weight and profit-to-weight ratio
    double total_weight = 0;
    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item];
    }
    double average_weight = total_weight / dimension;
    double profit_to_weight_ratio = profit[index_item] / (average_weight + 1e-9);

    // Calculate neighborhood effect
    double total_voisinage_profit = 0;
    double total_voisinage_weight = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        total_voisinage_profit += profit[voisinage[i]];
        for (int j = 0; j < dimension; j++) {
            total_voisinage_weight += weights[j][voisinage[i]];
        }
    }
    double neighborhood_ptw_ratio = (nb_voisinage > 0) ? total_voisinage_profit / (total_voisinage_weight + 1e-9) : 0;

    // Calculate heuristic value
    double h = profit_to_weight_ratio * (1 + neighborhood_ptw_ratio);

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < neighborhood_ptw_ratio * 0.5 || total_weight > dimension) {
        h = 0;
    }

    return h;
}
