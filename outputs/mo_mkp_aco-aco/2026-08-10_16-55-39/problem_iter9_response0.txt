double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Prioritize capacity checks
    int feasible = 1;
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > capacity[j]) {
            feasible = 0;
            break;
        }
    }

    if (!feasible) {
        return 0;
    }

    // Calculate the profit-to-weight ratio along each dimension
    double ptw_ratio[dimension];
    double min_ptw_ratio = DBL_MAX;
    double max_ptw_ratio = 0;
    for (int j = 0; j < dimension; j++) {
        ptw_ratio[j] = profit[index_item] / (weights[j][index_item] + 1e-9);
        if (ptw_ratio[j] < min_ptw_ratio) {
            min_ptw_ratio = ptw_ratio[j];
        }
        if (ptw_ratio[j] > max_ptw_ratio) {
            max_ptw_ratio = ptw_ratio[j];
        }
    }

    // Consider the minimum profit-to-weight ratio along all dimensions
    double h = min_ptw_ratio;

    // Consider neighborhood effect with simplified calculation
    if (nb_voisinage > 0) {
        double total_neighborhood_profit = 0;
        for (int i = 0; i < nb_voisinage; i++) {
            total_neighborhood_profit += profit[voisinage[i]];
        }
        double average_neighborhood_profit = total_neighborhood_profit / nb_voisinage;
        double neighborhood_ratio = average_neighborhood_profit / (average_neighborhood_profit + 1e-9);
        if (neighborhood_ratio > 0) {
            h = h / (neighborhood_ratio + 1e-9);
        }
    }

    // Sparsify the heuristic by setting unpromising elements to zero
    double average_weight = 0;
    for (int j = 0; j < dimension; j++) {
        average_weight += weights[j][index_item];
    }
    average_weight /= dimension;
    if (h < 0.5 || average_weight > 1.0 || max_ptw_ratio < min_ptw_ratio * 0.8) {
        h = 0;
    }

    return h;
}
