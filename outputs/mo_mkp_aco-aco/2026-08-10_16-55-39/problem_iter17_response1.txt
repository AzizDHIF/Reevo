double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate the minimum profit-to-weight ratio along all dimensions
    double min_ptw_ratio = DBL_MAX;
    double max_profit = 0.0;
    int feasible = 1;
    for (int j = 0; j < dimension; j++) {
        double ptw_ratio = profit[index_item] / (weights[j][index_item] + 1e-9);
        if (ptw_ratio < min_ptw_ratio) {
            min_ptw_ratio = ptw_ratio;
        }
        if (weights[j][index_item] > capacity[j]) {
            feasible = 0;
            break;
        }
        max_profit = (max_profit > profit[index_item]) ? max_profit : profit[index_item];
    }

    // If the item is not feasible, penalize it
    if (!feasible) {
        return 0.0;
    }

    // Consider neighborhood effect with simplified calculation
    double neighborhood_contribution = 0.0;
    if (nb_voisinage > 0) {
        double total_neighborhood_profit = 0;
        for (int i = 0; i < nb_voisinage; i++) {
            total_neighborhood_profit += profit[voisinage[i]];
        }
        double average_neighborhood_profit = total_neighborhood_profit / nb_voisinage;
        neighborhood_contribution = average_neighborhood_profit / (average_neighborhood_profit + 1e-9);
    }

    // Compute final heuristic value
    double h = min_ptw_ratio * (1.0 + neighborhood_contribution);

    // Sparsify the heuristic by setting unpromising elements to zero
    double average_weight = 0.0;
    for (int j = 0; j < dimension; j++) {
        average_weight += weights[j][index_item];
    }
    average_weight /= dimension;
    if (h < 0.5 || average_weight > 1.0 || max_profit < 0.5) {
        h = 0.0;
    }

    return h;
}
