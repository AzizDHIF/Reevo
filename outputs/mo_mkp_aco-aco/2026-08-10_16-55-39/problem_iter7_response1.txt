double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) 
{
    // Calculate the profit-to-weight ratio along each dimension with a small value added to the denominator for numerical stability
    double ptw_ratio[dimension];
    for (int j = 0; j < dimension; j++) {
        ptw_ratio[j] = profit[index_item] / (weights[j][index_item] + 1e-9);
    }

    // If the item's weight exceeds the capacity, immediately return 0 without further calculations
    double h = 1.0;
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > capacity[j]) {
            return 0;
        }
    }

    // Consider the minimum profit-to-weight ratio along all dimensions and use it directly as the heuristic value
    double min_ptw_ratio = ptw_ratio[0];
    for (int j = 1; j < dimension; j++) {
        if (ptw_ratio[j] < min_ptw_ratio) {
            min_ptw_ratio = ptw_ratio[j];
        }
    }
    h = min_ptw_ratio;

    // Consider neighborhood effect with simplified calculation, emphasizing high neighborhood profit
    if (nb_voisinage > 0) {
        double total_neighborhood_profit = 0;
        for (int i = 0; i < nb_voisinage; i++) {
            total_neighborhood_profit += profit[voisinage[i]];
        }
        double average_neighborhood_profit = total_neighborhood_profit / nb_voisinage;
        if (average_neighborhood_profit > profit[index_item]) {
            h *= 1.1; // Slightly increase the heuristic value for items with higher neighborhood profit
        } else {
            h *= 0.9; // Slightly decrease the heuristic value for items with lower neighborhood profit
        }
    }

    // Sparsify the heuristic by setting unpromising elements to zero based on a threshold
    double average_weight = 0;
    for (int j = 0; j < dimension; j++) {
        average_weight += weights[j][index_item];
    }
    average_weight /= dimension;
    if (h < 0.2 || average_weight > 1.5) {
        return 0;
    }

    return h;
}
