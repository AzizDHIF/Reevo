double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Check capacity constraints first
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > capacity[j]) return 0; // Item exceeds capacity
    }

    // Calculate the density of the item
    double density = 0;
    for (int j = 0; j < dimension; j++) {
        density += weights[j][index_item] / capacity[j];
    }
    density /= dimension;

    // Calculate the profit-to-weight ratio of the item
    double ptw_ratio = profit[index_item] / (density + 1e-9);

    // Calculate the neighborhood profit-to-weight ratio
    double total_profit = 0;
    double total_weight = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        total_profit += profit[voisinage[i]];
        for (int j = 0; j < dimension; j++) {
            total_weight += weights[j][voisinage[i]];
        }
    }
    double neighborhood_ptw_ratio = (nb_voisinage > 0 && total_weight > 0) ? total_profit / total_weight : 0;

    // Calculate the heuristic value
    double h = (1 - density) * ptw_ratio / (neighborhood_ptw_ratio + 1e-9);

    // Sparsify the heuristic by setting unpromising elements to zero
    if (density > 1.0 || (nb_voisinage > 0 && h < neighborhood_ptw_ratio * 0.5)) {
        h = 0;
    }

    return h;
}
