double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Prioritize density and profit ratios
    double max_profit_ratio = 0;
    for (int j = 0; j < dimension; j++) {
        double profit_ratio = profit[index_item] / (weights[j][index_item] + 1e-9);
        if (profit_ratio > max_profit_ratio) {
            max_profit_ratio = profit_ratio;
        }
    }

    // Consider neighborhood effect
    double neighborhood_ptw_ratio = 0;
    double total_neighborhood_profit = 0;
    double total_neighborhood_weight = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        total_neighborhood_profit += profit[voisinage[i]];
        for (int j = 0; j < dimension; j++) {
            total_neighborhood_weight += weights[j][voisinage[i]];
        }
    }
    if (nb_voisinage > 0) {
        double average_neighborhood_weight = total_neighborhood_weight / (nb_voisinage * dimension);
        neighborhood_ptw_ratio = total_neighborhood_profit / (total_neighborhood_weight + 1e-9);
    }

    // Calculate the heuristic value
    double h = max_profit_ratio / (neighborhood_ptw_ratio + 1e-9);

    // Check capacity constraints
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > capacity[j]) {
            h = 0;
            break;
        }
    }

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 0.5 || max_profit_ratio < 0.1) {
        h = 0;
    }

    return h;
}
