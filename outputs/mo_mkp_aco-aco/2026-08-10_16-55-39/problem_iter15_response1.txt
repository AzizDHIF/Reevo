double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Calculate the sum of weights for the item along all dimensions
    double sum_weights = 0;
    for (int j = 0; j < dimension; j++) {
        sum_weights += weights[j][index_item];
    }

    // Prioritize feasibility by checking if the item's weight exceeds the capacity
    double h = 1.0;
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > capacity[j]) {
            h = 0;
            break;
        }
    }

    // If the item is feasible, calculate the profit-to-weight ratio along each dimension
    if (h > 0) {
        double max_ptw_ratio = 0;
        for (int j = 0; j < dimension; j++) {
            double ptw_ratio = profit[index_item] / (weights[j][index_item] + 1e-9);
            if (ptw_ratio > max_ptw_ratio) {
                max_ptw_ratio = ptw_ratio;
            }
        }
        h = max_ptw_ratio / (sum_weights + 1e-9);
    }

    // Consider neighborhood effect with simplified calculation
    if (nb_voisinage > 0 && h > 0) {
        double total_neighborhood_profit = 0;
        for (int i = 0; i < nb_voisinage; i++) {
            total_neighborhood_profit += profit[voisinage[i]];
        }
        double average_neighborhood_profit = total_neighborhood_profit / nb_voisinage;
        double neighborhood_factor = average_neighborhood_profit / (average_neighborhood_profit + 1e-9);
        h *= neighborhood_factor;
    }

    // Sparsify the heuristic by setting unpromising elements to zero
    if (h < 0.1 || sum_weights > 1.5) {
        h = 0;
    }

    return h;
}
