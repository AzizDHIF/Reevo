double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] )
{
    double h = 0;
    double max_profit_weight_ratio = 0;

    // Calculate the profit-weight ratio of the selected item
    double item_weight = 0;
    for(int j = 0; j < dimension; j++){
        item_weight += weights[j][index_item];
    }
    if(item_weight > 0){
        max_profit_weight_ratio = profit[index_item] / item_weight;

}
