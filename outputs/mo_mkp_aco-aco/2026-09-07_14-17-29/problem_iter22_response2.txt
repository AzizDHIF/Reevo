double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double max_weight = 0;
    double max_profit = profit[voisinage[index_item]];
    
    // Find the maximum weight among neighbors in relevant dimensions
    for(int j = 0; j < dimension; j++) {
        if(weights[j][voisinage[index_item]] > max_weight && weights[j][voisinage[index_item]] <= capacity[j]) {
            max_weight = weights[j][voisinage[index_item]];
        }
    }
    
    // Sparse heuristic calculation with profit adjustment
    if(max_weight > 0) {
        h = max_profit / max_weight;
    }
    
    return h;
}
