double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double h=0, ratio = 0.0, weight_ratio = 0.0;
    
    for(int j=0;j<dimension;j++){
        weight_ratio += weights[j][index_item] / capacity[j]; 
    }
    
    for(int i=0;i<nb_voisinage;i++){
        int neighbor = voisinage[i];
        double neighbor_weight_ratio = 0.0;

        for(int j=0; j<dimension; j++){
            neighbor_weight_ratio += weights[j][neighbor] / capacity[j];
        }

        ratio = profit[neighbor] / (neighbor_weight_ratio+1e-8);  // small value to avoid divide by zero
        if (ratio > 1 || weight_ratio == 0){
            h += ratio;
        }
    }

    return h;
}
