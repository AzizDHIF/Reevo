double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] )
{
    double h = 0;
    double max_weight = 0;
    double max_profit = profit[voisinage[index_item]];

    // Calculate the useful ratios for heuristics
    double ratios[NBITEMS];
    for(int i = 0; i < NBITEMS; i++){
        double max_weight_neighbor = 0;
        for(int j = 0; j < dimension; j++){
            if(weights[j][voisinage[i]] > max_weight_neighbor){
                max_weight_neighbor = weights[j][voisinage[i]];
            }
        }
        if(max_weight_neighbor > 0){
            ratios[i] = profit[i] / max_weight_neighbor;
        }else{
            ratios[i] = 0;
        }
    }

    // Find the maximum weight among neighbors
    for(int j = 0; j < dimension; j++){
        if(weights[j][voisinage[index_item]] > max_weight){
            max_weight = weights[j][voisinage[index_item]];
        }
    }

    // Sparse heuristic calculation
    if(max_weight > 0){
        h = max_profit / max_weight;
    }

    return h;
}
