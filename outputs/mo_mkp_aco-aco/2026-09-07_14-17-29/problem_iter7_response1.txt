double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double max_profit_weight_ratio = 0;
    
    // Find the maximum profit-to-weight ratio among neighbors
    double profit_weight_ratio;
    for(int i = 0; i < nb_voisinage; i++){
        profit_weight_ratio = profit[voisinage[i]] / weights[0][voisinage[i]]; // Considering weight dimension 0 for simplicity
        if(profit_weight_ratio > max_profit_weight_ratio){
            max_profit_weight_ratio = profit_weight_ratio;
        }
    }
    
    h = max_profit_weight_ratio;
    
    return h;
}
