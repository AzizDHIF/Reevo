double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double sum_weights = 0;
    double max_weight = 0;

    for (int j = 0; j < dimension; j++) {
        sum_weights += weights[j][voisinage[index_item]];
        if (weights[j][voisinage[index_item]] > max_weight) {
            max_weight = weights[j][voisinage[index_item]];
        }
    }

    if (sum_weights > 0 && max_weight > 0) {
        h = profit[voisinage[index_item]] / (sum_weights / dimension + max_weight);
    }

    return h;
}
