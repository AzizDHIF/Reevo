double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j];
    }
    
    double importance_factor = 1.0; // Importance factor for balancing weight and profit importance
    double dimension_factor = 0.9; // Factor to penalize dimension impact
    
    return importance_factor * profit[voisinage[index_item]] / h + dimension_factor * dimension / NBITEMS;
}
