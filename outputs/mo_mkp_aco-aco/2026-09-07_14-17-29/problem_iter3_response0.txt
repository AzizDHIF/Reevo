double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double sum_weights = 0;
    double max_weight = 0;
    double max_profit = 0;
    
    // Getting the sum of weights and maximum weight among neighbors
    for (int j = 0; j < dimension; j++) {
        sum_weights += weights[j][voisinage[index_item]];
        max_weight = fmax(max_weight, weights[j][voisinage[index_item]]);
    }
    
    // Getting the maximum profit among neighbors
    max_profit = profit[voisinage[index_item]];
    
    // Sparse heuristic calculation with weighted factors
    if (sum_weights > 0 && max_weight > 0) {
        h = max_profit / (sum_weights + max_weight) * 2;
    }
    
    return h;
}
