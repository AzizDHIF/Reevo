double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_ratio = 0.6;
    double profit_ratio = 0.4;

    double total_ratio = weight_ratio + profit_ratio;
    double combined_weight = 0;

    for (int j = 0; j < dimension; j++) {
        combined_weight += weights[j][voisinage[index_item]] / capacity[j];
    }

    double weight_component = (combined_weight == 0) ? 0 : weights[0][voisinage[index_item]] / combined_weight;
    double profit_component = profit[voisinage[index_item]] / (combined_weight + 1);

    double heuristic_value = (weight_ratio * weight_component + profit_ratio * profit_component) / total_ratio;

    return (heuristic_value > 0) ? heuristic_value : 0;
}
