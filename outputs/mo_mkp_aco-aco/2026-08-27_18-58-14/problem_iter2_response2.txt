double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_ratio = 0.5;
    double profit_ratio = 0.5;
    
    double h = 0;
    for(int j = 0; j < dimension; j++){
        h += weights[j][voisinage[index_item]] / capacity[j];
    }

    double normalized_profit = profit[voisinage[index_item]] / h;

    double heuristic_value = weight_ratio * (weights[0][voisinage[index_item]] / h) + profit_ratio * normalized_profit;

    return heuristic_value > 0 ? heuristic_value : 0;
}
