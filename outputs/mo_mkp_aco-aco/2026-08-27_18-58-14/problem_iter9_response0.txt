double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double weight_ratio = 0.5;
    double profit_ratio = 0.5;

    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j];
    }

    double weight_component = h == 0 ? 0 : weights[0][voisinage[index_item]] / h;
    double profit_component = profit[voisinage[index_item]] / (h + 1);

    double normalized_weight_sum = 0;
    for (int i = 0; i < NBITEMS; i++) {
        for (int j = 0; j < dimension; j++) {
            normalized_weight_sum += weights[j][i] / capacity[j];
        }
    }

    double normalized_weight = normalized_weight_sum == 0 ? 0 : weights[0][voisinage[index_item]] / normalized_weight_sum;
    int promising = weight_component > 0 && profit_component > 0 ? 1 : 0;

    double heuristic_value = weight_ratio * normalized_weight + profit_ratio * profit_component * promising;

    return heuristic_value;
}
