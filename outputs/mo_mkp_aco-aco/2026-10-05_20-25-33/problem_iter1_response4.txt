double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double h=0;
    double sum_weights = 0;
    double sum_capacity = 0;

    for(int j=0;j<dimension;j++){
        sum_weights += weights[j][voisinage[index_item]];
        sum_capacity += capacity[j];
    }

    if(sum_capacity <= 0){
        return 0;
    }

    for(int j=0;j<dimension;j++){
        h += weights[j][voisinage[index_item]]/capacity[j]; 
    }

    return (sum_weights == 0) ? 0 : (profit[voisinage[index_item]]/h) * sum_weights / sum_capacity;
}
