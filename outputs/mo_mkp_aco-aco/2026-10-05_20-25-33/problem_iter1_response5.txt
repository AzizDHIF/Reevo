double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double h=0;
    double weight_div_capacity = 0.0;
    double profit_factor = 0.0;

        if(capacity[j] > 0.0){
            weight_div_capacity += weights[j][voisinage[index_item]]/capacity[j];
            h += weights[j][voisinage[index_item]];
        }
