double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double h=0;
	double promotes = 0.2; // Tunable promotion value
	double usefulness = 0;
		
	for(int j=0;j<dimension;j++){
		h += weights[j][voisinage[index_item]] / capacity[j]; 
		usefulness += weights[j][voisinage[index_item]] / h;
	}
	
	usefulness *= profit[voisinage[index_item]] / (double)dimension;
	
	if(usefulness < promotes){
		return 0.0;
	}else{
		return usefulness;
	}
}
