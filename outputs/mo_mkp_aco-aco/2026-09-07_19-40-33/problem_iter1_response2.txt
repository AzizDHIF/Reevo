double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double sum_weights = 0;
    double sum_capacity = 0;

    for(int j = 0; j < dimension; j++){
        sum_weights += weights[j][voisinage[index_item]];
        sum_capacity += capacity[j];
    }

    double promising_factor = sum_weights / sum_capacity;
    double potential_profit = profit[voisinage[index_item]];

    if(promising_factor < 0.5) {
        return 0.0;
    } else {
        return potential_profit * promising_factor / nb_voisinage;
    }
}
