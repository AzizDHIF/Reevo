double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double h=0;
    for(int j=0;j<dimension;j++){
        h += weights[j][voisinage[index_item]] / capacity[j]; 
    }
    double promising_factor = 2/(1+h); // Combining various factors
    
    if(promising_factor >= 0.4){
        return profit[voisinage[index_item]] * promising_factor;
    }
    else{
        return 0; // Sparsifying the heuristic by setting unpromising elements to zero
    }
}
