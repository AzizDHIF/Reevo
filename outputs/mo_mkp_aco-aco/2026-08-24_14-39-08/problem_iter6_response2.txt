double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double profit_factor = 0.8, weight_factor = 0.2;
    double h = 0, weight_div_capacity, score = 0;

    for (int j = 0; j < dimension; j++) {
        weight_div_capacity = weights[j][voisinage[index_item]] / capacity[j];
        h += weight_div_capacity;

        if (weight_div_capacity <= 0.01) {
            return 0; // sparsify if element unpromising
        }
    }

    if (h <= 0.01) {
        return 0; // sparsify if element unpromising
    }

    score = profit[voisinage[index_item]] * profit_factor - h * weight_factor;

    return score;
}
