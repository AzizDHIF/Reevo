double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double profit_factor = 0.8, weight_penalty = 0.01;
    double h = 0;

    for (int i = 0; i < nb_voisinage; i++) {
        int neighbour_index = voisinage[i];
        double neighbour_diff_ratio = 0;

        for (int j = 0; j < dimension; j++) {
            neighbour_diff_ratio += (weights[j][index_item] - weights[j][neighbour_index]) / capacity[j];
        }

        double neighbour_score = profit[neighbour_index] * profit_factor - neighbour_diff_ratio * (1 - profit_factor);
        if (neighbour_score > 0) {
            h += neighbour_score;
        }
    }

    double weight_ratio = 0;
    for (int j = 0; j < dimension; j++) {
        weight_ratio += weights[j][index_item] / capacity[j];
    }

    double score = profit[index_item] * profit_factor - weight_ratio * (1 - profit_factor) - h;

    return score;
}
