double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double profit_factor = 0.8, weight_factor = 0.2, weight_penalty = 0.01;
    double h = 0;

    for (int j = 0; j < dimension; j++) {
        double weight_ratio = weights[j][index_item] / capacity[j];
        h += weight_ratio;

        if (weight_ratio <= weight_penalty) return 0; // sparsify if element unpromising
    }

    if (h <= weight_penalty) return 0; // sparsify if element unpromising

    double normalized_profit = (profit[index_item] - profit[voisinage[0]]) / profit[voisinage[0]]; // Critical analysis
    double score = profit_factor * normalized_profit + weight_factor * (1 - h);

    return score;
}
