double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double h=0, cost_factor = 0.5, profit_death_threshold = 0.001;
    double weight_div_capacity[dimension];
    for(int j=0;j<dimension;j++){
        weight_div_capacity[j]=weights[j][voisinage[index_item]]/capacity[j];
	h += weight_div_capacity[j];
	if(weight_div_capacity[j] <= 0.01) return 0; //sparsify if element unpromising
    }
    if (h <=0.01) return 0; // sparsify if element unpromising
    double score = profit[voisinage[index_item]] * (1 - cost_factor) + 1/h * cost_factor;
    if(profit[voisinage[index_item]] < profit_death_threshold) return  0; // more flexible domination check
    return score;
}
