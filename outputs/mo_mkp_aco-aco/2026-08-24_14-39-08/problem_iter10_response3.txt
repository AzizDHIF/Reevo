double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double profit_factor = 0.8, weight_penalty = 0.01;
    double relevance = 0;

    for (int j = 0; j < dimension; j++) {
        double weight_ratio = weights[j][index_item] / capacity[j];
        relevance += weight_ratio;

        if (weight_ratio <= weight_penalty) {
            return 0; // sparsify if element unpromising
        }
    }

    if (relevance <= weight_penalty) {
        return 0; // sparsify if element unpromising
    }

    double score = profit[index_item] * profit_factor - relevance * (1 - profit_factor);

    return score;
}
