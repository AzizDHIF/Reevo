double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ) {

    double profit_factor = 0.7, weight_factor = 0.3;
    double weight_penalty = 0.01;
    double total_weight_contribution = 0;
    
    for (int j = 0; j < dimension; j++) {
        double weight_ratio = weights[j][index_item] / capacity[j];
        total_weight_contribution += weight_ratio;
        
        if (weight_ratio <= weight_penalty) return 0; // Sparsify if element unpromising
    }
    
    if (total_weight_contribution <= weight_penalty) return 0; // Sparsify if element unpromising
    
    double score = profit[index_item] * profit_factor - total_weight_contribution * weight_factor;
    
    return score;
}
