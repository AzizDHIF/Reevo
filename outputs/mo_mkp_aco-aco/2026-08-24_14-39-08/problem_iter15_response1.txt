double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ) {
    double profit_factor = 0.8, weight_factor = 0.2;
    double combined_weights = 0, profit_improvement, score;

    for (int j = 0; j < dimension; j++) {
        combined_weights += weights[j][index_item] / capacity[j];
    }

    if (combined_weights > 1) return 0; // Discard if total weight exceeds capacity

    profit_improvement = profit[index_item] - profit[voisinage[0]]; // Calculate profit improvement compared to the best neighbor

    // Penalize early for weight breach, prioritize profit increment, and balance weight and profit factors
    score = (profit_improvement > 0) ? profit_factor * profit_improvement - weight_factor * combined_weights : 0;

    return score;
}
