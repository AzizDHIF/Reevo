double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double profit_factor = 0.7, weight_factor = 0.3, profit_improvement, score;
    double weight_ratio = 0;

    for (int j = 0; j < dimension; j++) {
        weight_ratio += weights[j][index_item] / capacity[j];
    }

    if (weight_ratio > 1) return 0; // Discard if total weight exceeds capacity

    profit_improvement = profit[index_item] - profit[voisinage[0]]; // Calculate profit improvement compared to the best neighbor
    score = profit_factor * profit_improvement - weight_factor * weight_ratio; // Combine profit and weight factors

    return score;
}
