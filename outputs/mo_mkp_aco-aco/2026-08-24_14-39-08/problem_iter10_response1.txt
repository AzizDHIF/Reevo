double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double profit_factor = 0.8, weight_penalty = 0.01;
    double weight_contribution = 0, profit_contribution = 0;
    
    for (int j = 0; j < dimension; j++) {
        double weight_ratio = weights[j][index_item] / capacity[j];
        weight_contribution += weight_ratio;
        
        if (weight_ratio <= weight_penalty) return 0; // sparsify if element unpromising
    }
    
    if (weight_contribution <= weight_penalty) return 0; // sparsify if element unpromising
    
    profit_contribution = profit[index_item] * profit_factor - weight_contribution * (1 - profit_factor);
    
    return profit_contribution;
}
