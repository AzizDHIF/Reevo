double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ) {
    double weighted_profit_improvement = 0;
    double weight_penalty = 0;
    double score;

    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] > capacity[j]) {
            weight_penalty += weights[j][index_item] - capacity[j]; // Penalize weight breaches
        }
        weighted_profit_improvement += (weights[j][index_item] / capacity[j]) * (profit[index_item] - profit[voisinage[0]]); // Weighted profit improvement considering all dimensions
    }

    score = weighted_profit_improvement - weight_penalty; // Combine weighted profit improvement and weight penalty

    return score;
}
