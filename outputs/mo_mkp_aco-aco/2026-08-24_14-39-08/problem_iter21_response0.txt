double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double profit_factor = 0.6, weight_factor = 0.4;
    double score = 0;

    double total_weight = 0, total_profit = 0;
    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item];
    }

    if (total_weight > 1) return 0; // Discard if total weight exceeds capacity

    for (int i = 0; i < nb_voisinage; i++) {
        total_profit += profit[voisinage[i]];
    }

    double profit_score = profit[index_item] - total_profit / nb_voisinage; // Calculate profit improvement compared to average neighbor
    double weight_score = 1 - total_weight; // Weight score based on available capacity

    score = profit_factor * profit_score - weight_factor * weight_score; // Combine profit and weight factors

    return score;
}
