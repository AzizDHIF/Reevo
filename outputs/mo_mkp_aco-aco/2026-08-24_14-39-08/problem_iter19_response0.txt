double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {

    double profit_factor = 0.7, weight_factor = 0.3;
    double score = 0;
    double combined_weights = 0;
    double profit_improvement = 0;

    for (int j = 0; j < dimension; j++) {
        combined_weights += weights[j][index_item] / capacity[j];
    }

    if (combined_weights > 1) return 0; // Discard if total weight exceeds capacity

    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_combined_weights = 0;
        for (int j = 0; j < dimension; j++) {
            neighbor_combined_weights += weights[j][voisinage[i]] / capacity[j];
        }
        double neighbor_profit_improvement = profit[index_item] - profit[voisinage[i]];
        profit_improvement = fmax(profit_improvement, neighbor_profit_improvement);
        combined_weights = fmax(combined_weights, neighbor_combined_weights);
    }

    score = profit_factor * profit_improvement - weight_factor * combined_weights;

    return score;
}
