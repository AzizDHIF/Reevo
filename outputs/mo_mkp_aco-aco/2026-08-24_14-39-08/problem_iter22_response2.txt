double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double profit_weight_ratio = 0.0;

    for (int j = 0; j < dimension; j++) {
        profit_weight_ratio += profit[index_item] / weights[j][index_item];
    }

    if (profit_weight_ratio <= 0) return 0; // Discard if profit-weight ratio is non-positive

    return profit_weight_ratio;
}
