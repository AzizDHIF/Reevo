double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double profit_factor = 0.8, weight_penalty = 0.01;
    double sparsity_threshold = 0.1; // Threshold to determine disregarded items based on weight ratio
    double relevance_factor = 0.5; // Weight factor for determining item relevance
    double h = 0;

    for (int j = 0; j < dimension; j++) {
        double weight_ratio = weights[j][index_item] / capacity[j];
        h += weight_ratio;

        if (weight_ratio <= weight_penalty) return 0; // sparsify if element unpromising
    }

    if (h <= sparsity_threshold) return 0; // sparsify if element unpromising

    double normalized_profit = profit[index_item] / (1 + h); // Normalize profit based on combined weights

    double relevance = relevance_factor * normalized_profit + (1 - relevance_factor) * h; // Calculate item relevance

    return relevance;
}
