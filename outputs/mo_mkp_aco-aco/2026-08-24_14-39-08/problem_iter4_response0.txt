double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] )
{
    double h = 0;
    double promising_factor = 0.9;

    // Calculate the promising factor based on weight to capacity ratio for each dimension
    for (int j = 0; j < dimension; j++) {
        double w = weights[j][voisinage[index_item]] / capacity[j];
        h += w;

        // Scale promising factor based on weight to capacity ratio for each dimension
        promising_factor *= (1 - w);
    }

    return promising_factor * profit[voisinage[index_item]] / h;
}
