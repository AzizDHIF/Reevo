double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double profit_factor = 0.8, weight_factor = 0.2;
    double total_weight = 0, profit_improvement, score;

    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item];
    }

    if (total_weight > capacity[0]) return 0; // Discard if total weight exceeds capacity for a single knapsack

    profit_improvement = profit[index_item] - profit[voisinage[0]]; // Calculate profit improvement compared to the best neighbor
    score = profit_factor * profit_improvement - weight_factor * (total_weight / capacity[0]); // Combine profit and weight factors

    return score;
}
