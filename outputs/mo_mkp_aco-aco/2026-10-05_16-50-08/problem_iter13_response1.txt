double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weight_sum = 0;
    double distance_sum = 0;
    double low_profit_penalty = 0.1; // Penalty for low-profit items
    
    for (int j = 0; j < dimension; j++) {
        double weight_item = weights[j][voisinage[index_item]];
        if (weight_item > 0) {
            weight_sum += weight_item / capacity[j];
            distance_sum += fmax(0, weight_item - capacity[j]); // Using fmax to enforce non-negativity
        }
    }
    
    double weight_factor = 0.6;  // Weight given to weight-related score
    double distance_factor = 0.4;  // Weight given to distance-related score
    
    double profit_normalized = profit[voisinage[index_item]] / (double)NBITEMS;
    double weight_score = weight_factor / (1 + weight_sum);
    double distance_score = distance_factor / (1 + distance_sum);
    
    double heuristic_value = profit_normalized * weight_score - low_profit_penalty * profit[voisinage[index_item]] + distance_score; // Penalizing low-profit items
    
    if (weight_sum < 0.5 || distance_sum > 0.5) {
        heuristic_value = 0;
    }
    
    return heuristic_value;
}
