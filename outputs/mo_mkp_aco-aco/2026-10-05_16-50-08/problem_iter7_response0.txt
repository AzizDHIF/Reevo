double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weight_sum = 0;
    double distance_sum = 0;
    
    for (int j = 0; j < dimension; j++) {
        double normalized_weight = weights[j][voisinage[index_item]] / capacity[j];
        weight_sum += normalized_weight;
        
        double excess_weight = (weights[j][voisinage[index_item]] > capacity[j]) ? (weights[j][voisinage[index_item]] - capacity[j]) : 0;
        distance_sum += excess_weight;
    }
    
    // Incorporating factors and damping for item selection
    double weight_factor = 0.6;
    double distance_factor = 0.4;
    
    // Combining normalized factors with damping for heuristic value calculation
    double heuristic_value = (profit[voisinage[index_item]] / NBITEMS) * weight_factor / (1 + weight_sum) + distance_factor / (1 + distance_sum);
    
    // Sparsifying the heuristic by setting unpromising elements to zero based on sensitivity analysis
    if (weight_sum < 0.6 || distance_sum > 0.4) {
        heuristic_value = 0;
    }
    
    return heuristic_value;
}
