double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weight_sum = 0;
    double distance_sum = 0;
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][voisinage[index_item]] / capacity[j]; 
        distance_sum += (double)(weights[j][voisinage[index_item]] > capacity[j]) * (weights[j][voisinage[index_item]] - capacity[j]);
    }
    // Utilizing a combination of factors
    double heuristic_value = profit[voisinage[index_item]] / (weight_sum + distance_sum);

    // Sparsifying the heuristic by setting unpromising elements to zero
    if (weight_sum < 1) {
