double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weight_sum = 0;
    double distance_sum = 0;

    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][voisinage[index_item]] / capacity[j];
        distance_sum += fmax(weights[j][voisinage[index_item]] - capacity[j], 0);
    }

    // Updated factors for different components
    double profit_factor = 0.6;
    double weight_factor = 0.3;
    double distance_factor = 0.1;

    // Computing heuristic value with adjusted weights and factors
    double heuristic_value = (profit[voisinage[index_item]] / (double)NBITEMS) * profit_factor / (1 + weight_sum) + weight_factor / (1 + distance_sum);

    // Sparsifying unpromising elements based on component sensitivity
    if (weight_sum < 0.8 || distance_sum > 0.2) {
        heuristic_value = 0;
    }

    return heuristic_value;
}
