double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double added_weight = 0;
    double weight_ratio = 0;
    double promising_factor = 0;

    for (int j = 0; j < dimension; j++) {
        added_weight += weights[j][voisinage[index_item]];
        if (weights[j][voisinage[index_item]] > capacity[j]) {
            return 0; // Set unpromising elements to zero
        }
        if (weights[j][voisinage[index_item]] != 0) {
            weight_ratio += weights[j][voisinage[index_item]] / capacity[j];
        }
    }

    double weight_penalty = 1.0;
    for (int j = 0; j < dimension; j++) {
        if (weights[j][voisinage[index_item]] == 0) {
            weight_penalty = 0.5; // Penalize items with zero weight
            break;
        }
    }

    promising_factor = profit[voisinage[index_item]] * weight_penalty / (added_weight * weight_ratio + 1);

    return promising_factor;
}
