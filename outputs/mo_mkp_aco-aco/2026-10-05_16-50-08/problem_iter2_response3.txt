double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 0.0;
    
    // Calculate the sum ratio of weights to capacity
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] != 0) {
            h += weights[j][voisinage[index_item]] / capacity[j];
        }
    }
    
    // Calculate the promising factor
    double promising_factor = (h != 0) ? profit[voisinage[index_item]] / h : 0.0;

    return promising_factor;
}
