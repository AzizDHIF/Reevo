    double weight_sum = 0;
    double distance_sum = 0;

        distance_sum += fmax(0, weights[j][voisinage[index_item]] - capacity[j]); // Handling zero weights and balancing weight vs capacity
    }

    // Utilizing a combination of factors with adjusted influence factors
    double heuristic_value = profit[voisinage[index_item]] / (weight_sum + 0.1 * distance_sum); // Adjusting influence factors effectively
