double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double h = 0;
    double max_weight = 0;
    double total_profit = 0;
    for(int j = 0; j < dimension; j++){
        h += weights[j][index_item] / capacity[j];
        if(weights[j][index_item] > max_weight){
            max_weight = weights[j][index_item];
        }
        for(int i = 0; i < nb_voisinage; i++){
            total_profit += profit[voisinage[i]];
        }
    }
    double average_profit = total_profit / nb_voisinage;
    double h_value = (double)profit[index_item] / (h * average_profit);
    // Sparsifying the heuristic by setting unpromising elements to zero
    if(h_value < 0.5 || max_weight > 1.5){
        return 0;
    } else {
        return h_value;
    }
}
