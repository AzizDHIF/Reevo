double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) 
{
    double h = 0;
    double total_weight = 0;
    double max_profit = 0;
    double neighbor_impact_sum = 0;
    double neighbor_profit_sum = 0;
    int valid_neighbor_count = 0;

    // Calculate total weight and max profit
    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item];
        h += weights[j][index_item] / capacity[j];
    }

    // Simplify calculations for a balanced approach
    double density = (total_weight > 0) ? profit[index_item] / total_weight : 0;
    double average_profit = 0;

    // Calculate average profit and neighbor impact
    for (int i = 0; i < nb_voisinage; i++) {
        if (profit[voisinage[i]] > max_profit) {
            max_profit = profit[voisinage[i]];
        }
        if (profit[voisinage[i]] > 0) {
            neighbor_profit_sum += profit[voisinage[i]];
            valid_neighbor_count++;
        }
    }
    if (valid_neighbor_count > 0) {
        average_profit = neighbor_profit_sum / valid_neighbor_count;
    }

    // Combine density, profit, and neighbor impact
    double combined_heuristic = density + average_profit;

    // Penalize high max profits with low heuristics
    if (max_profit > profit[index_item] && combined_heuristic < 0.5) {
        combined_heuristic = 0;
    }

    // Sparsify the heuristic by setting unpromising elements to zero
    if (combined_heuristic < 0.1) {
        combined_heuristic = 0;
    }

    return combined_heuristic;
}
