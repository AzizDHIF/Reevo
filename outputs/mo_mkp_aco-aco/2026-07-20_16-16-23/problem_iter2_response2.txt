double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double total_weight = 0;
    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item];
        h += weights[j][index_item] / capacity[j];
    }
    double density = profit[index_item] / total_weight;
    double average_profit = 0;
    int count = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (profit[voisinage[i]] > 0) {
            average_profit += profit[voisinage[i]];
            count++;
        }
    }
    if (count > 0) {
        average_profit /= count;
    }
    double neighbor_impact = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        double total_weight_neighbor = total_weight + weights[0][voisinage[i]];
        for (int j = 1; j < dimension; j++) {
            total_weight_neighbor += weights[j][voisinage[i]];
        }
        neighbor_impact += profit[voisinage[i]] / total_weight_neighbor;
    }
    double overall_heuristic = (density + average_profit + neighbor_impact) / (h + 1);
    double max_profit = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (profit[voisinage[i]] > max_profit) {
            max_profit = profit[voisinage[i]];
        }
    }
    if (max_profit > profit[index_item] && overall_heuristic < 0.5) {
        overall_heuristic = 0;
    }
    return overall_heuristic;
}
