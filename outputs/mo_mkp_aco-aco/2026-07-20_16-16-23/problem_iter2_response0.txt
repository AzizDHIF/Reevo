double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double total_weight = 0;
    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item];
        h += weights[j][index_item] / capacity[j];
    }
    double density = profit[index_item] / total_weight;
    double neighbor_impact = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_density = profit[voisinage[i]] / total_weight;
        neighbor_impact += neighbor_density;
    }
    double average_profit = 0;
    int count = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        if (profit[voisinage[i]] > 0) {
            average_profit += profit[voisinage[i]];
            count++;
        }
    }
    if (count > 0) {
        average_profit /= count;
    }
    double overall_heuristic = (density + average_profit + neighbor_impact) / (h + 1);
    if (h > 1) {
        overall_heuristic *= (1 - (h - 1));
    }
    return overall_heuristic;
}
