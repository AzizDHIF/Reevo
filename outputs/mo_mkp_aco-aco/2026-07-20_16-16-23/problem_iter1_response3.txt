double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    if (nb_voisinage == 0) return 0.0;

    double total_profit = 0.0;
    double total_weight = 0.0;
    double max_weight_dim = 0.0;
    int num_voisinage_used = 0;

    for (int i = 0; i < nb_voisinage; i++) {
        int item_idx = voisinage[i];
        double item_weight = 0.0;
        double item_max_weight_dim = 0.0;

        for (int j = 0; j < dimension; j++) {
            double item_weight_j = weights[j][item_idx];
            item_weight += item_weight_j;
            if (item_weight_j > item_max_weight_dim) item_max_weight_dim = item_weight_j;
        }

        if (item_max_weight_dim > max_weight_dim) max_weight_dim = item_max_weight_dim;

        total_weight += item_weight;
        total_profit += profit[item_idx];
        num_voisinage_used++;
    }

    if (total_weight == 0.0) return 0.0;

    double density = total_profit / total_weight;
    double space LEFT = 0.0;

    for (int j = 0; j < dimension; j++) {
        double remaining_cap = capacity[j] - weights[j][index_item];
        if (remaining_cap > max_weight_dim && total_weight <= capacity[j]) {
            space_LEFT += remaining_cap - max_weight_dim;
        }
    }

    double h = density * space_LEFT;
    for (int i = 0; i < nb_voisinage; i++) {
        if (index_item == voisinage[i]) {
            return 0.0;
        }
    }

    return h > 0 ? h : 0.0;
}
