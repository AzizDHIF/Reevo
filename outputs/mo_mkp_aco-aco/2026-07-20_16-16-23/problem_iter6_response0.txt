double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double total_weight = 0;
    double h = 0;
    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item];
        h += weights[j][index_item] / capacity[j];
    }

    double neighbor_impact = 0;
    double average_profit = 0;
    int count = 0;
    double max_profit = 0;

    for (int i = 0; i < nb_voisinage; i++) {
        double total_weight_neighbor = 0;
        for (int j = 0; j < dimension; j++) {
            total_weight_neighbor += weights[j][voisinage[i]];
        }
        neighbor_impact += profit[voisinage[i]] / (total_weight + total_weight_neighbor);
        if (profit[voisinage[i]] > max_profit) {
            max_profit = profit[voisinage[i]];
        }
        if (profit[voisinage[i]] > 0) {
            average_profit += profit[voisinage[i]];
            count++;
        }
    }

    double density = (total_weight > 0) ? profit[index_item] / total_weight : 0;
    if (count > 0) {
        average_profit /= count;
    }

    // Introduce weighting and normalizing factors
    double weight_density = 0.4;
    double weight_average_profit = 0.3;
    double weight_neighbor_impact = 0.3;

    double overall_heuristic = (weight_density * density + weight_average_profit * average_profit + weight_neighbor_impact * neighbor_impact) / (h + 1);

    if (max_profit > profit[index_item] && overall_heuristic < 0.5) {
        overall_heuristic = 0;
    }

    return overall_heuristic;
}
