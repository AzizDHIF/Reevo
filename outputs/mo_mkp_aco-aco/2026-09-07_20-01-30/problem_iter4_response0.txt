double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_ratio = 0.0, profit_ratio = 0.0;
    
    for(int j = 0; j < dimension; j++) {
        weight_ratio += weights[j][voisinage[index_item]] / capacity[j];
    }
    
    if (weight_ratio <= 0) {
        return 0.0;
    }
    
    profit_ratio = profit[voisinage[index_item]] / weight_ratio;
    
    if(profit_ratio <= 0) {
        return 0.0;
    }
    
    return profit_ratio;
}
