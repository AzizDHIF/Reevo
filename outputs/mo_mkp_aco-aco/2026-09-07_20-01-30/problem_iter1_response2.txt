double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double sum_weights = 0;
    double sum_profit = 0;
    
    for(int j=0; j<dimension; j++){
        sum_weights += weights[j][index_item] / capacity[j];
    }
    
    for(int i=0; i<nb_voisinage; i++){
        int neighbor_index = voisinage[i];
        double neighbor_weight_ratio = 0;
        double neighbor_profit = profit[neighbor_index];
    
        if(neighbor_index != index_item){
            neighbor_weight_ratio = sum_weights * (weights[0][neighbor_index]/capacity[0]) + (weights[1][neighbor_index]/capacity[1]);
            sum_profit += neighbor_profit / neighbor_weight_ratio;
        }
    }
   
    if(sum_profit == 0 || sum_weights == 0){
        return 0;
    }
    
    return sum_profit / sum_weights;
}
