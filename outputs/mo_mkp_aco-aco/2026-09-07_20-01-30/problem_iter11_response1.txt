double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0.0;
    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j];
    }

    double factor1 = profit[voisinage[index_item]] / h;
    double factor2 = (capacity[0] - weights[0][voisinage[index_item]]) * profit[voisinage[index_item]];
    double factor3 = weights[1][voisinage[index_item]] * profit[voisinage[index_item]];
    double factor4 = weights[2][voisinage[index_item]] / capacity[1];

    // Consider diverse factors with varied contributions
    double factor5 = profit[voisinage[index_item]] * weights[0][voisinage[index_item]] / capacity[0];
    double factor6 = profit[voisinage[index_item]] * weights[1][voisinage[index_item]] / capacity[1];

    // Sparsify heuristic based on multiple factors and item properties
    if (factor1 + factor2 + factor3 + factor4 + factor5 + factor6 > 0) {
        return factor1 + factor2 + factor3 + factor4 + factor5 + factor6;
    } else {
        return 0.0;
    }
}
