double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0.0;
    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j];
    }

    double factor1 = profit[voisinage[index_item]] / h;
    double factor2 = (capacity[0] - weights[0][voisinage[index_item]]) * profit[voisinage[index_item]];
    double factor3 = weights[1][voisinage[index_item]] * profit[voisinage[index_item]];

    // Sparse heuristic based on combined factors
    double heuristic_value = factor1 + factor2 + factor3;
    return (heuristic_value > 0) ? heuristic_value : 0.0;
}
