double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double h = 0;
    for (int j = 0; j < dimension; j++)
    {
        h += weights[j][index_item] / capacity[j]; // Consider capacity in weight analysis
    }

    double factor1 = profit[index_item] / h;
    double factor2 = 0;
    for (int k = 0; k < nb_voisinage; k++)
    {
        int neighbor = voisinage[k];
        factor2 += (capacity[0] - weights[0][neighbor]) * profit[neighbor]; // Include neighbors and remaining capacity
    }

    if (factor1 > 0 && factor2 > 0)
    {
        return factor1 + factor2;
    }
    else
    {
        return 0.0;
    }
}
