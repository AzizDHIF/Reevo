double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ) 
{
    double h=0;
    for(int j=0; j<dimension; j++){
        h += weights[j][voisinage[index_item]] / capacity[j];
    }

    double weight_profit_ratio = profit[voisinage[index_item]] / h;
    double weight_remaining_profit = (capacity[0] - weights[0][voisinage[index_item]]) * profit[voisinage[index_item]];

    // Promising selection criteria based on weight-profit ratio and remaining profit potential
    if(weight_profit_ratio > 0 && weight_remaining_profit > 0) {
        return weight_profit_ratio + weight_remaining_profit;
    } else {
        return 0.0;
    }
}
