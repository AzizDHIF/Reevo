double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double weight_ratio = 0;
    double profit_score = 0;

    for (int j = 0; j < dimension; j++) {
        weight_ratio += weights[j][voisinage[index_item]] / capacity[j];
    }

    profit_score = profit[voisinage[index_item]] / weight_ratio;

    // Balancing objective and constraint ratio
    if (weight_ratio > 1.5) {
        return 0.0;
    }

    // Checking stability in profit score
    if (profit_score < 0.1) {
        return 0.0;
    }

    return profit_score;
}
