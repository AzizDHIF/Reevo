double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]){
    double h = 0;
	
    // Calculate a combined factor based on different criteria
    for(int j = 0; j < dimension; j++){
        h = h + weights[j][voisinage[index_item]] / capacity[j]; 
    }
	
    // Combine profit and weight factors to determine item promise
    double factor_1 = profit[voisinage[index_item]] / h;
	
    // Set unpromising items based on weight threshold
    double threshold = 0.3; // Threshold for considering factor_1
    if (h > capacity[0] * threshold) {
        factor_1 = 0.0;
    }
	
    return factor_1;
}
