double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] )
{
    double h = 0;
    
    // Considering load of neighbors and normalized weights for each dimension
    for(int j = 0; j < dimension; j++){
        h += weights[j][index_item] / capacity[j];
    }
    
    // Combining load, profit, and weight of neighboring item
    double load_ratio = (h > 0) ? 1 / h : 0; // Avoid division by zero
    double promising_factor = profit[index_item] * load_ratio / weights[0][index_item]; // Considering weight of the item
    
    return promising_factor;
}
