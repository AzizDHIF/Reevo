double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] ){
    double h=0, sum_weight_ratios=0, cutoff_ratio = 0.7;
    
    for(int j=0; j<dimension; j++){
        h += weights[j][index_item] / capacity[j]; 
        sum_weight_ratios += weights[j][index_item] / weights[j][voisinage[index_item]];
    }
    
    if(sum_weight_ratios < dimension * cutoff_ratio){
        return 0.0;
    }
    
    return (profit[index_item] - sum_weight_ratios) / h;
}
