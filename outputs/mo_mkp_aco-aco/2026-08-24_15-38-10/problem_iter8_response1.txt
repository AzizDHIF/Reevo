double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS] )
{
    double impact = 0;
    
    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] != 0) {
            double profit_weight_ratio = profit[index_item] / weights[j][index_item];
            double capacity_ratio = capacity[j] > 0 ? 1 / capacity[j] : 0;
            double item_contribution = profit_weight_ratio * capacity_ratio;
            impact += item_contribution;
        }
    }
    
    return impact;
}
