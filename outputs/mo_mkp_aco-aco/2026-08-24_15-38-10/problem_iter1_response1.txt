double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]){
    double h=0;
    double combined_factor = 0;
    
    for(int i=0; i<nb_voisinage; i++){
        double weight_sum = 0;
        
        for(int j=0; j<dimension; j++){
            weight_sum += weights[j][voisinage[i]] / capacity[j];
        }
        
        double promise_factor = profit[voisinage[i]] / weight_sum;
        if(promise_factor > 0){
            combined_factor += promise_factor;
        }
    }
    
    return combined_factor;
}
