double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]){
    double h=0, total_weight=0, total_profit=0;
    
    for(int j=0; j<dimension; j++){
        h += weights[j][voisinage[index_item]] / capacity[j]; 
        total_weight += weights[j][voisinage[index_item]];
    }

    total_profit = profit[voisinage[index_item]];

    if(total_weight <= capacity[0]){
        return total_profit * (1/h) + (total_weight/capacity[0]);
    } else{
        return 0;
    }
}
