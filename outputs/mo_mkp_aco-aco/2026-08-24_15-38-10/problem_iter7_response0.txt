double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double impact = 0;

    for (int j = 0; j < dimension; j++) {
        double weight_j = weights[j][index_item];
        if (weight_j != 0) {
            double profit_weight_ratio = profit[index_item] / weight_j;
            double weight_contribution = weight_j / capacity[j];
            double item_contribution = profit_weight_ratio * weight_contribution;
            impact += item_contribution;
        }
    }

    return impact;
}
