double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double impact = 0.0;
    
    double max_profit = 0.0;
    for (int i = 0; i < nb_voisinage; i++) {
        int neighbor_index = voisinage[i];
        double profit_diff = profit[neighbor_index] - profit[index_item];
        double weight_diff = 0.0;
        for (int j = 0; j < dimension; j++) {
            weight_diff += weights[j][neighbor_index] - weights[j][index_item];
        }
        double profit_weight_ratio = (profit_diff > 0 && weight_diff < 0) ? profit_diff / weight_diff : 0.0;
        double item_contribution = profit_weight_ratio * (1.0 - weight_diff);
        
        if (item_contribution > 0 && item_contribution > max_profit) {
            max_profit = item_contribution;
        }
    }
    
    impact = max_profit;
    
    return impact;
}
