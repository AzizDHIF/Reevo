double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double impact = 0;

    for (int j = 0; j < dimension; j++) {
        if (weights[j][index_item] != 0) {
            double profit_weight_ratio = profit[index_item] / weights[j][index_item];
            double weight_contribution = weights[j][index_item] / capacity[j];
            double item_contribution = profit_weight_ratio * weight_contribution;

            // Additional consideration for sparsity
            if (profit[j] == 0) {
                item_contribution = 0;
            }

            impact += item_contribution;
        }
    }

    return impact;
}
