double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double h = 0;
    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j];
    }

    double w_ratio = 0.3;
    double p_ratio = 0.7;

    double normalized_weight = weights[0][index_item] / capacity[0];
    double profit_density = profit[voisinage[index_item]] / weights[0][voisinage[index_item]];

    return (p_ratio * profit_density) - (w_ratio * normalized_weight);
}
