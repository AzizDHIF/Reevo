double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS])
{
    double total_weight = 0.0;
    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][index_item];
    }

    double combined_score = 0.0;
    for (int i = 0; i < nb_voisinage; i++) {
        double neighbor_effectiveness = profit[voisinage[i]] / total_weight;
        double neighbor_custom_factor = capacity[0] - weights[0][voisinage[i]];

        if (neighbor_custom_factor >= 0) {
            double neighbor_contribution = 0;
            for (int j = 0; j < dimension; j++) {
                neighbor_contribution += weights[j][voisinage[i]] / capacity[j];
            }

            double w_ratio = 0.6; // Adjusted weight ratio
            double p_ratio = 0.4; // Adjusted profit ratio

            double neighbor_score = ((1 - w_ratio) * neighbor_effectiveness) + (w_ratio * p_ratio);

            combined_score = fmax(combined_score, neighbor_score / neighbor_contribution);
        }
    }

    return combined_score;
}
