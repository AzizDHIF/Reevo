double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double total_weight = 0;

    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][voisinage[index_item]] / capacity[j];
    }

    double weight_ratio = 0.5;
    double profit_ratio = 0.5;

    double profit_density = profit[voisinage[index_item]] / total_weight;

    double remaining_capacity_ratio = (capacity[0] - weights[0][index_item]) / capacity[0];

    if (remaining_capacity_ratio < 0) {
        return 0.0;
    }

    return ((1 - weight_ratio) * profit_density) + (weight_ratio * profit_ratio);
}
