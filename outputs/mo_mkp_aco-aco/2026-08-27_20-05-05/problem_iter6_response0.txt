double heuristic_v2(int index_item, double weights[dimension][NBITEMS], double capacity[dimension], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double remaining_capacity = capacity[0]; // Considering only the first dimension for simplicity

    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j];
    }

    double effectiveness = profit[voisinage[index_item]] / h;

    double custom_factor = remaining_capacity - weights[0][index_item]; // Custom factor combining weight and capacity

    if (custom_factor < 0) {
        return 0.0;
    }

    return effectiveness * custom_factor;
}
