double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 0, weight_sum = 0, density_ratio = 0;

    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][voisinage[index_item]]; 
        h += weights[j][voisinage[index_item]] / capacity[j];
    }
    
    if (weight_sum > capacity[0]) {
        return 0.0;
    }

    density_ratio = weight_sum > 0 ? h / weight_sum : 0;
    
    return 0.6 * density_ratio + 0.4 * (profit[voisinage[index_item]] / (weight_sum + 1));
}
