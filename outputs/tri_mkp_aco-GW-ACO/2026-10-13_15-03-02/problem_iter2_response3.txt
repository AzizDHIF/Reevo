double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double sum_weight_ratio = 0.0;
    
    for(int j = 0; j < dimension; j++){
        sum_weight_ratio += weights[j][voisinage[index_item]] / capacity[j];
    }

    double promising_score = profit[voisinage[index_item]] / sum_weight_ratio;

    if (promising_score > 0.3) {
        return promising_score;
    } else {
        return 0.0;
    }
}
