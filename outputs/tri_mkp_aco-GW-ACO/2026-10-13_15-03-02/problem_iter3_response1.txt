double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double sum_weights = 0.0;
    double sum_ratios = 0.0;

    for (int j = 0; j < dimension; j++) {
        if (weights[j][voisinage[index_item]] <= capacity[j]) {
            sum_weights += weights[j][voisinage[index_item]];
            sum_ratios += weights[j][voisinage[index_item]] / capacity[j];
        }
    }

    if (sum_weights <= 0.0 || sum_ratios <= 0.0) {
        return 0.0;
    }

    return profit[voisinage[index_item]] / sum_ratios;
}
