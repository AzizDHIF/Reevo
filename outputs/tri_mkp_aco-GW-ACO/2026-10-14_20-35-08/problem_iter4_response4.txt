double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double sum_weighted_weights = 0;

    for (int j = 0; j < dimension; j++) {
        sum_weighted_weights += weights[j][voisinage[index_item]] / capacity[j];
    }

    return (sum_weighted_weights > 0) ? profit[voisinage[index_item]] / sum_weighted_weights : 0;
}
