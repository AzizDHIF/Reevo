double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double h=0, sum_weights = 0, sum_capacity = 0;
    for(int j=0; j<dimension; j++){	
    	sum_weights += weights[j][voisinage[index_item]];
    	sum_capacity += capacity[j];
    }
    
    if(sum_capacity == 0) {
    	return 0.0; // Capacity completely filled, no more items can be added
    }
    
    for(int j=0; j<dimension; j++){
        if (capacity[j]!=0) {
            h += weights[j][voisinage[index_item]] / (double)capacity[j];
        }
    }
    
    return profit[voisinage[index_item]] / (1 + h + sum_weights/sum_capacity);
}
