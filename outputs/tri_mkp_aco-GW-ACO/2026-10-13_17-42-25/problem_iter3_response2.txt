double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit)
{
    double h=0, sum_weights = 0, sum_capacity = 0, prom;

    sum_weights = weights[0][voisinage[index_item]] + weights[1][voisinage[index_item]]; // Only considering first two dimensions

    sum_capacity = capacity[0] + capacity[1]; // Considering first two capacity dimensions

    prom = (sum_capacity > 0) ? sum_weights / sum_capacity : 0; // Calculate promise based on ratio of weights to capacities

    return profit[voisinage[index_item]] / prom;
}
