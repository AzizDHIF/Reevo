double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double h=0;
    double density=0;
    for(int j=0; j<dimension; j++){
        h += weights[j][voisinage[index_item]]/capacity[j]; 
        density += weights[j][index_item]; 
    }
    
    double prom_factor = 0.5 * profit[voisinage[index_item]] * density;  // Combined factor
    double heuristic_value =  prom_factor / ((h > 0) ? h : 1e-6); // Avoid division by zero
    
    if(density < epsilon || prom_factor < beta ){ // sparsifying
        return 0;
    }
    
    return heuristic_value;
}
