double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
	double h=0;
	double weight_diff = 0;
	double benefit_ratio = 0;
	
	for(int j=0; j<dimension; j++){
		weight_diff += weights[j][index_item] - weights[j][voisinage[index_item]];
		h += weights[j][voisinage[index_item]] / capacity[j]; 
	}
	
	if(weight_diff <= 0){
		return 0;
	}
	
	benefit_ratio = (profit[voisinage[index_item]] - profit[index_item]) / weight_diff;
	
	return benefit_ratio / h;
}
