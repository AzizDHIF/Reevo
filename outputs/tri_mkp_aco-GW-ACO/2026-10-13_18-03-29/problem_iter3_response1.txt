double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double total_weight_ratio = 0.0;
    double profit_density = 0.0;

    for (int j = 0; j < dimension; j++) {
        total_weight_ratio += weights[j][voisinage[index_item]] / capacity[j];
    }

    profit_density = profit[voisinage[index_item]] / (1 + total_weight_ratio); // Add 1 to avoid division by zero

    // a simple combination of weight ratio and profit density with adjusted weighting factor
    return 0.7 * total_weight_ratio + 0.3 * profit_density;
}
