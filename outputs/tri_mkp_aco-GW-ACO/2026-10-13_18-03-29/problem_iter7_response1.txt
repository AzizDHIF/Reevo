    double profit_density = 0.0;

    for (int j = 0; j < dimension; j++) {

    }

    total_weight_ratio = 0.8 * total_weight_ratio + 0.2;

    profit_density = profit[voisinage[index_item]] / (1 + total_weight_ratio);
