double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weight_ratio = 0.0;
    double profit_density = 0.0;
    
    for (int j = 0; j < dimension; j++) {
        weight_ratio += weights[j][voisinage[index_item]] / capacity[j];
    }
    
    profit_density = profit[voisinage[index_item]] / (1.0 + weight_ratio);
    
    return 0.6 * weight_ratio + 0.4 * profit_density;
}
