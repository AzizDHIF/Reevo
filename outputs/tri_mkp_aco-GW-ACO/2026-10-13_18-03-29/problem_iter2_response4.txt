double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 0;
    double g = 0;

    double weight_sum = 0;
    for (int i = 0; i < dimension; i++) {
        weight_sum += weights[i][voisinage[index_item]];
    }

    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j];
        g += weights[j][voisinage[index_item]] / weight_sum;
    }

    double heuristic_value = (profit[voisinage[index_item]] / h) * g;

    // Sparsifying the heuristic by setting unpromising elements to zero
    if (h == 0) {
        heuristic_value = 0;
    }

    return heuristic_value;
}
