double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profits){
    double h=0, remaining_capacity=0, prod_weights=1;
    
    for(int j=0; j<dimension; j++){
        h += weights[j][voisinage[index_item]] / capacity[j]; 
        remaining_capacity += capacity[j];
        prod_weights *= weights[j][index_item];
    }
    
    double promising_factor = remaining_capacity == 0 ? 0 : prod_weights / remaining_capacity;
    
    return profits[voisinage[index_item]] * promising_factor / h;
}
