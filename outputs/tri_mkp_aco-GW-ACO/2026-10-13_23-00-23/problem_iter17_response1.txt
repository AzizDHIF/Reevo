double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double total_weight = 0.0;
    double sum_weights_by_capacity = 0.0;
    
    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][voisinage[index_item]];
        sum_weights_by_capacity += weights[j][voisinage[index_item]] / capacity[j];
    }
    
    if (sum_weights_by_capacity == 0.0) {
        return 0.0;
    }
    
    return profit[voisinage[index_item]] / (total_weight * sum_weights_by_capacity);
}
