double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 0.0;
    double weighted_capacity = 0.0;
    
    for (int j = 0; j < dimension; j++) {
        if (weights[j][voisinage[index_item]] > 0) {
            h += (1.0 / weights[j][voisinage[index_item]]) * (1.0 / capacity[j]);
            weighted_capacity += 1.0 / capacity[j];
        }
    }
    
    if (weighted_capacity > 0) {
        return profit[voisinage[index_item]] / (h * weighted_capacity);
    } else {
        return 0.0;
    }
}
