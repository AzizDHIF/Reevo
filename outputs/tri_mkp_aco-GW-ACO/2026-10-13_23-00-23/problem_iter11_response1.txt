double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 0;
    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j];
    }
    
    // Adding a balancing factor to avoid weight dominance
    double balance_factor = 0.5;
    
    // Combining various factors to determine item selection promise
    double combined_factor = profit[voisinage[index_item]] / (h + balance_factor);

    return combined_factor;
}
