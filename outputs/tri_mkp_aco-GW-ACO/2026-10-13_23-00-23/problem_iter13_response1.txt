double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 0;
    double weighted_capacity = 0;

    for (int j = 0; j < dimension; j++) {
        weighted_capacity += (capacity[j] == 0) ? 0 : weights[j][voisinage[index_item]] / capacity[j];
    }

    return (weighted_capacity == 0) ? profit[voisinage[index_item]] * NBITEMS : profit[voisinage[index_item]] / weighted_capacity;
}
