double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weighted_sum = 0;

    for (int j = 0; j < dimension; j++) {
        weighted_sum += weights[j][voisinage[index_item]] / capacity[j];
    }

    return weighted_sum == 0 ? 0 : profit[voisinage[index_item]] / weighted_sum;
}
