    double w_factor = 0.6;
    double p_factor = 0.4;

    for(int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] * w_factor / capacity[j];
    }
    
    // Calculate heuristic value based on combined factors
    double h_value = profit[voisinage[index_item]] * p_factor / h;
