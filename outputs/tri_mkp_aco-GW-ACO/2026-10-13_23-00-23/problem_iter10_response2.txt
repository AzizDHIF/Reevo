double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double cumulative_weight = 0.0;
    
    for(int j = 0; j < dimension; j++) {
        cumulative_weight += weights[j][voisinage[index_item]] / capacity[j];
    }
    
    return profit[voisinage[index_item]] / cumulative_weight;
}
