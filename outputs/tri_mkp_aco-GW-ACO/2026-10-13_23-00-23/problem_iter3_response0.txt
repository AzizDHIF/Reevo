double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 1e-6; // Smoothing factor to avoid division by zero
    int promising_factors = 0;

    for (int j = 0; j < dimension; j++) {
        if (weights[j][voisinage[index_item]] > 0) {
            h += weights[j][voisinage[index_item]] / capacity[j];
            promising_factors++;
        }
    }

    if (promising_factors == 0) {
        return 0; // Return 0 if no promising factors found
    }

    return profit[voisinage[index_item]] / (h * promising_factors);
}
