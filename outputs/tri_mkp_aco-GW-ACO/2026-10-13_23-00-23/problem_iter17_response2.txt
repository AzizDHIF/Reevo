double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weighted_factor = 0;
    
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] != 0) {
            weighted_factor += (weights[j][voisinage[index_item]] / capacity[j]);
        }
    }
    
    return profit[voisinage[index_item]] / weighted_factor;
}
