import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Improved heuristics function for Solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.
    
    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).
    
    Returns:
    np.ndarray: Heuristics values, shape (n,).
    """
    
    # Normalize the weights to balance efficiency and variance
    normalized_weight = weight / np.sum(weight, axis=0, keepdims=True)
    
    # Calculate the total normalized weight of each item
    total_normalized_weight = np.sum(normalized_weight, axis=1)
    
    # Calculate the ratio of prize to total normalized weight for each item
    prize_to_total_normalized_weight_ratio = prize / (total_normalized_weight + 1e-6)
    
    # Calculate the mean normalized weight of each item
    mean_normalized_weight = np.mean(normalized_weight, axis=1)
    
    # Calculate the ratio of prize to mean normalized weight for each item
    prize_to_mean_normalized_weight_ratio = prize / (mean_normalized_weight + 1e-6)
    
    # Calculate the variance of normalized weights for each item
    normalized_weight_variance = np.var(normalized_weight, axis=1)
    
    # Calculate the ratio of prize to normalized weight variance for each item
    prize_to_normalized_weight_variance_ratio = prize / (normalized_weight_variance + 1e-6)
    
    # Combine the ratios to determine the heuristics values
    heuristics = (prize_to_total_normalized_weight_ratio + prize_to_mean_normalized_weight_ratio +
                  prize_to_normalized_weight_variance_ratio) / 3
    
    # Consider the constraint by scaling the heuristics
    heuristics = heuristics * (1 - total_normalized_weight)
    
    # Consider the efficiency by scaling the heuristics
    efficiency = prize / (np.sum(normalized_weight ** 2, axis=1) + 1e-6)
    heuristics = heuristics * efficiency
    
    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 25)
    heuristics[heuristics < threshold] = 0
    
    return heuristics
