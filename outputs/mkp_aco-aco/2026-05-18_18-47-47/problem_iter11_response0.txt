import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Improved heuristics function for Solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.

    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).

    Returns:
    np.ndarray: Heuristics values, shape (n,).
    """

    # Calculate the total weight of each item
    total_weight = np.sum(weight, axis=1)

    # Calculate the efficiency of each item using L1 norm for better interpretability
    efficiency = prize / (np.sum(np.abs(weight), axis=1) + 1e-6)

    # Calculate the ratio of prize to total weight for each item, considering the dimension of weights
    prize_to_total_weight_ratio = prize / (np.max(weight, axis=1) + 1e-6)

    # Calculate the normalized prize-to-weight ratio
    normalized_prize_to_total_weight_ratio = prize_to_total_weight_ratio / np.max(prize_to_total_weight_ratio)

    # Calculate the normalized efficiency
    normalized_efficiency = efficiency / np.max(efficiency)

    # Combine the ratios to determine the heuristics values, prioritizing normalized prize-to-weight ratio and normalized efficiency
    heuristics = 0.6 * normalized_prize_to_total_weight_ratio + 0.4 * normalized_efficiency

    # Consider the constraint by scaling the heuristics
    constraint_factor = 1 - np.max(weight, axis=1)
    heuristics = heuristics * constraint_factor

    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 10)
    heuristics = np.where(heuristics < threshold, 0, heuristics)

    return heuristics
