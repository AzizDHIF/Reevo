import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Improved heuristics function for Solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.

    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).

    Returns:
    np.ndarray: Heuristics values, shape (n,).
    """

    # Calculate the total weight of each item
    total_weight = np.sum(weight, axis=1)

    # Calculate the efficiency of each item using L1, L2, and L-infinity norms for better robustness
    efficiency_l1 = prize / (np.sum(np.abs(weight), axis=1) + 1e-6)
    efficiency_l2 = prize / (np.sqrt(np.sum(np.square(weight), axis=1)) + 1e-6)
    efficiency_linf = prize / (np.max(np.abs(weight), axis=1) + 1e-6)
    efficiency = (efficiency_l1 + efficiency_l2 + efficiency_linf) / 3

    # Calculate the ratio of prize to total weight for each item, considering the dimension of weights
    prize_to_total_weight_ratio = prize / (np.max(weight, axis=1) + 1e-6)

    # Calculate the constraint satisfaction for each item
    constraint_satisfaction = 1 - np.max(weight, axis=1)

    # Calculate the penalty for each item based on the number of constraints it violates
    penalty = np.sum(weight > 1, axis=1)

    # Combine the ratios to determine the heuristics values, prioritizing efficiency, prize-to-weight ratio, and weight constraints
    heuristics = (efficiency * prize_to_total_weight_ratio * constraint_satisfaction) / (1 + penalty + np.sum(np.abs(weight), axis=1))

    # Normalize the heuristics values for better comparison
    heuristics = heuristics / (np.sum(heuristics) + 1e-6)

    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 20)
    heuristics = np.where(heuristics < threshold, 0, heuristics)

    return heuristics
