import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Improved heuristics function for Solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.
    
    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).
    
    Returns:
    np.ndarray: Heuristics values, shape (n,).
    """
    
    # Calculate the total weight of each item
    total_weight = np.sum(weight, axis=1)
    
    # Calculate the ratio of prize to total weight for each item
    prize_to_total_weight_ratio = prize / (total_weight + 1e-6)  # Add a small value for numerical stability
    
    # Calculate the maximum weight of each item
    max_weight = np.max(weight, axis=1)
    
    # Calculate the ratio of prize to maximum weight for each item
    prize_to_max_weight_ratio = prize / (max_weight + 1e-6)  # Add a small value for numerical stability
    
    # Calculate the variance of weights for each item
    weight_variance = np.var(weight, axis=1)
    
    # Calculate the ratio of prize to weight variance for each item
    prize_to_weight_variance_ratio = prize / (weight_variance + 1e-6)  # Add a small value for numerical stability
    
    # Calculate the mean weight of each item
    mean_weight = np.mean(weight, axis=1)
    
    # Calculate the ratio of prize to mean weight for each item
    prize_to_mean_weight_ratio = prize / (mean_weight + 1e-6)  # Add a small value for numerical stability
    
    # Calculate the median weight of each item
    median_weight = np.median(weight, axis=1)
    
    # Calculate the ratio of prize to median weight for each item
    prize_to_median_weight_ratio = prize / (median_weight + 1e-6)  # Add a small value for numerical stability
    
    # Combine the ratios to determine the heuristics values
    heuristics = (0.2 * prize_to_total_weight_ratio + 0.2 * prize_to_max_weight_ratio + 
                  0.1 * prize_to_weight_variance_ratio + 0.2 * prize_to_mean_weight_ratio + 
                  0.3 * prize_to_median_weight_ratio)
    
    # Consider the constraint by scaling the heuristics
    heuristics = heuristics * (1 - total_weight)
    
    # Consider the efficiency by scaling the heuristics
    efficiency = prize / (np.sum(weight ** 2, axis=1) + 1e-6)  # Use L2 norm for numerical stability
    heuristics = heuristics * efficiency
    
    # Consider the variance by scaling the heuristics
    variance_factor = 1 / (weight_variance + 1e-6)
    heuristics = heuristics * variance_factor
    
    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 10)  # Use the 10th percentile as the threshold
    heuristics[heuristics < threshold] = 0
    
    return heuristics
