import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Improved heuristics function for Solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.

    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).

    Returns:
    np.ndarray: The heuristics values, shape (n,).
    """
    
    # Calculate the total weight of each item
    total_weight = np.sum(weight, axis=1)
    
    # Calculate the prize-to-weight ratio for each item, considering the constraint
    prize_to_weight_ratio = prize / (total_weight + 1e-6)
    
    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(prize_to_weight_ratio, 25)  # Use the 25th percentile as the threshold
    heuristics = np.where(prize_to_weight_ratio < threshold, 0, prize_to_weight_ratio)
    
    return heuristics
