import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Improved heuristics function for Solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.
    
    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).
    
    Returns:
    np.ndarray: Heuristics values, shape (n,).
    """

    # Calculate the total weight of each item
    total_weight = np.sum(weight, axis=1)

    # Calculate the efficiency of each item
    efficiency = prize / (np.sum(weight ** 2, axis=1) + 1e-6)  # Use L2 norm for numerical stability

    # Calculate the ratio of prize to total weight for each item
    prize_to_total_weight_ratio = prize / (total_weight + 1e-6)  # Add a small value for numerical stability

    # Combine the ratios to determine the heuristics values, prioritizing prize-to-weight ratio and efficiency
    heuristics = prize_to_total_weight_ratio * efficiency

    # Consider the constraint by scaling the heuristics
    heuristics = heuristics * (1 - total_weight)

    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 10)  # Use the 10th percentile as the threshold
    heuristics[heuristics < threshold] = 0

    return heuristics
