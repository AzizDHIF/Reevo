import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Heuristics function for Solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.
    
    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).
    
    Returns:
    np.ndarray: Heuristics of shape (n,), where heuristics[i] indicates how promising it is to include item i in the solution.
    """
    
    # Calculate the mean weight of each item
    mean_weight = np.mean(weight, axis=1)
    
    # Calculate the maximum weight of each item
    max_weight = np.max(weight, axis=1)
    
    # Calculate the minimum weight of each item
    min_weight = np.min(weight, axis=1)
    
    # Calculate the variance of weights for each item
    weight_variance = np.var(weight, axis=1)
    
    # Calculate the ratio of prize to mean weight, with a small value added to the denominator for stability
    ratio_mean = prize / (mean_weight + 1e-8)
    
    # Calculate the ratio of prize to maximum weight, with a small value added to the denominator for stability
    ratio_max = prize / (max_weight + 1e-8)
    
    # Calculate the ratio of prize to minimum weight, with a small value added to the denominator for stability
    ratio_min = prize / (min_weight + 1e-8)
    
    # Combine the ratios and variance to determine how promising it is to include an item
    heuristics = (ratio_mean + ratio_max + ratio_min) / 3 * (1 - weight_variance / (1 + weight_variance))
    
    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.mean(heuristics)  # Calculate the threshold as the mean of the heuristics
    heuristics[heuristics < threshold] = 0  # Set the unpromising elements to zero
    
    return heuristics
