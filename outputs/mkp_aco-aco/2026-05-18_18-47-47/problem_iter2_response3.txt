import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Heuristics function for Solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.
    
    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,)
    weight (np.ndarray): The weight of each item, shape (n, m)
    
    Returns:
    np.ndarray: Heuristics of shape (n,), heuristics[i] indicates how promising it is to include item i in the solution
    """
    # Calculate the average weight of each item
    avg_weight = np.mean(weight, axis=1)
    
    # Calculate the efficiency of each item (prize per unit weight)
    efficiency = prize / avg_weight
    
    # Calculate the variance of weights for each item
    weight_variance = np.var(weight, axis=1)
    
    # Calculate the sparsity of weights for each item
    sparsity = np.count_nonzero(weight, axis=1) / weight.shape[1]
    
    # Combine various factors to determine how promising it is to select an item
    heuristics = efficiency * (1 - weight_variance) * (1 - sparsity)
    
    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.mean(heuristics)  # Set the threshold to the mean heuristics value
    heuristics[heuristics < threshold] = 0
    
    return heuristics
