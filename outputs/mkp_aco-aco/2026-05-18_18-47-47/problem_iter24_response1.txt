import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Improved heuristics function for Solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.

    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).

    Returns:
    np.ndarray: Heuristics values, shape (n,).
    """

    # Calculate the total weight of each item using L2 norm for better robustness to outliers
    total_weight = np.linalg.norm(weight, axis=1)

    # Calculate the efficiency of each item using L2 norm for better robustness to outliers
    efficiency = prize / (total_weight + 1e-6)

    # Calculate the ratio of prize to total weight for each item, considering the dimension of weights
    prize_to_max_weight_ratio = prize / (np.max(np.abs(weight), axis=1) + 1e-6)
    prize_to_total_weight_ratio = prize / (total_weight + 1e-6)

    # Calculate the constraint satisfaction for each item, prioritizing items with lower weights
    constraint_satisfaction = 1 - np.max(np.abs(weight), axis=1)

    # Combine the ratios to determine the heuristics values, prioritizing efficiency and constraint satisfaction
    heuristics = efficiency * constraint_satisfaction * (prize_to_max_weight_ratio + prize_to_total_weight_ratio) / (2 + total_weight)

    # Apply a non-linear transformation to amplify the difference between promising and unpromising items
    heuristics = np.power(heuristics, 2)

    # Normalize the heuristics values for better comparison
    heuristics = heuristics / (np.sum(heuristics) + 1e-6)

    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 20)
    heuristics = np.where(heuristics < threshold, 0, heuristics)

    return heuristics
