import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Heuristics function for Solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.
    
    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).
    
    Returns:
    np.ndarray: Heuristics values, shape (n,).
    """
    
    # Calculate the total weight of each item
    total_weight = np.sum(weight, axis=1)
    
    # Calculate the ratio of prize to total weight for each item
    prize_to_total_weight_ratio = prize / total_weight
    
    # Calculate the maximum weight of each item
    max_weight = np.max(weight, axis=1)
    
    # Calculate the ratio of prize to maximum weight for each item
    prize_to_max_weight_ratio = prize / max_weight
    
    # Combine the ratios to determine the heuristics values
    heuristics = (prize_to_total_weight_ratio + prize_to_max_weight_ratio) / 2
    
    # Consider the constraint by scaling the heuristics
    heuristics = heuristics * (1 - total_weight)
    
    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 25)  # Use the 25th percentile as the threshold
    heuristics[heuristics < threshold] = 0
    
    return heuristics
