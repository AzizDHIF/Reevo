import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Heuristics function for Solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.
    
    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).
    
    Returns:
    np.ndarray: The heuristics values, shape (n,).
    """
    
    # Calculate the prize-to-weight ratio for each item
    total_weight = np.sum(weight, axis=1)
    prize_to_weight_ratio = prize / total_weight
    
    # Calculate the weight constraint violation ratio for each item
    max_weight = np.max(weight, axis=1)
    weight_constraint_violation = max_weight / total_weight
    
    # Calculate the average weight ratio for each item
    average_weight_ratio = np.mean(weight, axis=1) / total_weight
    
    # Combine the factors to determine the heuristics values
    heuristics = prize_to_weight_ratio / (1 + weight_constraint_violation + average_weight_ratio)
    
    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 25)  # Use the 25th percentile as the threshold
    heuristics[heuristics < threshold] = 0
    
    return heuristics
