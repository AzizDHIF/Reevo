import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Improved heuristics function for Solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.
    
    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).
    
    Returns:
    np.ndarray: Heuristics values, shape (n,).
    """

    # Calculate the total weight of each item using L1 norm for stability
    total_weight = np.sum(np.abs(weight), axis=1)

    # Calculate the efficiency of each item using L2 norm for numerical stability
    efficiency = prize / (np.sum(weight ** 2, axis=1) + 1e-6)

    # Calculate the ratio of prize to total weight for each item
    prize_to_total_weight_ratio = prize / (total_weight + 1e-6)  # Add a small value for numerical stability

    # Prioritize efficiency and prize-to-weight ratio with a focus on stability
    heuristics = efficiency * prize_to_total_weight_ratio

    # Consider the constraint by scaling the heuristics with a non-linear transformation
    max_weight = np.max(np.abs(weight), axis=1)
    max_weight_constraint_ratio = np.tanh(max_weight / (total_weight + 1e-6))  # Use tanh for non-linear transformation
    heuristics = heuristics * (1 - max_weight_constraint_ratio)  # Directly consider the max weight constraint

    # Sparsify the heuristics by setting unpromising elements to zero with a dynamic threshold
    threshold = np.percentile(heuristics, 10)  # Use the 10th percentile as the dynamic threshold
    heuristics = np.where(heuristics < threshold, 0, heuristics)

    return heuristics
