import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    This function computes heuristics for solving Multiple Knapsack Problems (MKP)
    using stochastic solution sampling. It considers various factors such as 
    prize, weight, and sparsity to determine how promising it is to include 
    each item in the solution.

    Parameters:
    prize (np.ndarray): A 1D array of shape (n,) representing the prize of each item.
    weight (np.ndarray): A 2D array of shape (n, m) representing the weights of each item.

    Returns:
    heuristics (np.ndarray): A 1D array of shape (n,) where each element represents 
    the heuristics value for the corresponding item.
    """

    # Calculate the prize-to-weight ratio for each item
    prize_to_weight_ratio = prize / np.sum(weight, axis=1)

    # Calculate the average weight of each item
    average_weight = np.mean(weight, axis=1)

    # Calculate the sparsity of the weights for each item
    sparsity = np.count_nonzero(weight, axis=1) / weight.shape[1]

    # Combine the factors to determine the heuristics
    heuristics = prize_to_weight_ratio * (1 - average_weight) * (1 - sparsity)

    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 25)
    heuristics[heuristics < threshold] = 0

    return heuristics
