import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Improved heuristics function for Solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.

    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).

    Returns:
    np.ndarray: Heuristics values, shape (n,).
    """

    # Calculate the total weight of each item
    total_weight = np.sum(np.abs(weight), axis=1)

    # Calculate the efficiency of each item
    efficiency = prize / (total_weight + 1e-6)

    # Calculate the prize-to-weight ratio for each item
    prize_to_weight_ratio = prize / (np.max(np.abs(weight), axis=1) + 1e-6)

    # Calculate the constraint satisfaction for each item
    constraint_satisfaction = 1 - np.max(np.abs(weight), axis=1)

    # Combine the factors to determine the heuristics values
    heuristics = efficiency * prize_to_weight_ratio * constraint_satisfaction

    # Apply a non-linear transformation to amplify the differences between items
    heuristics = np.tanh(heuristics)

    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 25)
    heuristics = np.where(heuristics < threshold, 0, heuristics)

    return heuristics
