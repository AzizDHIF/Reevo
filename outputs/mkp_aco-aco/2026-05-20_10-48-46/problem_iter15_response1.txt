import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    This heuristics function calculates the promise of each item based on prize, 
    weight, and the ratio of prize to weight for each dimension, with added diversity and simplification considerations.

    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).

    Returns:
    np.ndarray: The heuristics of each item, shape (n,).
    """

    # Calculate the total weight of each item
    total_weight = np.sum(weight, axis=1)

    # Calculate the average weight of each item for each dimension
    avg_weight = np.mean(weight, axis=1)

    # Calculate the dimension diversity of each item
    dimension_diversity = np.std(weight, axis=1)

    # Calculate the prize-to-weight ratio for each item
    prize_to_weight_ratio = prize / (total_weight + 1e-8)  # avoid division by zero

    # Calculate the prize-to-average-weight ratio for each item
    prize_to_avg_weight_ratio = prize / (avg_weight + 1e-8)  # avoid division by zero

    # Calculate the prize-to-dimension-diversity ratio for each item
    prize_to_diversity_ratio = prize / (dimension_diversity + 1e-8)  # avoid division by zero

    # Combine the factors using a weighted average
    heuristics = (
        0.3 * prize_to_weight_ratio + 
        0.25 * prize_to_avg_weight_ratio + 
        0.2 * prize_to_diversity_ratio + 
        0.25 * prize / (np.max(weight, axis=1) + 1e-8)  # consider maximum weight in each dimension
    )

    # Apply a soft threshold to emphasize top items
    threshold = np.percentile(heuristics, 30)
    heuristics = np.where(heuristics < threshold, heuristics * 0.5, heuristics)

    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 20)
    heuristics = np.where(heuristics < threshold, 0, heuristics)

    return heuristics
