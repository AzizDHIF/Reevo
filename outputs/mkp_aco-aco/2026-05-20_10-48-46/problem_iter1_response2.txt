```python
import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    # Calculate the promise of each item based on prize-to-weight ratio
    promise = prize / np.sum(weight, axis=1)
    
    # Introduce a sparsity factor to filter out unpromising items
    sparsity_factor = 0.5
    threshold = np.percentile(promise, 100 * (1 - sparsity_factor))
    
    # Set unpromising items to zero
    promise[promise < threshold] = 0
    
    # Consider the diversity of weights for each item
    weight_diversity = np.sum(weight > 0.5, axis=1)
    weight_diversity_factor = 0.2
    promise += weight_diversity_factor * weight_diversity
    
    # Introduce a factor to promote items with higher prizes
    prize_factor = 0.8
    promise += prize_factor * prize / np.max(prize)
    
    return promise
```
