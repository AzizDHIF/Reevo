import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    This heuristics function calculates the promise of each item based on prize, 
    weight, and the ratio of prize to weight for each dimension.

    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).

    Returns:
    np.ndarray: The heuristics of each item, shape (n,).
    """

    # Calculate the total weight of each item
    total_weight = np.sum(weight, axis=1)

    # Calculate the average weight of each item for each dimension
    avg_weight = np.mean(weight, axis=1)

    # Calculate the dimension diversity of each item
    dimension_diversity = np.std(weight, axis=1)

    # Calculate the prize-to-weight ratio for each item
    prize_to_weight_ratio = prize / (total_weight + 1e-6)  # Add a small value for stability

    # Calculate the prize-to-average-weight ratio for each item
    prize_to_avg_weight_ratio = prize / (avg_weight + 1e-6)  # Add a small value for stability

    # Calculate the prize-to-dimension-diversity ratio for each item
    prize_to_diversity_ratio = prize / (dimension_diversity + 1e-6)  # Add a small value for stability

    # Calculate the normalized prize
    normalized_prize = prize / np.max(prize)

    # Combine the factors using a weighted average with adaptive weighting
    weights = np.array([0.3, 0.25, 0.2, 0.25])  # Adaptive weights
    weights = weights / np.sum(weights)  # Normalize the weights
    heuristics = weights[0] * prize_to_weight_ratio + weights[1] * prize_to_avg_weight_ratio + weights[2] * prize_to_diversity_ratio + weights[3] * normalized_prize

    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 15)  # Decrease the threshold for more aggressive sparsification
    heuristics[heuristics < threshold] = 0

    return heuristics
