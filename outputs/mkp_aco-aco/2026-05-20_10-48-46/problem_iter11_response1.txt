```python
import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    This heuristics function calculates the promise of each item based on prize, 
    weight, and the ratio of prize to weight for each dimension, with adaptive weighting 
    and diversity consideration.

    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).

    Returns:
    np.ndarray: The heuristics of each item, shape (n,).
    """

    # Calculate the total weight of each item
    total_weight = np.sum(weight, axis=1)

    # Calculate the average weight of each item for each dimension
    avg_weight = np.mean(weight, axis=1)

    # Calculate the dimension diversity of each item
    dimension_diversity = np.std(weight, axis=1)

    # Calculate the prize-to-weight ratio for each item
    prize_to_weight_ratio = prize / total_weight

    # Calculate the prize-to-average-weight ratio for each item
    prize_to_avg_weight_ratio = prize / avg_weight

    # Calculate the prize-to-dimension-diversity ratio for each item
    prize_to_diversity_ratio = prize / dimension_diversity

    # Calculate the dimension-specific prize-to-weight ratios
    dim_specific_ratios = prize[:, np.newaxis] / weight

    # Calculate the average dimension-specific prize-to-weight ratio for each item
    avg_dim_specific_ratio = np.mean(dim_specific_ratios, axis=1)

    # Combine the factors using a weighted average with adaptive weighting
    weight_ratios = 0.3 * prize_to_weight_ratio
    avg_weight_ratios = 0.2 * prize_to_avg_weight_ratio
    diversity_ratios = 0.2 * prize_to_diversity_ratio
    dim_specific_ratios = 0.3 * avg_dim_specific_ratio

    # Add a direct prize factor with a smaller weight
    direct_prize_factor = 0.1 * prize

    heuristics = weight_ratios + avg_weight_ratios + diversity_ratios + dim_specific_ratios + direct_prize_factor

    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 15)  # Adjusted threshold to 15th percentile
    heuristics[heuristics < threshold] = 0

    return heuristics
```
