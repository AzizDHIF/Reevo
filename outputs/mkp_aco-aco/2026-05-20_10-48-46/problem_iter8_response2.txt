import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    This heuristics function calculates the promise of each item based on prize, 
    weight, and the ratio of prize to weight for each dimension.

    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).

    Returns:
    np.ndarray: The heuristics of each item, shape (n,).
    """

    # Calculate the total weight of each item
    total_weight = np.sum(weight, axis=1)

    # Calculate the average weight of each item for each dimension
    avg_weight = np.mean(weight, axis=1)

    # Calculate the dimension diversity of each item
    dimension_diversity = np.std(weight, axis=1)

    # Calculate the coefficient of variation of weight for each item
    coefficient_variation = np.std(weight, axis=1) / np.mean(weight, axis=1)

    # Calculate the prize-to-weight ratio for each item
    prize_to_weight_ratio = prize / total_weight

    # Calculate the prize-to-average-weight ratio for each item
    prize_to_avg_weight_ratio = prize / avg_weight

    # Calculate the prize-to-dimension-diversity ratio for each item
    prize_to_diversity_ratio = prize / dimension_diversity

    # Calculate the prize-to-coefficient-variation ratio for each item
    prize_to_variation_ratio = prize / coefficient_variation

    # Use adaptive weighting based on the relative importance of each factor
    weight_ratio_importance = np.max(prize_to_weight_ratio) / (np.max(prize_to_weight_ratio) + np.max(prize_to_avg_weight_ratio) + np.max(prize_to_diversity_ratio) + np.max(prize_to_variation_ratio))
    avg_weight_ratio_importance = np.max(prize_to_avg_weight_ratio) / (np.max(prize_to_weight_ratio) + np.max(prize_to_avg_weight_ratio) + np.max(prize_to_diversity_ratio) + np.max(prize_to_variation_ratio))
    diversity_ratio_importance = np.max(prize_to_diversity_ratio) / (np.max(prize_to_weight_ratio) + np.max(prize_to_avg_weight_ratio) + np.max(prize_to_diversity_ratio) + np.max(prize_to_variation_ratio))
    variation_ratio_importance = np.max(prize_to_variation_ratio) / (np.max(prize_to_weight_ratio) + np.max(prize_to_avg_weight_ratio) + np.max(prize_to_diversity_ratio) + np.max(prize_to_variation_ratio))
    prize_importance = 0.2  # fixed importance for prize

    # Combine the factors using a weighted average
    heuristics = weight_ratio_importance * prize_to_weight_ratio + avg_weight_ratio_importance * prize_to_avg_weight_ratio + diversity_ratio_importance * prize_to_diversity_ratio + variation_ratio_importance * prize_to_variation_ratio + prize_importance * prize

    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 20)
    heuristics[heuristics < threshold] = 0

    return heuristics
