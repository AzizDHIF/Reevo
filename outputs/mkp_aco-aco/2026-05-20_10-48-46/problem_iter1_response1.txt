```python
import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    This function calculates a heuristic value for each item in the multiple knapsack problem.
    The heuristic values represent how promising it is to include each item in the solution.
    
    Parameters:
    prize (np.ndarray): The prizes of the items, shape (n,).
    weight (np.ndarray): The weights of the items, shape (n, m).
    
    Returns:
    np.ndarray: Heuristic values for each item, shape (n,).
    """
    
    # Calculate the average weight of each item across all dimensions
    avg_weight = np.mean(weight, axis=1)
    
    # Calculate the prize-to-average-weight ratio for each item
    prize_to_weight_ratio = prize / avg_weight
    
    # Calculate the maximum weight in each dimension
    max_weight = np.max(weight, axis=0)
    
    # Calculate the weight-to-max-weight ratio for each item in each dimension
    weight_to_max_weight_ratio = weight / max_weight
    
    # Calculate the average weight-to-max-weight ratio for each item across all dimensions
    avg_weight_to_max_weight_ratio = np.mean(weight_to_max_weight_ratio, axis=1)
    
    # Calculate the heuristic value for each item based on the prize-to-average-weight ratio and the average weight-to-max-weight ratio
    heuristic = prize_to_weight_ratio / (1 + avg_weight_to_max_weight_ratio)
    
    # Sparsify the heuristic by setting unpromising elements to zero (items with low prize-to-weight ratio and high average weight-to-max-weight ratio)
    sparse_heuristic = np.where((prize_to_weight_ratio > np.mean(prize_to_weight_ratio)) & (avg_weight_to_max_weight_ratio < np.mean(avg_weight_to_max_weight_ratio)), heuristic, 0)
    
    return sparse_heuristic
```
