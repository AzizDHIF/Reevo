import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    This heuristics function calculates the promise of each item based on prize, 
    weight, and dimension-specific prize-to-weight ratios.

    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).

    Returns:
    np.ndarray: The heuristics of each item, shape (n,).
    """

    # Calculate the dimension-specific prize-to-weight ratios for each item
    prize_to_weight_ratios = prize[:, np.newaxis] / weight

    # Calculate the average prize-to-weight ratio for each item
    avg_prize_to_weight_ratio = np.mean(prize_to_weight_ratios, axis=1)

    # Calculate the standard deviation of prize-to-weight ratios for each item
    std_prize_to_weight_ratio = np.std(prize_to_weight_ratios, axis=1)

    # Calculate the total weight of each item
    total_weight = np.sum(weight, axis=1)

    # Calculate the prize-to-total-weight ratio for each item
    prize_to_total_weight_ratio = prize / total_weight

    # Combine the factors using a weighted average
    heuristics = 0.4 * avg_prize_to_weight_ratio + 0.3 * prize_to_total_weight_ratio + 0.2 * std_prize_to_weight_ratio + 0.1 * prize

    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 20)
    heuristics = np.where(heuristics >= threshold, heuristics, 0)

    return heuristics
