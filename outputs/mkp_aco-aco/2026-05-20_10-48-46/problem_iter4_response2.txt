import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    This heuristics function calculates the promise of each item based on prize, 
    weight, and the ratio of prize to weight for each dimension, emphasizing prize and weight diversity.

    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).

    Returns:
    np.ndarray: The heuristics of each item, shape (n,).
    """

    # Calculate the total weight of each item
    total_weight = np.sum(weight, axis=1)

    # Calculate the average weight of each item for each dimension
    avg_weight = np.mean(weight, axis=1)

    # Calculate the prize-to-weight ratio for each item
    prize_to_weight_ratio = prize / total_weight

    # Calculate the prize-to-average-weight ratio for each item
    prize_to_avg_weight_ratio = prize / avg_weight

    # Calculate the diversity of weights for each item
    weight_diversity = np.sum(weight > 0.5, axis=1)

    # Calculate the diversity of prizes for each item
    prize_diversity = np.abs(prize - np.mean(prize)) / np.std(prize)

    # Combine the factors using a weighted average, emphasizing prize and weight diversity
    heuristics = 0.2 * prize_to_weight_ratio + 0.2 * prize_to_avg_weight_ratio + 0.3 * prize_diversity + 0.3 * weight_diversity

    # Introduce a factor to promote items with higher prizes and more diverse weights
    prize_factor = 0.8
    weight_factor = 0.6
    heuristics += prize_factor * prize / np.max(prize) + weight_factor * weight_diversity / np.max(weight_diversity)

    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 25)
    heuristics[heuristics < threshold] = 0

    return heuristics
