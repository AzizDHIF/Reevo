import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    This heuristics function calculates the promise of each item based on prize, 
    weight, and the ratio of prize to weight for each dimension, with enhanced diversity, 
    sparsity, and dimension-wise considerations.

    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).

    Returns:
    np.ndarray: The heuristics of each item, shape (n,).
    """

    # Calculate the total weight of each item
    total_weight = np.sum(weight, axis=1)

    # Calculate the average weight of each item for each dimension
    avg_weight = np.mean(weight, axis=1)

    # Calculate the dimension diversity of each item
    dimension_diversity = np.std(weight, axis=1)

    # Calculate the prize-to-weight ratio for each item
    prize_to_weight_ratio = prize / (total_weight + 1e-6)  # Avoid division by zero

    # Calculate the prize-to-average-weight ratio for each item
    prize_to_avg_weight_ratio = prize / (avg_weight + 1e-6)  # Avoid division by zero

    # Calculate the prize-to-dimension-diversity ratio for each item
    prize_to_diversity_ratio = prize / (dimension_diversity + 1e-6)  # Avoid division by zero

    # Calculate the dimension-wise prize-to-weight ratios
    dimension_wise_ratios = prize[:, None] / (weight + 1e-6)  # Avoid division by zero

    # Calculate the average dimension-wise prize-to-weight ratio
    avg_dimension_wise_ratio = np.mean(dimension_wise_ratios, axis=1)

    # Calculate the minimum dimension-wise prize-to-weight ratio
    min_dimension_wise_ratio = np.min(dimension_wise_ratios, axis=1)

    # Combine the factors using a weighted average
    heuristics = (
        0.25 * prize_to_weight_ratio + 
        0.2 * prize_to_avg_weight_ratio + 
        0.2 * prize_to_diversity_ratio + 
        0.15 * avg_dimension_wise_ratio + 
        0.1 * min_dimension_wise_ratio + 
        0.1 * prize
    )

    # Apply a bonus to items with high dimension diversity
    diversity_bonus = 0.1 * (dimension_diversity / np.max(dimension_diversity))
    heuristics += diversity_bonus

    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 10)
    heuristics[heuristics < threshold] = 0

    return heuristics
