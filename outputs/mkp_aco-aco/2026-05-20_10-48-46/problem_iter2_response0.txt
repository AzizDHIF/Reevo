import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    This function calculates an improved heuristic value for each item in the multiple knapsack problem.
    
    Parameters:
    prize (np.ndarray): The prizes of the items, shape (n,).
    weight (np.ndarray): The weights of the items, shape (n, m).
    
    Returns:
    np.ndarray: Heuristic values for each item, shape (n,).
    """
    
    # Calculate the promise of each item based on simplified prize-to-weight ratio
    promise = prize / np.sum(weight, axis=1)
    
    # Introduce a sparsity factor to filter out unpromising items
    sparsity_factor = 0.5
    threshold = np.percentile(promise, 100 * (1 - sparsity_factor))
    
    # Set unpromising items to zero and add diversity factor
    weight_diversity = np.sum(weight > 0.5, axis=1)
    weight_diversity_factor = 0.2
    sparse_promise = np.where(promise >= threshold, promise + weight_diversity_factor * weight_diversity, 0)
    
    # Prioritize high-reward items with a factor
    prize_factor = 0.8
    prioritized_promise = sparse_promise + prize_factor * prize / np.max(prize)
    
    return prioritized_promise
