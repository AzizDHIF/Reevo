import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Heuristics function to determine how promising it is to include each item in the solution.
    
    The heuristics function combines the prize of each item, the weight of each item in each dimension,
    and the sparsity of the solution to determine how promising it is to include each item.
    
    Parameters:
    prize (np.ndarray): The prize of each item. Shape: (n,)
    weight (np.ndarray): The weight of each item in each dimension. Shape: (n, m)
    
    Returns:
    np.ndarray: The heuristics value for each item. Shape: (n,)
    """
    
    # Calculate the average weight of each item across all dimensions
    avg_weight = np.mean(weight, axis=1)
    
    # Calculate the heuristics value as the prize divided by the average weight
    heuristics = prize / avg_weight
    
    # Calculate the density of each item as the prize divided by the sum of weights in all dimensions
    density = prize / np.sum(weight, axis=1)
    
    # Calculate the variance of weights for each item
    weight_var = np.var(weight, axis=1)
    
    # Combine the heuristics value, density, and variance to determine the final heuristics value
    heuristics = heuristics * density * (1 - weight_var)
    
    # Sparsify the heuristics by setting unpromising elements to zero
    sparsity_threshold = np.percentile(heuristics, 25)
    heuristics[heuristics < sparsity_threshold] = 0
    
    return heuristics
