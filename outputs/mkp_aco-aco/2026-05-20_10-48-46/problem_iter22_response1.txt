import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    This heuristics function calculates the promise of each item based on prize, 
    weight, and the ratio of prize to weight for each dimension, with added diversity and simplification considerations.

    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).

    Returns:
    np.ndarray: The heuristics of each item, shape (n,).
    """

    # Calculate the total weight of each item
    total_weight = np.sum(weight, axis=1)

    # Calculate the prize-to-weight ratio for each item
    prize_to_weight_ratio = prize / (total_weight + 1e-8)  # avoid division by zero

    # Calculate the prize-to-max-weight ratio for each item
    prize_to_max_weight_ratio = prize / (np.max(weight, axis=1) + 1e-8)  # consider maximum weight in each dimension

    # Calculate the average prize-to-weight ratio for each dimension
    avg_prize_to_weight_ratio = np.mean(prize[:, np.newaxis] / (weight + 1e-8), axis=1)

    # Calculate the dimension-wise weights to promote diversity
    dim_weights = np.sum(weight, axis=0)  # calculate the sum of weights for each dimension
    dim_weights = dim_weights / np.sum(dim_weights)  # normalize the dimension weights

    # Calculate the dimension-wise prize-to-weight ratio
    dim_prize_to_weight_ratio = prize / (np.sum(weight * dim_weights, axis=1) + 1e-8)

    # Combine the factors using a weighted average
    heuristics = (
        0.25 * prize_to_weight_ratio + 
        0.25 * prize_to_max_weight_ratio + 
        0.25 * avg_prize_to_weight_ratio + 
        0.25 * dim_prize_to_weight_ratio
    )

    # Apply a soft threshold to emphasize top items
    threshold = np.percentile(heuristics, 30)
    heuristics = np.where(heuristics < threshold, heuristics * 0.6, heuristics)

    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.percentile(heuristics, 10)
    heuristics = np.where(heuristics < threshold, 0, heuristics)

    # Consider diversity by amplifying items with higher dimension-wise weights
    heuristics = np.where(heuristics > 0, heuristics * (1 + 0.3 * dim_prize_to_weight_ratio), heuristics)

    return heuristics
