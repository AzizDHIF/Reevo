import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    # Calculate the density of each item
    density = prize / np.sum(weight, axis=1)
    
    # Calculate the average weight of each item
    avg_weight = np.mean(weight, axis=1)
    
    # Calculate the variance of weights for each item
    weight_var = np.var(weight, axis=1)
    
    # Combine various factors to determine the promising value of each item
    promising_value = density * (1 - avg_weight) * (1 / (1 + weight_var))
    
    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.mean(promising_value)  # set threshold to the mean of promising values
    promising_value[promising_value < threshold] = 0
    
    return promising_value
