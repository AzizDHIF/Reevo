import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Heuristics function for Solving Multiple Knapsack Problems (MKP) 
    through stochastic solution sampling based on "heuristics".

    Parameters:
    prize (np.ndarray): Prize of each item, shape (n,)
    weight (np.ndarray): Weight of each item, shape (n, m)

    Returns:
    np.ndarray: Heuristics of shape (n,), heuristics[i] indicates how promising 
    it is to include item i in the solution.
    """

    # Calculate prize-to-weight ratio for each item
    prize_to_weight_ratio = prize / np.sum(weight, axis=1)

    # Normalize weights to [0, 1] range
    normalized_weight = weight / np.max(weight, axis=0)

    # Calculate the average normalized weight for each item
    average_normalized_weight = np.mean(normalized_weight, axis=1)

    # Combine factors: prize-to-weight ratio and average normalized weight
    combined_heuristics = prize_to_weight_ratio / (average_normalized_weight + 1e-6)

    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.mean(combined_heuristics)
    sparsified_heuristics = np.where(combined_heuristics > threshold, combined_heuristics, 0)

    return sparsified_heuristics
