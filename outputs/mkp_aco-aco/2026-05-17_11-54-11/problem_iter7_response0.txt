import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Heuristics function for Solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.
    
    Parameters:
    prize (np.ndarray): Prize of each item, shape (n,)
    weight (np.ndarray): Weight of each item, shape (n, m)
    
    Returns:
    heuristics (np.ndarray): Heuristics value for each item, shape (n,)
    """
    # Calculate the sum of weights for each item
    sum_weights = np.sum(weight, axis=1)
    
    # Handle edge case where sum of weights is zero
    sum_weights = np.maximum(sum_weights, 1e-8)
    
    # Calculate the prize-to-weight ratio for each item
    prize_to_weight_ratio = prize / sum_weights
    
    # Calculate the average weight for each item
    average_weight = np.mean(weight, axis=1)
    
    # Combine the prize-to-weight ratio and average weight to determine the heuristics
    heuristics = prize_to_weight_ratio * (1 - average_weight)
    
    # Sparsify the heuristics by setting unpromising elements to zero
    heuristics[heuristics < np.median(heuristics)] = 0
    
    return heuristics
