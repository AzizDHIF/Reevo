import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Heuristics function for Solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.

    Parameters:
    prize (np.ndarray): The prize of each item. Shape: (n,)
    weight (np.ndarray): The weight of each item in each dimension. Shape: (n, m)

    Returns:
    heuristics (np.ndarray): Heuristics of shape (n,) indicating how promising it is to include item i in the solution.
    """
    # Calculate the average weight of each item across all dimensions
    avg_weight = np.mean(weight, axis=1)

    # Calculate the relative prize of each item compared to its average weight
    relative_prize = prize / avg_weight

    # Calculate the standard deviation of weights for each item across all dimensions
    weight_std = np.std(weight, axis=1)

    # Combine relative prize and standard deviation of weights to determine the promise of each item
    promise = relative_prize / (weight_std + 1e-8)  # avoid division by zero

    # Sparsify the heuristics by setting unpromising elements to zero (e.g., items with very high weights)
    max_avg_weight = np.max(avg_weight)
    min_promise = np.min(promise)
    max_promise = np.max(promise)
    threshold = np.median(promise)  # use median as threshold
    promise[avg_weight > 0.75 * max_avg_weight] = 0  # items with high average weights
    promise[promise < threshold] = 0  # items with low promise

    # Normalize the promise values to ensure they sum up to 1
    if np.sum(promise) > 0:
        heuristics = promise / np.sum(promise)
    else:
        heuristics = np.zeros_like(promise)

    return heuristics
