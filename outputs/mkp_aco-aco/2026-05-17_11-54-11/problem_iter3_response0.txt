import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Heuristics function for Solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.

    Parameters:
    prize (np.ndarray): The prize of each item, shape (n,).
    weight (np.ndarray): The weight of each item, shape (n, m).

    Returns:
    np.ndarray: The heuristics of shape (n,), indicating how promising it is to include each item in the solution.
    """
    
    # Calculate the prize-to-weight ratio for each item
    ratio = prize / np.sum(weight, axis=1)
    
    # Normalize the weights to get a sparse representation
    normalized_weights = weight / np.sum(weight, axis=0)
    
    # Combine the prize-to-weight ratio with the normalized weights
    combined_heuristics = ratio * np.sum(normalized_weights, axis=1)
    
    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.mean(combined_heuristics)
    sparsified_heuristics = np.where(combined_heuristics > threshold, combined_heuristics, 0)
    
    return sparsified_heuristics
