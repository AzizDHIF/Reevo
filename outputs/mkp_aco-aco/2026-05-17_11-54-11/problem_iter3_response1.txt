import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    # Calculate prize-to-weight ratios for each item
    ratios = prize / np.sum(weight, axis=1)
    
    # Normalize weights to have a maximum value of 1
    normalized_weights = weight / np.max(weight, axis=0)
    
    # Combine factors: prize-to-weight ratio and normalized weights
    factors = np.column_stack((ratios, np.sum(normalized_weights, axis=1)))
    
    # Calculate heuristics by taking the maximum of the combined factors
    heuristics = np.max(factors, axis=1)
    
    # Sparsify heuristics by setting unpromising elements to zero
    threshold = np.mean(heuristics)
    heuristics[heuristics < threshold] = 0
    
    return heuristics
