import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Heuristics function for solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.
    
    Parameters:
    prize (np.ndarray): The prize of each item, of shape (n,).
    weight (np.ndarray): The weight of each item, of shape (n, m).
    
    Returns:
    np.ndarray: The heuristics value of each item, of shape (n,).
    """
    
    # Calculate the average weight of each item
    avg_weight = np.mean(weight, axis=1)
    
    # Calculate the variance of the weights of each item
    var_weight = np.var(weight, axis=1)
    
    # Combine various factors to determine how promising it is to select an item
    factor1 = prize / avg_weight  # prize-to-weight ratio
    factor2 = prize / (avg_weight + var_weight)  # prize-to-(weight + variance) ratio
    factor3 = prize / np.max(weight, axis=1)  # prize-to-max-weight ratio
    
    # Calculate the final heuristics by combining the factors
    heuristics = factor1 + factor2 + factor3
    
    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.mean(heuristics)  # set threshold to the mean heuristics value
    heuristics[heuristics < threshold] = 0  # set heuristics values below threshold to zero
    
    return heuristics
