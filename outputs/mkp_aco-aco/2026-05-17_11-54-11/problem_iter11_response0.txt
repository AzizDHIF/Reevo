```python
import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Heuristics function for solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.
    
    Parameters:
    prize (np.ndarray): The prize of each item, of shape (n,).
    weight (np.ndarray): The weight of each item, of shape (n, m).
    
    Returns:
    np.ndarray: The heuristics value of each item, of shape (n,).
    """
    
    # Calculate the key factor: prize-to-weight ratio considering all dimensions
    total_weight = np.sum(weight, axis=1)
    total_weight[total_weight == 0] = 1e-9  # avoid division by zero with a small value
    
    # Calculate the heuristics value based on the prize-to-weight ratio
    # and simplify calculations
    prize_to_weight_ratio = prize / total_weight
    
    # Combine various factors to determine how promising it is to select an item
    heuristics = prize_to_weight_ratio * (1 / np.max(weight, axis=1))
    
    # Sparsify the heuristics by setting unpromising elements to zero
    mask = prize_to_weight_ratio > np.mean(prize_to_weight_ratio)
    heuristics[~mask] = 0
    
    return heuristics
```
