import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Heuristics function for Solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.

    Parameters:
    prize (np.ndarray): array of shape (n,) representing the prize of each item
    weight (np.ndarray): array of shape (n, m) representing the weight of each item in each dimension

    Returns:
    np.ndarray: array of shape (n,) representing the heuristics of each item
    """
    # Calculate the maximum weight constraint for each dimension
    max_weight = np.ones(weight.shape[1])
    
    # Normalize weights
    normalized_weight = weight / max_weight
    
    # Calculate the maximum normalized weight for each item
    max_normalized_weight = np.max(normalized_weight, axis=1)
    
    # Calculate the average normalized weight for each item
    avg_normalized_weight = np.mean(normalized_weight, axis=1)
    
    # Calculate the variance of normalized weights for each item
    var_normalized_weight = np.var(normalized_weight, axis=1)
    
    # Calculate the heuristics for each item based on prize, max normalized weight, and average normalized weight
    factor1 = prize / max_normalized_weight  # prize-to-max-weight ratio
    factor2 = prize / avg_normalized_weight  # prize-to-avg-weight ratio
    factor3 = prize / (avg_normalized_weight + var_normalized_weight)  # prize-to-(avg-weight + variance) ratio
    
    # Combine factors intuitively
    heuristics = factor1 * 0.4 + factor2 * 0.3 + factor3 * 0.3  # Assign weights to each factor
    
    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.mean(heuristics)  # set threshold to the mean heuristics value
    heuristics[heuristics < threshold] = 0  # set heuristics values below threshold to zero
    
    return heuristics
