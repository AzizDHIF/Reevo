import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    This function calculates a heuristic value for each item in the Multiple Knapsack Problem.
    The heuristic value is a measure of how promising it is to include an item in the solution.
    
    Parameters:
    prize (np.ndarray): A 1D array of shape (n,) where n is the number of items, representing the prize of each item.
    weight (np.ndarray): A 2D array of shape (n, m) where n is the number of items and m is the number of dimensions, representing the weight of each item in each dimension.
    
    Returns:
    np.ndarray: A 1D array of shape (n,) representing the heuristic value of each item.
    """
    
    # Calculate the prize-to-weight ratio for each item
    prize_to_weight_ratio = prize / np.sum(weight, axis=1)
    
    # Calculate the L1 normalization of the weights for each item
    weight_l1_norm = np.sum(weight, axis=1)
    weight_l1_normalized = weight / weight_l1_norm[:, np.newaxis]
    
    # Calculate the average weight in each dimension for all items
    avg_weight = np.mean(weight, axis=0)
    
    # Calculate the weight deviation from the average for each item
    weight_deviation = np.sum(np.abs(weight - avg_weight), axis=1)
    
    # Combine the factors to determine the heuristic value
    heuristic_value = prize_to_weight_ratio * (1 - weight_deviation)
    
    # Sparsify the heuristic by setting unpromising elements to zero
    threshold = np.mean(heuristic_value)
    heuristic_value[heuristic_value < threshold] = 0
    
    return heuristic_value
