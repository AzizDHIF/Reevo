import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Heuristics function for solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.
    
    Parameters:
    prize (np.ndarray): The prize of each item, of shape (n,).
    weight (np.ndarray): The weight of each item, of shape (n, m).
    
    Returns:
    np.ndarray: The heuristics value of each item, of shape (n,).
    """
    
    # Calculate the total weight considering all dimensions
    total_weight = np.sum(weight, axis=1)
    total_weight[total_weight == 0] = 1  # avoid division by zero
    
    # Calculate the average weight across dimensions
    avg_weight = np.mean(weight, axis=1)
    
    # Calculate the prize-to-weight ratio across all dimensions
    ratio = prize / total_weight
    
    # Calculate the prize-to-average-weight ratio
    avg_ratio = prize / avg_weight
    
    # Combine the ratios to determine the heuristics value
    heuristics = np.minimum(ratio, avg_ratio)
    
    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.mean(heuristics)  # calculate a threshold based on the mean of heuristics
    heuristics[heuristics < threshold] = 0  # set heuristics values below the threshold to zero
    
    return heuristics
