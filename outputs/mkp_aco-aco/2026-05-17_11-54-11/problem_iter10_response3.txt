import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Heuristics function for solving Multiple Knapsack Problems (MKP) through stochastic solution sampling.
    
    Parameters:
    prize (np.ndarray): The prize of each item, of shape (n,).
    weight (np.ndarray): The weight of each item, of shape (n, m).
    
    Returns:
    np.ndarray: The heuristics value of each item, of shape (n,).
    """
    
    # Calculate total weight and avoid division by zero
    total_weight = np.sum(weight, axis=1)
    total_weight = np.where(total_weight == 0, np.inf, total_weight)
    
    # Calculate heuristics value based on prize-to-weight ratio, handling division by zero
    with np.errstate(divide='ignore', invalid='ignore'):
        heuristics = prize / total_weight
        heuristics = np.where(np.isinf(heuristics), 0, heuristics)
        heuristics = np.where(np.isnan(heuristics), 0, heuristics)
    
    return heuristics
