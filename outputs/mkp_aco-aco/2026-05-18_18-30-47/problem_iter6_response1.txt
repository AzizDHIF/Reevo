import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Improved heuristic for Multiple Knapsack Problems (MKP) focusing on prize-to-weight ratio.
    
    This heuristic calculates the promise of each item based on its prize-to-weight ratio,
    ensuring numerical stability by adding a small epsilon value to the denominator.

    Args:
    prize (np.ndarray): A 1D array representing the prize of each item.
    weight (np.ndarray): A 2D array representing the weights of each item.

    Returns:
    np.ndarray: A 1D array where each element represents how promising it is to include the corresponding item in the solution.
    """
    # Calculate the sum of weights for each item
    weight_sum = np.sum(weight, axis=1)
    
    # Define a small epsilon value to avoid division by zero and ensure numerical stability
    epsilon = 1e-8  
    
    # Calculate the heuristics as prize-to-weight ratio, adding epsilon to the denominator for stability
    heuristics = prize / (weight_sum + epsilon)
    
    return heuristics
