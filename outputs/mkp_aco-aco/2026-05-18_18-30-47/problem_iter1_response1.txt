import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Heuristics function for the Multiple Knapsack Problem.

    Parameters:
    prize (np.ndarray): The prizes of the items.
    weight (np.ndarray): The weights of the items.

    Returns:
    np.ndarray: The heuristics values of the items.
    """
    n, m = weight.shape

    # Calculate the efficiency of each item
    efficiency = prize / np.linalg.norm(weight, axis=1)

    # Calculate the sparsity of each item
    sparsity = 1 - np.sum(weight, axis=1) / m

    # Combine the efficiency and sparsity
    heuristics = efficiency * sparsity

    # Set unpromising elements to zero
    threshold = np.median(heuristics)
    heuristics[heuristics < threshold] = 0

    return heuristics
