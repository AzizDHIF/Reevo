import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Stochastic solution sampling heuristic for Multiple Knapsack Problems (MKP).
    
    This heuristic combines multiple factors (prize, L2 norm of weights, 
    the number of non-zero weights, and the maximum weight) with normalization 
    to determine an item's promise.
    
    Args:
    prize (np.ndarray): A 1D array representing the prize of each item.
    weight (np.ndarray): A 2D array representing the weights of each item.

    Returns:
    np.ndarray: A 1D array where each element represents how promising it is to 
    include the corresponding item in the solution.
    """
    n, m = weight.shape
    
    # Calculate the L2 norm of weights for each item
    weight_norm = np.linalg.norm(weight, axis=1)
    
    # Normalize prizes and inverse of L2 norms
    prize_normalized = prize / np.mean(prize) if np.mean(prize) != 0 else 0
    weight_norm_inverse = 1 / (weight_norm + 1e-6)
    weight_norm_inverse_normalized = weight_norm_inverse / np.mean(weight_norm_inverse) if np.mean(weight_norm_inverse) != 0 else 0
    
    # Calculate the number of non-zero weights for each item and normalize
    weight_nonzero = np.sum(weight > 0, axis=1)
    weight_nonzero_normalized = weight_nonzero / np.mean(weight_nonzero) if np.mean(weight_nonzero) != 0 else 0
    
    # Calculate the maximum weight for each item and normalize
    weight_max = np.max(weight, axis=1)
    weight_max_normalized = weight_max / np.mean(weight_max) if np.mean(weight_max) != 0 else 0
    
    # Calculate the average weight for each item and normalize
    weight_avg = np.mean(weight, axis=1)
    weight_avg_normalized = weight_avg / np.mean(weight_avg) if np.mean(weight_avg) != 0 else 0
    
    # Combine the factors with normalization
    factor1 = prize_normalized
    factor2 = weight_norm_inverse_normalized
    factor3 = 1 - weight_nonzero_normalized  # Prefer fewer nonzero weights
    factor4 = 1 - weight_max_normalized  # Prefer smaller maximum weights
    factor5 = 1 - weight_avg_normalized  # Prefer smaller average weights
    heuristics = factor1 * factor2 * factor3 * factor4 * factor5
    
    # Sparsify the heuristics with a threshold
    threshold = np.mean(heuristics) / 2 if np.mean(heuristics) != 0 else 0
    heuristics = np.where(heuristics < threshold, 0, heuristics)
    
    return heuristics
