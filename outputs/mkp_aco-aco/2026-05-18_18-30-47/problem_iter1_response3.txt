import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    A heuristic function for solving Multiple Knapsack Problems (MKP).
    
    Parameters:
    prize (np.ndarray): a 1D array of shape (n,) representing the prizes of items.
    weight (np.ndarray): a 2D array of shape (n, m) representing the weights of items.
    
    Returns:
    heuristics (np.ndarray): a 1D array of shape (n,) indicating how promising it is to include each item in the solution.
    """

    # Calculate the average weight of each item
    avg_weight = np.mean(weight, axis=1)
    
    # Calculate the prize-to-weight ratio for each item
    prize_to_weight_ratio = prize / avg_weight
    
    # Calculate the sparsity of each item's weights (i.e., the number of non-zero weights)
    weight_sparsity = np.count_nonzero(weight, axis=1)
    
    # Combine the prize-to-weight ratio and weight sparsity to determine the heuristic value
    heuristic_values = prize_to_weight_ratio * (1 - weight_sparsity / weight.shape[1])
    
    # Sparsify the heuristic values by setting unpromising elements to zero
    threshold = np.mean(heuristic_values)  # Set the threshold to the mean of the heuristic values
    heuristic_values[heuristic_values < threshold] = 0
    
    return heuristic_values
