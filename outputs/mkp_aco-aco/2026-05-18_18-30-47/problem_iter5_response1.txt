import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Stochastic solution sampling heuristic for Multiple Knapsack Problems (MKP).
    
    This heuristic combines the normalized prize of each item, the inverse of 
    the L2 norm of its weights, the number of dimensions where the item's 
    weight is non-zero, and a small value to avoid division by zero to determine 
    its promise. Unpromising items are filtered out.

    Args:
    prize (np.ndarray): A 1D array representing the prize of each item.
    weight (np.ndarray): A 2D array representing the weights of each item.

    Returns:
    np.ndarray: A 1D array where each element represents how promising it is to 
    include the corresponding item in the solution.
    """
    n, m = weight.shape
    
    # Add a small value to the prizes to avoid division by zero
    prize = prize + 1e-6
    
    # Normalize prizes to have a mean of 1
    prize_normalized = prize / np.mean(prize)
    
    # Calculate the L2 norm of weights for each item
    weight_norm = np.linalg.norm(weight, axis=1)
    
    # Clip weights to avoid division by zero
    weight_norm = np.clip(weight_norm, a_min=1e-6, a_max=None)
    
    # Calculate the inverse of the L2 norm
    weight_norm_inverse = 1 / weight_norm
    
    # Normalize the inverse of the L2 norm
    weight_norm_inverse_normalized = weight_norm_inverse / np.mean(weight_norm_inverse)
    
    # Calculate the number of non-zero weights for each item
    weight_nonzero = np.sum(weight > 0, axis=1)
    
    # Normalize the number of non-zero weights to have a mean of 1
    weight_nonzero_normalized = weight_nonzero / np.mean(weight_nonzero)
    
    # Calculate the prize-to-weight ratio for each item
    prize_to_weight_ratio = prize / np.sum(weight, axis=1)
    
    # Normalize the prize-to-weight ratio
    prize_to_weight_ratio_normalized = prize_to_weight_ratio / np.mean(prize_to_weight_ratio)
    
    # Combine the factors with normalization and sparsification
    heuristics = (prize_normalized * 
                  weight_norm_inverse_normalized * 
                  (1 + weight_nonzero_normalized) * 
                  prize_to_weight_ratio_normalized)
    
    # Sparsify the heuristics with a threshold
    threshold = np.mean(heuristics) / 2
    heuristics = np.where(heuristics < threshold, 0, heuristics)
    
    return heuristics
