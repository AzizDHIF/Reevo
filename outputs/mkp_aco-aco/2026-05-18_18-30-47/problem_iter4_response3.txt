import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Improved stochastic solution sampling heuristic for Multiple Knapsack Problems (MKP).
    
    This heuristic prioritizes the prize-to-weight ratio to determine the promise of each item.

    Args:
    prize (np.ndarray): A 1D array representing the prize of each item.
    weight (np.ndarray): A 2D array representing the weights of each item.

    Returns:
    np.ndarray: A 1D array where each element represents how promising it is to 
    include the corresponding item in the solution.
    """
    # Calculate the sum of weights for each item
    weight_sum = np.sum(weight, axis=1)
    
    # Clip weights to avoid division by zero and ensure numerical stability
    weight_sum = np.clip(weight_sum, a_min=1e-9, a_max=None)
    
    # Calculate the prize-to-weight ratio for each item, handling potential divide-by-zero
    heuristics = np.where(weight_sum > 0, prize / weight_sum, 0)
    
    return heuristics
