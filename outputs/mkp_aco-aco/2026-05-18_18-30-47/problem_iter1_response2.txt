import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    This heuristics function generates a promising score for each item 
    based on a combination of their prize, average weight, and weight uniformity.
    
    Parameters:
    prize (np.ndarray): The prize of each item (shape: (n,)).
    weight (np.ndarray): The weight of each item in multiple dimensions (shape: (n, m)).
    
    Returns:
    np.ndarray: The heuristics score for each item (shape: (n,)).
    """

    # Calculate the average weight for each item
    avg_weight = np.mean(weight, axis=1)

    # Calculate the uniformity of weights for each item
    weight_uniformity = np.std(weight, axis=1)

    # Generate a score for each item based on a balance of prize and average weight
    score = prize / (avg_weight + 1e-6)  # adding a small value to avoid division by zero

    # Factor in the weight uniformity to penalize items with high variability in weights
    score *= 1 / (weight_uniformity + 1)

    # Scale the scores to be between 0 and 1
    score = score / (np.max(score) + 1e-6)

    # Sparsify the heuristics by setting unpromising elements to zero
    threshold = np.mean(score)
    score[score < threshold] = 0

    return score
