import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Improved stochastic solution sampling heuristic for Multiple Knapsack Problems (MKP).
    
    This heuristic combines the normalized prize-to-weight ratio of each item, 
    the inverse of the L2 norm of its weights, and the number of dimensions where 
    the item's weight is non-zero to determine its promise. Unpromising items are 
    filtered out. The heuristics are then sparsified by setting unpromising elements 
    to zero.

    Args:
    prize (np.ndarray): A 1D array representing the prize of each item.
    weight (np.ndarray): A 2D array representing the weights of each item.

    Returns:
    np.ndarray: A 1D array where each element represents how promising it is to 
    include the corresponding item in the solution.
    """
    n, m = weight.shape
    
    # Clip weights to avoid division by zero
    weight_clipped = np.clip(weight, a_min=1e-6, a_max=None)
    
    # Calculate the prize-to-weight ratio for each item
    prize_to_weight = prize / np.sum(weight_clipped, axis=1)
    
    # Normalize the prize-to-weight ratio to have a mean of 1
    prize_to_weight_normalized = prize_to_weight / np.mean(prize_to_weight)
    
    # Calculate the L2 norm of weights for each item
    weight_norm = np.linalg.norm(weight_clipped, axis=1)
    
    # Calculate the inverse of the L2 norm
    weight_norm_inverse = 1 / weight_norm
    
    # Normalize the inverse of the L2 norm to have a mean of 1
    weight_norm_inverse_normalized = weight_norm_inverse / np.mean(weight_norm_inverse)
    
    # Calculate the number of non-zero weights for each item
    weight_nonzero = np.sum(weight > 0, axis=1)
    
    # Normalize the number of non-zero weights to have a mean of 1
    weight_nonzero_normalized = weight_nonzero / np.mean(weight_nonzero)
    
    # Combine the factors with normalization
    heuristics = prize_to_weight_normalized * weight_norm_inverse_normalized * (1 + weight_nonzero_normalized)
    
    # Add small values to the heuristics to avoid division by zero
    heuristics = heuristics + 1e-6
    
    # Sparsify the heuristics with a threshold
    threshold = np.mean(heuristics) / 2
    heuristics = np.where(heuristics < threshold, 0, heuristics)
    
    return heuristics
