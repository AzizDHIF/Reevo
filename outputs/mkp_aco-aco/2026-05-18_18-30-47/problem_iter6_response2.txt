import numpy as np

def heuristics_v2(prize: np.ndarray, weight: np.ndarray) -> np.ndarray:
    """
    Improved stochastic solution sampling heuristic for Multiple Knapsack Problems (MKP).
    
    This heuristic combines the normalized prize of each item, the inverse of 
    the L2 norm of its weights, and the number of dimensions where the item's 
    weight is non-zero to determine its promise. Unpromising items are filtered out.

    Args:
    prize (np.ndarray): A 1D array representing the prize of each item.
    weight (np.ndarray): A 2D array representing the weights of each item.

    Returns:
    np.ndarray: A 1D array where each element represents how promising it is to 
    include the corresponding item in the solution.
    """
    n, m = weight.shape
    
    # Normalize prizes to have a mean of 1
    prize_normalized = prize / np.mean(prize)
    
    # Calculate the L2 norm of weights for each item
    weight_norm = np.linalg.norm(weight, axis=1)
    
    # Calculate the inverse of the L2 norm
    weight_norm_inverse = 1 / (weight_norm + 1e-6)
    
    # Normalize the inverse of the L2 norm
    weight_norm_inverse_normalized = weight_norm_inverse / np.mean(weight_norm_inverse)
    
    # Calculate the L1 norm of weights for each item
    weight_l1_norm = np.sum(weight, axis=1)
    
    # Calculate the inverse of the L1 norm
    weight_l1_norm_inverse = 1 / (weight_l1_norm + 1e-6)
    
    # Normalize the inverse of the L1 norm
    weight_l1_norm_inverse_normalized = weight_l1_norm_inverse / np.mean(weight_l1_norm_inverse)
    
    # Calculate the number of non-zero weights for each item
    weight_nonzero = np.sum(weight > 0, axis=1)
    
    # Normalize the number of non-zero weights to have a mean of 1
    weight_nonzero_normalized = weight_nonzero / np.mean(weight_nonzero)
    
    # Combine the factors with normalization
    heuristics = prize_normalized * (weight_norm_inverse_normalized + weight_l1_norm_inverse_normalized) * (1 + weight_nonzero_normalized)
    
    # Normalize heuristics to have a mean of 1
    heuristics_normalized = heuristics / np.mean(heuristics)
    
    # Sparsify the heuristics with a threshold based on the top 50% of items
    threshold = np.percentile(heuristics_normalized, 50)
    heuristics_sparse = np.where(heuristics_normalized < threshold, 0, heuristics_normalized)
    
    return heuristics_sparse
