double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weighted_sum = 0;
    double capacity_ratio = 1.0;
    
    for(int j = 0; j < dimension; j++) {
        capacity_ratio *= capacity[j];
        weighted_sum += weights[j][voisinage[index_item]];
    }
    
    if(capacity_ratio == 0) {
        return 0.0;
    }
    
    double utility_per_capacity = profit[voisinage[index_item]] / (weighted_sum / capacity_ratio);
    
    return utility_per_capacity;
}
