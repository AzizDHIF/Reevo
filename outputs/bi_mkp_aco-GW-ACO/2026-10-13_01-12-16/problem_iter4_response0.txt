double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weight_inverse_sum = 0;
    for (int j = 0; j < dimension; j++) {
        weight_inverse_sum += 1 / (weights[j][voisinage[index_item]] / capacity[j]);
    }
    
    double objective_contribution = profit[voisinage[index_item]] / (1 + weight_inverse_sum);
    
    return objective_contribution;
}
