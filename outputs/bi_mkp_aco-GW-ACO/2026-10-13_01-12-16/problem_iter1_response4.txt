    double sparsified_factor = 0;

	}
	
    double profit_factor = profit[voisinage[index_item]] / combined_factor;
    double weight_factor = 1 / combined_factor; // Assuming minimizing total weight is desirable

    // Sparsifying the heuristic
    if(combined_factor > threshold){
