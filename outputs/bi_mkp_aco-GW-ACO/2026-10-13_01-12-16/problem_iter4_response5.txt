double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double utility = 0.0;

    for (int j = 0; j < dimension; j++) {
        utility += weights[j][voisinage[index_item]] / capacity[j];
    }

    double promising_factor = profit[voisinage[index_item]] / utility;
    double sparsity_threshold = 0.1;

    return (promising_factor > sparsity_threshold) ? promising_factor : 0;
}
