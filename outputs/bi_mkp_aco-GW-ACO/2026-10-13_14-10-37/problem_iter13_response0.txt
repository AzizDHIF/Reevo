double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 0;
    
    for (int j = 0; j < dimension; j++) {
        double weight_ratio = weights[j][voisinage[index_item]] / capacity[j];
        h += weight_ratio * weight_ratio; // Squared weight ratio
    }
    
    double normalized_h = h / dimension;  // Normalize sum of squared weight ratios
    
    return profit[voisinage[index_item]] / (1 + normalized_h); // Normalize profit by adding 1 to normalized sum
}
