double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit)
{
    double h = 0;
    
    for(int j = 0; j < dimension; j++) {
        double weight_ratio = weights[j][voisinage[index_item]] / capacity[j];
        h += weight_ratio * weight_ratio; // Use squared weight ratio
    }

    double normalized_sqrt_h = sqrt(h / dimension); // Normalize by taking square root of average squared weight ratios

    return (profit[voisinage[index_item]] * normalized_sqrt_h); // Normalize profit with normalized_sqrt_h
}
