double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 0.0;
    
    for (int j = 0; j < dimension; j++) {
        if (weights[j][voisinage[index_item]] > capacity[j]) {
            return 0.0; // Set unpromising elements to 0
        }
        h += weights[j][voisinage[index_item]] / capacity[j]; // Consider weight-to-capacity ratio
    }
    
    if (h == 0.0) {
        return 0.0; // Avoid unnecessary computations if the weight-to-capacity ratio is 0
    }
    
    return profit[voisinage[index_item]] / h;
}
