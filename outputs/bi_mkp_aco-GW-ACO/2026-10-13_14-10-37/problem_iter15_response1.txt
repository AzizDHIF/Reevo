double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {

    double h = 1.0; // Initialize h with 1 for geometric mean

    for (int j = 0; j < dimension; j++) {
        double weight_ratio = weights[j][voisinage[index_item]] / capacity[j];
        
        // Use geometric mean instead of sum of squared weight ratios
        h *= sqrt(weight_ratio);
    }

    return profit[voisinage[index_item]] / h; // Normalize by multiplying with geometric mean
}
