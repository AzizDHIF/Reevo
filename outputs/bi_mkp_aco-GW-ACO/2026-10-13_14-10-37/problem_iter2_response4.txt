double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 0;
    int cnt = 0;

    for (int j = 0; j < dimension; j++) {
        double weight_j = weights[j][voisinage[index_item]];
        h += weight_j / capacity[j];
        cnt += weight_j > capacity[j];
    }

    double normalization = dimension * 0.5;

    return cnt > normalization ? 0 : profit[voisinage[index_item]] / h;
}
