double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double h=0;
    double sum_weight = 0;
    double sum_profit_capability = 0;
    double max_profit=0;
    for(int j=0;j<dimension;j++){
        sum_weight += weights[j][index_item];
        h += weights[j][voisinage[index_item]]/capacity[j];
        if(capacity[j] >= weights[j][index_item])
            sum_profit_capability += profit[index_item];
    }
    
    for(int k=0; k<nb_voisinage; k++){
        if(profit[k] > max_profit)
            max_profit = profit[k];
    }
    
    if(sum_weight > sum_profit_capability)
        return 0;
    return sum_profit_capability/max_profit/h;
}
