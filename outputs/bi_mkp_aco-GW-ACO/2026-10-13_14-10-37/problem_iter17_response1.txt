double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 0;
    
    for (int j = 0; j < dimension; j++) {
        double weight_ratio = weights[j][voisinage[index_item]] / sqrt(capacity[j]); // Utilize square root normalization
        h += weight_ratio; // Use weighted sum of normalized weight ratios
    }
    
    return profit[voisinage[index_item]] * pow(h, 1.0 / dimension); // Use geometric mean for weight ratios
}
