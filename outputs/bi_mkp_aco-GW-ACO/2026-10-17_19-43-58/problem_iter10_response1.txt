double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weight_sum = 0, profit_sum = 0, weight_capacity_contribution = 0;

    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][voisinage[index_item]];
        profit_sum += profit[voisinage[index_item]];
        weight_capacity_contribution += weights[j][voisinage[index_item]] / capacity[j];
    }

    double weight_norm = weight_sum / dimension;
    double profit_norm = profit_sum / dimension;

    return profit_norm / (1 + weight_norm + weight_capacity_contribution);
}
