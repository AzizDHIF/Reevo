double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double promising_factor = 0;

    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            promising_factor += weights[j][voisinage[index_item]] / capacity[j];
        }
    }

    double profit_weight_ratio = 0;
    if (promising_factor > 0) {
        profit_weight_ratio = profit[voisinage[index_item]] / promising_factor;
    }

    return profit_weight_ratio;
}
