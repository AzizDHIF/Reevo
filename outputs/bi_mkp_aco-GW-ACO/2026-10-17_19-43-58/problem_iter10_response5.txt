double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit)
{
    double weight_capacity_ratio = 0, profit_sum = 0;

    for (int j = 0; j < dimension; j++) {
        weight_capacity_ratio += weights[j][voisinage[index_item]] / capacity[j];
        profit_sum += profit[voisinage[index_item]];
    }

    double profit_norm = profit_sum / dimension;
    double weight_norm = weight_capacity_ratio / dimension;

    return profit_norm / (1 + weight_norm);
}
