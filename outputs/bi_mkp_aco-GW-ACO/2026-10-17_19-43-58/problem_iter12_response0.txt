double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weight_sum = 0, profit_sum = 0, promising_ratio = 0;

    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][voisinage[index_item]] / dimension;
        profit_sum += profit[voisinage[index_item]] / dimension;
        if (capacity[j] > 0) {
            promising_ratio += weights[j][voisinage[index_item]] / dimension / capacity[j];
        }
    }

    return (profit_sum - weight_sum) * (1 + promising_ratio);
}
