double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit)
{
    double weight_sum = 0, profit_sum = 0, promising_ratio = 0;

    for (int j = 0; j < dimension; j++) {
        double item_weight = weights[j][voisinage[index_item]];
        weight_sum += item_weight;
        profit_sum += profit[voisinage[index_item]];

        if (item_weight > 0 && capacity[j] > 0) {
            promising_ratio += item_weight / capacity[j];
        }
    }

    double weight_norm = weight_sum / dimension;
    double profit_norm = profit_sum / dimension;

    return (profit_norm - weight_norm) * (1 + promising_ratio);
}
