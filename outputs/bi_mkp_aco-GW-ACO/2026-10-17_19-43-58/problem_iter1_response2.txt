double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double h=0, a=0, b=0, c=0;
    
    for(int j=0;j<dimension;j++){
        h += weights[j][voisinage[index_item]] / capacity[j];
        a += weights[j][voisinage[index_item]];
        b += weights[j][voisinage[index_item]] * j;
        c += weights[j][voisinage[index_item]] * j * j;
    }
    
    if(h == 0){
        return 0;
    }
    
    return 1.0 / (1.0 + a) + b / h - c * capacity[0] / h / 1000 + profit[voisinage[index_item]] / h;
}
