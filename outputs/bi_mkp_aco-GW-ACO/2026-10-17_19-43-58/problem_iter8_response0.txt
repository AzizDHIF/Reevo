double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weight_sum = 0, profit_sum = 0, weight_norm = 0, profit_norm = 0, h = 0;

    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][voisinage[index_item]];
        profit_sum += profit[voisinage[index_item]];

        if (capacity[j] > 0)
            h += weights[j][voisinage[index_item]] / capacity[j];
    }

    weight_norm = weight_sum / dimension;
    profit_norm = profit_sum / dimension;

    return profit_norm / (1 + weight_sum / dimension + h);
}
