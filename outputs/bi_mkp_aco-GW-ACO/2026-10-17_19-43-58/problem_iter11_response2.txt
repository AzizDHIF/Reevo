double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weight_sum = 0, profit_sum = 0, weight_to_capacity_ratio = 0;
    double promising_factor = 0;
    
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][voisinage[index_item]];
        profit_sum += profit[voisinage[index_item]];
        if (capacity[j] > 0) {
            weight_to_capacity_ratio = weights[j][voisinage[index_item]] / capacity[j];
            promising_factor += weight_to_capacity_ratio;
        }
    }
    
    double weight_norm = weight_sum / dimension;
    double profit_norm = profit_sum / dimension;
    
    return (profit_norm - weight_norm) * (1 + promising_factor);
}
