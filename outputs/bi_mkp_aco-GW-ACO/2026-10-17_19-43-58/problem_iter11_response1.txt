double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weight_sum = 0, profit_sum = 0, weight_norm, profit_norm;
    double promising_factor = 0;

    for (int j = 0; j < nb_voisinage; j++) {
        weight_sum += weights[0][voisinage[j]];
        profit_sum += profit[voisinage[j]];
        if (capacity[0] > 0 && weights[0][voisinage[j]] > 0) {
            promising_factor += profit[voisinage[j]] / weights[0][voisinage[j]];
        }
    }

    weight_norm = weight_sum / nb_voisinage;
    profit_norm = profit_sum / nb_voisinage;

    return (profit_norm - weight_norm) * (1 + promising_factor);
}
