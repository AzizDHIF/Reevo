double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weight_norm = 0, profit_norm = 0;

    for (int j = 0; j < dimension; j++) {
        weight_norm += weights[j][voisinage[index_item]] / capacity[j];
        profit_norm += profit[voisinage[index_item]] / capacity[j];
    }

    return profit_norm / (1 + weight_norm);
}
