double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double h=0;
    for(int j=0;j<dimension;j++){
			h=h+weights[j][voisinage[index_item]]/capacity[j]; 
		}
    
    double promising_factor = profit[voisinage[index_item]] / h;
    double weight_factor = h;
    double capacity_factor = 1;
    for(int j=0;j<dimension;j++){
			capacity_factor = capacity_factor * capacity[j]; 
		}
    
    return promising_factor * weight_factor * capacity_factor;
}
