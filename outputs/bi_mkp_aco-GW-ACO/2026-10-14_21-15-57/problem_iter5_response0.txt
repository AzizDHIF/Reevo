double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double promise = 0;
    
    for (int j = 0; j < dimension; j++) {
        promise += (capacity[j] >= weights[j][voisinage[index_item]]) ? profit[voisinage[index_item]] / weights[j][voisinage[index_item]] : 0;
    }
    
    return promise / dimension;
}
