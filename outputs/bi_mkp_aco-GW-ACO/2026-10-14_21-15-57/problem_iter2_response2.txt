double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 0;
    double weighted_profit = 0;

    // Calculate total weighted capacity
    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j];
    }

    // Sparsification strategy
    if (h > dimension) {
        return 0;

    // Calculate weighted profit based on profit density
    weighted_profit = profit[voisinage[index_item]] / h;
