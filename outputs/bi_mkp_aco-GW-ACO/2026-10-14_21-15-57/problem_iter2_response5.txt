double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double balance_factor = 0;
    double total_weight = 0;

    for (int j = 0; j < dimension; j++) {
        balance_factor += (weights[j][voisinage[index_item]] > 0) ? 1 : 0;
        total_weight += weights[j][voisinage[index_item]];
    }

    balance_factor /= dimension;
    double weight_factor = total_weight / dimension;

    double heuristic_value = (0.6 * balance_factor) + (0.4 * weight_factor);

    if (balance_factor < 0.5) {
        return 0;
    }

    return heuristic_value * profit[voisinage[index_item]];
}
