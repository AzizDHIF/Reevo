double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 0;
    double total_weight = 0;

    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j];
        total_weight += weights[j][voisinage[index_item]];
    }

    // Sparsification: If the combined weight exceeds capacity in any dimension, item is unlikely to be selected
    for (int j = 0; j < dimension; j++) {
        if (weights[j][voisinage[index_item]] > capacity[j]) {
            return 0;
        }
    }

    return profit[voisinage[index_item]] / (h + 1); // Adding 1 to avoid division by zero
}
