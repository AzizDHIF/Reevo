double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 0;
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] == 0) {
            return 0; // Early pruning if capacity is already exhausted
        }
        h += weights[j][voisinage[index_item]]; // Simplified calculation without division by capacity
    }
    return (h != 0) ? profit[voisinage[index_item]] / h : 0; // Avoid division by zero
}
