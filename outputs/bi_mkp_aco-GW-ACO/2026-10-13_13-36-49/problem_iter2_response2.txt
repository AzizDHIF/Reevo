double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 0;
    
    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j];
    }
    
    double importance_factor = 0.7; // Weight given to capacity constraints
    double profit_factor = 0.5; // Weight given to profit
    
    return (1 - h) * importance_factor + profit[voisinage[index_item]] * profit_factor;
}
