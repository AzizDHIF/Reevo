    double profit_weight_ratio = profit[voisinage[index_item]] / capacity[0];
    double h = 0;
    
    // Calculate the sum of normalized weights for the item
    for (int j = 0; j < dimension; j++) {
        if (weights[j][voisinage[index_item]] != 0) {
