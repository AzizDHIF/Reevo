double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double total_weight = 0, potential_profit = 0;

    for(int j=0; j<dimension; j++){
        total_weight += weights[j][voisinage[index_item]] / capacity[j];
    }

    if(total_weight == 0){
        return 0;
    }

    potential_profit = profit[voisinage[index_item]] / total_weight;

    return potential_profit;
}
