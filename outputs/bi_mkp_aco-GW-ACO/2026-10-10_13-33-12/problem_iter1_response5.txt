double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double h = 0;
    double alpha = 0.6;
    double beta = 0.4;
    double sum_weights = 0;
    
    for(int j=0; j<dimension; j++){
        sum_weights += weights[j][voisinage[index_item]] / capacity[j];
    }
    
    double rank_index = (double)(index_item + 1) / NBITEMS;
    
    h = alpha * sum_weights + beta * rank_index;
    
    if(index_item >= nb_voisinage) {
        return 0;
    }
    
    return profit[voisinage[index_item]] / h;
}
