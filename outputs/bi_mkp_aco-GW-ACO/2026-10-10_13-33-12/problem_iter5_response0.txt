double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weighted_sum = 0.0;
    double inverse_capacity_sum = 0.0;
    
    for (int j = 0; j < dimension; j++) {
        weighted_sum += weights[j][voisinage[index_item]] * capacity[j];
        inverse_capacity_sum += 1.0 / capacity[j];
    }
    
    if (inverse_capacity_sum != 0.0) {
        return profit[voisinage[index_item]] * (weighted_sum / inverse_capacity_sum);
    } else {
        return 0.0;
    }
}
