    }
    
    dim_factor = dimension / (1.0 + h_denom);  // Adjust based on the number of dimensions
    
    for (int j = 0; j < dimension; j++) {
        inv_weight_factor = 1.0 - weights[j][voisinage[index_item]] / capacity[j];
        h_numer += inv_weight_factor > 0 ? 1.0 / inv_weight_factor : 0;
    }
