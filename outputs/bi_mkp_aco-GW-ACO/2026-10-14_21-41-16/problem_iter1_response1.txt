double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double sum_weight = 0.0, sum_profit = 0.0;
    
    for(int j = 0; j < dimension; j++){
        if(weights[j][voisinage[index_item]] > capacity[j]) {
            return 0.0; // Penalize if weight exceeds capacity
        }
        sum_weight += weights[j][voisinage[index_item]] / capacity[j];
    }

    if(sum_weight == 0.0){
        return 0.0; // Avoid division by zero
    }
		
    for(int k = 0; k < nb_voisinage; k++){
        sum_profit += profit[voisinage[k]];
    }
    
    return sum_profit / sum_weight;
}
