double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double h=0;
    double weight_constraint = 0;
    for(int j=0;j<dimension;j++){
        h = h + weights[j][voisinage[index_item]]/capacity[j];
        weight_constraint += weights[j][voisinage[index_item]];
    }
    
    if(weight_constraint > capacity[0] || weight_constraint > capacity[1]){
        return 0;
    }
    
    return profit[voisinage[index_item]]/h;
}
