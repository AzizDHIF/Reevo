double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double promising_factor = 0.5;
    
    double h = 0;
    double feasibility_factor = 1;
    
    for (int j = 0; j < dimension; j++) {
        double weight_ratio = weights[j][voisinage[index_item]] / capacity[j];
        h += weight_ratio;
        feasibility_factor *= (1 - weight_ratio);
    }
    
    double h2 = (feasibility_factor > 0) ? 1 : 0;
    
    return promising_factor * profit[voisinage[index_item]] * h * feasibility_factor * h2;
}
