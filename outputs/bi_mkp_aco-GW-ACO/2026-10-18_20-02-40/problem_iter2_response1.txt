double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double ratio = 0;
    double weight_sum = 0;
    double profit_sum = 0;

    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][voisinage[index_item]];
        profit_sum += profit[voisinage[index_item]];
        ratio += weights[j][voisinage[index_item]] / capacity[j];
    }

    if (weight_sum > 0) {
        ratio = profit_sum * ratio / weight_sum;
    }

    return ratio;
}
