double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double promising_factor = 0.5;
    double normalized_weight = 0;
    double diversity_factor = 0;
    int feasibility = 1;

    for (int j = 0; j < dimension; j++) {
        normalized_weight += weights[j][voisinage[index_item]] / capacity[j];
        diversity_factor += (capacity[j] - weights[j][voisinage[index_item]]) / capacity[j];
        feasibility &= (weights[j][voisinage[index_item]] <= capacity[j]);
    }

    return feasibility ? promising_factor * (profit[voisinage[index_item]] / normalized_weight) * diversity_factor : 0;
}
