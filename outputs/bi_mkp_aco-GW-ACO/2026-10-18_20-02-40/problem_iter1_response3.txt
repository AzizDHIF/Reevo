double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double prom_ct=0, prom_aw=0, unw_ct=0, unw_aw=0, weight_diff, profit_ratio, h;
    
    for(int j=0;j<dimension;j++){
        weight_diff = weights[j][voisinage[index_item]] - capacity[j];
        profit_ratio = profit[voisinage[index_item]] / weights[j][voisinage[index_item]];
        
        if(weight_diff <= 0){ // Promising for capacity constraint
            prom_ct += weight_diff / capacity[j];
        } else {
            unw_ct += weight_diff / capacity[j];
        }
        
        if(profit_ratio >= 1){ // Promising for profit constraint
            prom_aw += (profit_ratio - 1) * capacity[j];
        } else {
            unw_aw += (profit_ratio - 1) * capacity[j];
        }
    }
    
    h = (prom_aw / fabs(prom_aw + unw_aw)) / (prom_ct / fabs(prom_ct + unw_ct) + 1);
    
    return (h > 0) ? h : 0;
}
