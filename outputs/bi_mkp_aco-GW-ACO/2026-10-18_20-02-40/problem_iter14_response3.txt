double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double promising_factor = 0.5;
    double numerator = 0;
    double denominator = 0;
    double diversity_factor = dimension;

    for (int j = 0; j < dimension; j++) {
        double weight_ratio = weights[j][voisinage[index_item]] / capacity[j];
        numerator += weight_ratio;
        if (weights[j][voisinage[index_item]] > capacity[j]) {
            return 0;
        }
        diversity_factor *= (capacity[j] - weights[j][voisinage[index_item]]) / capacity[j];
    }

    denominator = dimension - numerator;
    return promising_factor * (profit[voisinage[index_item]] / numerator) * denominator * diversity_factor;
}
