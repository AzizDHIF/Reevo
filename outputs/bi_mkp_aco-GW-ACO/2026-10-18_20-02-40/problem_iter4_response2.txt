double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double promising_factor = 0.5;
    double total_weight = 0;
    double capacity_factor = 1;
    double feasibility_factor = 1;

    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][voisinage[index_item]];
        capacity_factor *= (capacity[j] >= weights[j][voisinage[index_item]]) ? 1 : 0;
        feasibility_factor *= (capacity[j] - weights[j][voisinage[index_item]]) / capacity[j];
    }

    return promising_factor * (profit[voisinage[index_item]] / total_weight) * capacity_factor * feasibility_factor;
}
