double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double promising_factor = 0.5;
    double value = 0;

    for (int j = 0; j < dimension; j++) {
        double remaining_capacity_ratio = (capacity[j] - weights[j][voisinage[index_item]]) / capacity[j];
        double weight_ratio = weights[j][voisinage[index_item]] > capacity[j] ? 0 : 1 - weights[j][voisinage[index_item]] / capacity[j];
        value += remaining_capacity_ratio * weight_ratio;
    }

    return promising_factor * value * profit[voisinage[index_item]];
}
