double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit)
{
    double h = 0;
    double h2 = 0;
    double promising_factor = 0.5;

    for (int j = 0; j < dimension; j++) {
        double capacity_ratio = weights[j][voisinage[index_item]] / capacity[j];
        h += capacity_ratio;
        h2 += (1 - capacity_ratio);
    }

    return promising_factor * profit[voisinage[index_item]] * h * h2;
}
