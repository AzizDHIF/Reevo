double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double promising_factor = 0.5;
    double h = 0;
    double h2 = 0;
    double h3 = 0;

    for (int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j];
        h2 += (capacity[j] - weights[j][voisinage[index_item]]) / capacity[j];
        if (weights[j][voisinage[index_item]] > capacity[j]) {
            h3 = 0;
            break;
        }
    }

    return promising_factor * (profit[voisinage[index_item]] / h) * h2 * h3;
}
