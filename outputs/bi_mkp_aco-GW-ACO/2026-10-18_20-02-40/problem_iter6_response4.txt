double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double ratio = 0;

    for (int j = 0; j < dimension; j++) {
        double item_weight = weights[j][voisinage[index_item]];
        if (item_weight > capacity[j]) {
            return 0;
        }
        ratio += item_weight / capacity[j];
    }

    return profit[voisinage[index_item]] / ratio;
}
