double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double promising_factor = 0.5;
    double h = 0;
    
    for (int j = 0; j < dimension; j++) {
        double remaining_capacity_ratio = (capacity[j] != 0) ? weights[j][voisinage[index_item]] / capacity[j] : 0;
        h += remaining_capacity_ratio;
    }
    
    return promising_factor * (profit[voisinage[index_item]] / h);
}
