double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double promising_factor = 0.5;
    double normalized_weight = 0;
    double profit_ratio = 0;
    double feasibility_factor = 0;
    
    for (int j = 0; j < dimension; j++) {
        double remaining_capacity = capacity[j] - weights[j][voisinage[index_item]];
        normalized_weight += weights[j][voisinage[index_item]] / capacity[j];
        feasibility_factor += (remaining_capacity >= 0) ? 1 : 0;
    }
    
    profit_ratio = (feasibility_factor > 0) ? profit[voisinage[index_item]] / normalized_weight : 0;
    
    return promising_factor * profit_ratio * feasibility_factor;
}
