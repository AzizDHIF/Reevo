    double promising_factor = 0.5;
    double h = 0;

    // Check if the item fits within the capacity constraints

        if (weights[j][voisinage[index_item]] > capacity[j]) {
            return 0.0; // Return 0 if the item cannot fit in one dimension
        }
        h += weights[j][voisinage[index_item]] / capacity[j]; // Total weight utilization ratio
    }

    // Include diversification by penalizing already heavily loaded knapsacks
    double diversification_penalty = 1;
    for (int i = 0; i < nb_voisinage; i++) {
        if (i != index_item) {
            double other_h = 0;
            for (int j = 0; j < dimension; j++) {
                other_h += weights[j][voisinage[i]] / capacity[j]; // Calculate utilization ratio for other items
            }
            diversification_penalty *= 1 - other_h;
        }
    }

    // Calculate the promising score based on profit, weight utilization, and diversification
    return promising_factor * (profit[voisinage[index_item]] / h) * diversification_penalty;
}
