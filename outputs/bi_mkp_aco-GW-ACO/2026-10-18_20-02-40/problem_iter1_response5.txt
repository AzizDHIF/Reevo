double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double weight_sum = 0;
    double benefit_sum = 0;
    double weight_penalty = 0;
    
    for(int j=0; j<dimension; j++){
		weight_sum += weights[j][voisinage[index_item]];
		
		// Penalize weights that exceed capacity to make them less promising
		double dim_penalty = fmax(0, weights[j][voisinage[index_item]] - capacity[j]);
		weight_penalty += dim_penalty;
	}
    
	if(weight_penalty > 0){
		return 0;
	}
    
    return benefit_sum + 1/(weight_sum + 1);
}
