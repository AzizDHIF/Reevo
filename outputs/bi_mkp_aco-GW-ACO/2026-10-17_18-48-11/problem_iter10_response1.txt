    double value = 0;

    // Factor 1: ratio of profit to combined weight
    double combined_weight = 0;

    }
    double ratio1 = profit[voisinage[index_item]] / combined_weight;

    // Factor 2: sum of weights ratio to capacity
    double sum_weight_ratio = 0;
    for (int j = 0; j < dimension; j++) {
        sum_weight_ratio += weights[j][voisinage[index_item]] / capacity[j];
    }

    // Sparsifying based on combined factors
    if (ratio1 < 0.5 || sum_weight_ratio > 1.5) {
