double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double value = 0;

    // Factor 1: profit-to-weight ratio
    double ratio1 = profit[voisinage[index_item]] / weights[0][voisinage[index_item]]; // Assuming dimension is 1

    // Factor 2: weight ratio to capacity
    double sum_weight_ratio = weights[0][voisinage[index_item]] / capacity[0]; // Assuming dimension is 1

    // Dynamic threshold adjustment based on the number of neighbors
    double threshold_ratio = nb_voisinage <= 5 ? 0.5 : 0.2;
    double threshold_sum_weight = nb_voisinage <= 5 ? 2 : 5;

    // Sparsifying: Setting unlikely promising elements to zero
    if (ratio1 < threshold_ratio || sum_weight_ratio > threshold_sum_weight) {
        return 0;
    }

    value = 0.7 * ratio1 + 0.3 * sum_weight_ratio;

    return value;
}
