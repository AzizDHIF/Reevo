double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double h=0;
    double threshold = 0.0001; // Smoothing factor to avoid divide by zero
    for(int j=0; j<dimension; j++){
        h += weights[j][voisinage[index_item]] / (capacity[j] + threshold); 
    }
    
    double weight_factor = 1.0;
    for(int j=0; j<dimension; j++){
        weight_factor *= (1 - weights[j][voisinage[index_item]] / (capacity[j] + threshold));
    }

    double profit_factor = profit[voisinage[index_item]] > 0 ? profit[voisinage[index_item]] : 0;
    
    return profit_factor * weight_factor / h;
}
