    double value = 0;

    // Factor 1: ratio of profit to weights
    double ratio1 = profit[voisinage[index_item]] / 1;  // Considering weight dimension of 1 as per the problem description

    // Factor 2: sum of weights ratio to capacity
    double ratio2 = 0;
    for (int j = 0; j < dimension; j++) {
        ratio2 += weights[j][voisinage[index_item]] / capacity[j];
    }

    // Sparsifying: Setting unlikely promising elements to zero dynamically based on nb_voisinage
    double sparsify_threshold1 = nb_voisinage <= 5 ? 0.5 : 0.2;
    double sparsify_threshold2 = nb_voisinage <= 5 ? 2 : 5;

    if (ratio1 < sparsify_threshold1 || ratio2 > sparsify_threshold2) {
