double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double value = 0;

    // Factor 1: ratio of profit to total aggregated weight
    double weight_sum = 0;
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][voisinage[index_item]];
    }
    double ratio1 = profit[voisinage[index_item]] / weight_sum;

    // Factor 2: sum of weights ratio to total aggregated capacity
    double sum_weight_ratio = 0;
    for (int j = 0; j < dimension; j++) {
        sum_weight_ratio += weights[j][voisinage[index_item]] / capacity[j];
    }

    // Dynamic threshold adjustment based on the number of neighbors
    double threshold_ratio = nb_voisinage <= 5 ? 0.5 : 0.2;
    double threshold_sum_weight = nb_voisinage <= 5 ? 2 : 5;

    // Sparsifying: Setting unlikely promising elements to zero
    if (ratio1 < threshold_ratio || sum_weight_ratio > threshold_sum_weight) {
        return 0;
    }

    value = 0.7 * ratio1 + 0.3 * sum_weight_ratio;

    return value;
}
