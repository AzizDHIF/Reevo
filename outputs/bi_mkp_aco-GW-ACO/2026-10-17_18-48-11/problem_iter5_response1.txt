double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double value = 0;
    
    // Factor 1: ratio of profit to weight
    double weight_sum = 0;
    for (int j = 0; j < dimension; j++) {
        weight_sum += weights[j][voisinage[index_item]];
    }
    double ratio1 = profit[voisinage[index_item]] / weight_sum;
    
    // Factor 2: maximum weight ratio to capacity
    double max_weight_ratio = 0;
    for (int j = 0; j < dimension; j++) {
        double weight_ratio = weights[j][voisinage[index_item]] / capacity[j];
        if (weight_ratio > max_weight_ratio) {
            max_weight_ratio = weight_ratio;
        }
    }
    
    // Sparsifying: Setting unlikely promising elements to zero
    if (nb_voisinage <= 5) {
        if (ratio1 < 0.5 || max_weight_ratio > 1) {
            return 0;
        }
    } else {
        if (ratio1 < 0.3 || max_weight_ratio > 2) {
            return 0;
        }
    }
    
    value = 0.6 * ratio1 + 0.4 * max_weight_ratio;
    
    return value;
}
