double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profits){
    double h = 0;
    double total_weight_ratio = 0;
    double profit_ratio = 0;
	
    for(int j = 0; j < dimension; j++) {
        h += weights[j][voisinage[index_item]] / capacity[j]; 
        total_weight_ratio += (capacity[j] - weights[j][voisinage[index_item]]) / capacity[j];
    }
	
    profit_ratio = profits[voisinage[index_item]] / (1 + h);

    if(total_weight_ratio < 0.5) {
        return profit_ratio;
    } else {
        return 0;
    }
}
