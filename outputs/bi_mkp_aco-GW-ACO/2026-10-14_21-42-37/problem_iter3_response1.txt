double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weight_contribution = 0;
    for (int j = 0; j < dimension; j++) {
        weight_contribution += weights[j][voisinage[index_item]] / capacity[j];
    }

    if (weight_contribution == 0) {
        return 0;
    }

    return profit[voisinage[index_item]] / weight_contribution;
}
