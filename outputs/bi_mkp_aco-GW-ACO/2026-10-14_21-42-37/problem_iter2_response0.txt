double heuristic_v2(int index_item, double **weights, double *capacity, double *profit)
{
    double weight_factor = 0.5;
    double profit_factor = 0.5;
    double h = 0;

    for (int j = 0; j < dimension; j++) {
        h += weight_factor * weights[j][index_item] * (1 / capacity[j]);
    }

    return (profit[index_item] + h) / (1 + h);
}
