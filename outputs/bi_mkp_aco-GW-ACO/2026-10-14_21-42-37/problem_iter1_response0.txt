double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){
    double h=0;
    double prom_factor1 = 0.7;
    double prom_factor2 = 0.3;

    for(int j=0;j<dimension;j++){
        h = h + weights[j][voisinage[index_item]] / capacity[j]; 
    }

    double promising = prof[voisinage[index_item]] / h;
  
    // Sparsifying the heuristic
    double threshold = 0.5;
    if(promising < threshold){
        return 0.0;
    }

    return prom_factor1 * promising + prom_factor2;
}
