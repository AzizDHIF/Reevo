double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double total_weight_ratio = 0;
    double total_profit = 0;

    for (int j = 0; j < dimension; j++) {
        total_weight_ratio += weights[j][voisinage[index_item]] / capacity[j];
    }

    if (total_weight_ratio == 0) {
        return 0; // Avoid division by zero
    }

    total_profit = profit[voisinage[index_item]];
    return total_profit / total_weight_ratio;
}
