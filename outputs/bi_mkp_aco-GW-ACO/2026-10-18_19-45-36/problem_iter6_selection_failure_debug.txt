    double weighted_sum = 0.0;
    
    for(int j = 0; j < dimension; j++) {
        double weight_ratio = weights[j][voisinage[index_item]] / capacity[j];
        weighted_sum += weight_ratio;
        
        // Sparsify zero-weight ratio elements
        if (weight_ratio == 0.0) {
            return 0.0;
        }
    }
    
    return profit[voisinage[index_item]] / weighted_sum;
}
