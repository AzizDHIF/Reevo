double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double weight_factor = 0.7;
    double profit_factor = 0.3;
    double total_weight = 0.0;

    for (int j = 0; j < dimension; j++) {
        total_weight += weights[j][voisinage[index_item]] / capacity[j];
    }

    if (total_weight == 0.0 || profit[voisinage[index_item]] == 0.0) {
        return 0.0;
    }

    return (weight_factor * total_weight + profit_factor * profit[voisinage[index_item]]) / total_weight;
}
