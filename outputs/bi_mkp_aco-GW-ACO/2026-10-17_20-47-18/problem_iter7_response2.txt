double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 1.0;    // Initialize h as absolute value for normalization
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            h *= weights[j][voisinage[index_item]] / capacity[j]; // Combine weights and capacity effectively
        }
    }
    if (h < 1e-6) { // Early termination if h is too small
        return 0.0;
    }
    return profit[voisinage[index_item]] * h; // Emphasize objective-specific normalization
}
