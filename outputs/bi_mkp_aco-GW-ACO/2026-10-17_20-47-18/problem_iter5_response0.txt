double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 1e-6; // Initialize h with a small value to avoid division by zero
    int max_profit_idx = -1;
    double max_profit = 0;

    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            h += weights[j][voisinage[index_item]] * (1.0 / capacity[j]); // Combine weight and capacity directly
        }
    }

    for (int i = 0; i < nb_voisinage; i++) {
        int neighbor = voisinage[i];
        if (profit[neighbor] > max_profit) {
            max_profit = profit[neighbor];
            max_profit_idx = neighbor;
        }
    }

    // Normalize the profit with the highest profit among neighbors
    if (max_profit > 0) {
        return profit[voisinage[index_item]] / max_profit / h;
    } else {
        return 0.0;
    }
}
