double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double alpha = 0.5; // Adjust weight impact strategically
    double h = 1e-6; // Initialize h with a small value to avoid division by zero
    double total_weight = 0.0;

    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            h += weights[j][voisinage[index_item]] * (1.0 / capacity[j]); // Combine weight and capacity directly
            total_weight += weights[j][voisinage[index_item]]; // Calculate total weight separately
        }
    }

    double capacity_factor = h > 1.0 ? 1.0 / h : 1.0; // Normalize capacity factor
    double weight_factor = (total_weight > 0) ? alpha * alpha / total_weight : 0.0; // Weight factor based on total weight

    double normalized_profit = profit[voisinage[index_item]] * capacity_factor * (1.0 - weight_factor); // Combine factors strategically

    return normalized_profit;
}
