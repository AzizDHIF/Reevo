double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double eps = 1e-6; // Small value to avoid division by zero
    double h = eps; // Initialize h with eps to avoid zero division

    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            h += weights[j][voisinage[index_item]] / capacity[j]; // Combine weight and capacity directly
        }
    }

    return profit[voisinage[index_item]] / h;
}
