double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit){

    double alpha = 0.5;
    double h = 1e-6;
    double promising_factor = 0.0;

    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            h += weights[j][voisinage[index_item]] * (1.0 / capacity[j]);
        }
    }

    double normalized_profit = profit[voisinage[index_item]] / h;

    promising_factor = 1.0 - alpha * alpha * alpha * weights[0][voisinage[index_item]];

    normalized_profit *= promising_factor;

    return normalized_profit;
}
