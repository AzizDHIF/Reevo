double heuristic_v2(int index_item, double **weights, double *capacity, int nb_voisinage, int *voisinage, double *profit) {
    double h = 1e-6; // Initialize h with a small value to avoid division by zero
    for (int j = 0; j < dimension; j++) {
        if (capacity[j] > 0) {
            h += weights[j][voisinage[index_item]] * (1.0 / capacity[j]); // Combine weight and capacity directly
        }
    }

    double normalized_profit = profit[voisinage[index_item]] / h;

    // Objective-specific normalization for MO-MKP
    normalized_profit *= (1.0 - 0.5 * weights[0][voisinage[index_item]]); // Prioritize scarce dimension

    return normalized_profit;
}
