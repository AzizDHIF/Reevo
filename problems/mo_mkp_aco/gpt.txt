#include "HBACO.h"
double heuristic_v2(int index_item, double weights[/*dimension*/][NBITEMS], double capacity[/*dimension*/], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    // Simplify calculations by directly checking if item's weights fit within capacity for all dimensions
    double total_weight = 0.0;
    double total_capacity = 0.0;
    for (int j = 0; j < /*dimension*/; j++) {
        total_weight += weights[j][index_item];
        total_capacity += capacity[j];
    }

    if (total_weight == 0.0 || total_capacity == 0.0) {
        return 1e-9;
    }

    // Calculate how well the item fits into the knapsack
    double fit = total_weight / total_capacity;

    // Calculate a pressure factor based on how full the knapsacks are
    double pressure = total_weight / total_capacity;

    // Combine the item's profit with its fit and pressure to get the heuristic value
    double neighborhood_conflict = 0.0;
    for (int i = 0; i < nb_voisinage; i++) {
        int item = voisinage[i];
        if (item == index_item) continue;
        double competing_weight = 0.0;
        for (int j = 0; j < /*dimension*/; j++) {
            competing_weight += weights[j][item];
        }
        if (competing_weight > 0.0) {
            neighborhood_conflict += competing_weight / total_capacity;
        }
    }

    double final_heuristic = profit[index_item] * (1.0 - fit) * (1.0 - neighborhood_conflict);
    return final_heuristic > 0 ? final_heuristic : 1e-9;
}
