#include "HBACO.h"
double heuristic_v2(int index_item, double weights[/*dimension*/][NBITEMS], double capacity[/*dimension*/], int nb_voisinage, int voisinage[NBITEMS], double profit[NBITEMS]) {
    double h = 0;
    double sum_weights = 0;
    double density = 0;
    for (int j = 0; j < /*dimension*/; j++) {
        sum_weights += weights[j][index_item];
    }
    density = profit[index_item] / sum_weights;

    for (int i = 0; i < nb_voisinage; i++) {
        double sum_weights_voisinage = 0;
        for (int j = 0; j < /*dimension*/; j++) {
            h += weights[j][voisinage[i]] / capacity[j];
            sum_weights_voisinage += weights[j][voisinage[i]];
        }
        double density_voisinage = profit[voisinage[i]] / sum_weights_voisinage;
        if (density_voisinage < density) {
            h -= 2 * density_voisinage; // discourage neighbors that are less promising
        }
    }

    double avg_profit = 0;
    for (int i = 0; i < nb_voisinage; i++) {
        avg_profit += profit[voisinage[i]];
    }
    avg_profit /= nb_voisinage;
    if (profit[index_item] < avg_profit) {
        h -= avg_profit; // discourage items that are less profitable than average neighbors
    }

    // sparsify the heuristic
    if (profit[index_item] < 0.5 * avg_profit) {
        h = 0; // set unpromising elements to zero
    }

    return profit[index_item] / (h + 1e-6); // add small value to avoid division by zero
}
